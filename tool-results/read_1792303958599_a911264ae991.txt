   889→   889→// Forecast model (screen 81) — derived from canonical targets
   890→   890→// + opportunities. Actual / Forecast / Target never mixed.
   891→   891→// ============================================================
   892→   892→
   893→   893→export function computeForecast(rows: TargetRow[]) {
   894→   894→  const employees = rows.filter((r) => r.kind === "employee");
   895→   895→  const target = employees.reduce((s, r) => s + r.quotaCr, 0);
   896→   896→  const closed = employees.reduce((s, r) => s + r.achievedCr, 0);
   897→   897→  const weighted = employees.reduce((s, r) => s + r.pipelineCr * 0.55, 0); // blended demo weight
   898→   898→  const commit = closed + employees.reduce((s, r) => s + (r.forecastCr - r.achievedCr) * 0.6, 0);
   899→   899→  const bestCase = closed + employees.reduce((s, r) => s + (r.pipelineCr), 0);
   900→   900→  return { target, closed, weighted, commit, bestCase, gap: target - commit };
   901→   901→}
