    98→  const history = s.approvals.filter((a) => a.status !== "Pending");
    99→  return { queue, history, note: null };

   144→export function decideApproval(
   145→  role: WorkspaceRole,
   146→  input: { id: string; decision: "Approved" | "Rejected" | "Changes requested"; note?: string; stepUpPassword?: string },
   147→) {
   148→  const s = store();
   149→  const req = s.approvals.find((a) => a.id === input.id);
   150→  if (!req) return { ok: false as const, status: 404, code: "not_found", message: "Approval request not found." };
   151→  if (req.status !== "Pending") {
   152→    return { ok: false as const, status: 409, code: "already_decided", message: `This request was already ${req.status.toLowerCase()} by another approver.` };
   153→  }
   154→
   155→  // Server-side authorization: employee boundaries hold no approval authority.
   156→  if (role === "employee" || role === "sales-exec") {
   157→    writeAudit({ actorId: SESSION_USER_ID, action: "Approval decision denied", resource: req.title, result: "Blocked", severity: "Warning", context: "Role holds no approval authority for this request type" });
   158→    return { ok: false as const, status: 403, code: "forbidden", message: "Your role does not hold approval authority for this request." };
   159→  }
   160→
   161→  // Step-up authentication for sensitive approvals (server-enforced).
   162→  if (req.stepUpRequired) {
   163→    if (!input.stepUpPassword || input.stepUpPassword.length < 8) {
   164→      return { ok: false as const, status: 401, code: "step_up_required", message: "Reauthentication is required for this approval." };
   165→    }
   166→  }
   167→
   168→  // High-value decisions require a note for the audit trail.
   169→  if (input.decision === "Rejected" && (!input.note || input.note.trim().length < 4)) {
   170→    return { ok: false as const, status: 422, code: "note_required", message: "A reason is required when rejecting a request." };
   171→  }
   172→
   173→  req.status = input.decision;
   174→  req.decidedAt = Date.now();
   175→  req.decidedBy = SESSION_USER_ID;
   176→  req.decisionNote = input.note?.trim() || undefined;
   177→  req.history = [
   178→    ...(req.history ?? []),
   179→    { at: Date.now(), actor: "Sagar Singh", event: input.decision + (input.note ? ` — ${input.note.trim()}` : "") },
   180→  ];
   181→
   182→  writeAudit({
   183→    actorId: SESSION_USER_ID,
   184→    action: `Approval ${input.decision.toLowerCase()}`,
   185→    resource: `${req.kind} · ${req.title}`,
   186→    result: "Success",
   187→    severity: req.stepUpRequired ? "High" : "Info",
   188→    source: "Web · Mumbai",
   189→    context: req.stepUpRequired
   190→      ? "Step-up authentication verified before the decision"
   191→      : `Routed via Approval Matrix (${req.rule})`,
   192→  });
   193→
   194→  return { ok: true as const, approval: req };
   195→}

   205→export function createTask(role: WorkspaceRole, input: { title: string; detail?: string; priority?: Task["priority"]; due?: number | null }) {
   206→  const title = input.title?.trim() ?? "";
   207→  if (title.length < 4) return { ok: false as const, status: 422, code: "invalid_title", message: "Task title must be at least 4 characters." };
   208→  if (title.length > 140) return { ok: false as const, status: 422, code: "invalid_title", message: "Task title must be 140 characters or fewer." };
   209→  if (input.due !== null && input.due !== undefined && Number.isNaN(input.due)) {
   210→    return { ok: false as const, status: 422, code: "invalid_due", message: "Due date is invalid." };
   211→  }
   212→
   213→  const t: Task = {
   214→    id: `task-${Math.floor(Date.now() / 1000) % 100000}`,
   215→    title,
   216→    detail: input.detail?.trim() || "",
   217→    status: "Not Started",
   218→    priority: input.priority ?? "Medium",
   219→    due: input.due ?? null,
   220→    source: "Manual",
   221→    ownerId: SESSION_USER_ID,
   222→    assigneeId: SESSION_USER_ID,
   223→  };
   224→  store().tasks.unshift(t);
   225→  writeAudit({ actorId: SESSION_USER_ID, action: "Task created", resource: t.title, result: "Success", context: "Created from Employee Home quick action" });
   226→  return { ok: true as const, task: t };
   227→}

   229→const TASK_TRANSITIONS: Record<string, Task["status"][]> = {
   230→  "Not Started": ["In Progress", "Cancelled"],
   231→  "In Progress": ["Waiting", "Blocked", "Completed", "Cancelled"],
   232→  Blocked: ["In Progress", "Waiting", "Cancelled"],
   233→  Waiting: ["In Progress", "Blocked", "Cancelled"],
   234→  Completed: ["In Progress"],

   238→export function updateTask(role: WorkspaceRole, input: { id: string; status: Task["status"] }) {
   239→  const t = store().tasks.find((x) => x.id === input.id);
   240→  if (!t) return { ok: false as const, status: 404, code: "not_found", message: "Task not found." };
   241→  // Authorization: only the assignee transitions their own task.
   242→  if (t.assigneeId !== SESSION_USER_ID) {
   243→    writeAudit({ actorId: SESSION_USER_ID, action: "Task update denied", resource: t.title, result: "Blocked", severity: "Warning", context: "Caller is not the task assignee" });
   244→    return { ok: false as const, status: 403, code: "forbidden", message: "Only the task assignee can update this task." };
   245→  }
   246→  const allowed = TASK_TRANSITIONS[t.status] ?? [];
   247→  if (!allowed.includes(input.status)) {
   248→    return { ok: false as const, status: 409, code: "invalid_transition", message: `A task cannot move from ${t.status} to ${input.status}.` };
   249→  }
   250→  t.status = input.status;
   251→  if (input.status === "Completed") t.completedAt = Date.now();
   252→  writeAudit({ actorId: SESSION_USER_ID, action: "Task status changed", resource: t.title, result: "Success", context: `${t.status === "Completed" ? "Marked completed" : `Moved to ${input.status}`}` });
   253→  return { ok: true as const, task: t };
   254→}

   257→// Mail — search/sending respects the session mailbox scope.
   258→// The search index can never become an authorization bypass:

   262→export function listMail(role: WorkspaceRole, params: { folder?: string; q?: string; unreadOnly?: boolean; label?: string; limit?: number }) {
   263→  const s = store();
   264→  const folder = params.folder ?? "inbox";
   265→  let items = folder === "all" ? [...s.mail] : s.mail.filter((m) => m.folder === folder);
   266→  // Authorization scope: the session mailbox only (demo: single-user store).
   267→  const q = params.q?.trim().toLowerCase();
   268→  if (q) {
   269→    items = items.filter((m) => {
   270→      const fromName = m.fromId ? employees.find((e) => e.id === m.fromId)?.name ?? "" : m.fromExternal?.name ?? "";
   271→      const fromEmail = m.fromId ? employees.find((e) => e.id === m.fromId)?.email ?? "" : m.fromExternal?.email ?? "";
   272→      return [m.subject, m.preview, m.body, fromName, fromEmail, ...(m.to ?? [])].join(" ").toLowerCase().includes(q);
   273→    });
   274→  }
   275→  if (params.unreadOnly) items = items.filter((m) => !m.read);
   276→  if (params.label) items = items.filter((m) => m.labels?.includes(params.label as never));
   277→  items.sort((a, b) => b.ts - a.ts);
   278→  const limit = params.limit && params.limit > 0 ? Math.min(params.limit, 200) : 100;
   279→  return { items: items.slice(0, limit), total: items.length, scopeNote: "Search is scoped to your authorized mailbox server-side." };
   280→}

   282→export function sendMail(role: WorkspaceRole, input: { to: string[]; cc?: string[]; bcc?: string[]; subject: string; body: string; attachments?: { name: string; sizeKb: number; type: string }[] }) {
   283→  const to = (input.to ?? []).map((v) => v.trim()).filter(Boolean);
   284→  const cc = (input.cc ?? []).map((v) => v.trim()).filter(Boolean);
   285→  if (to.length === 0) return { ok: false as const, status: 422, code: "no_recipients", message: "Add at least one recipient." };
   286→  if (to.some((v) => !emailLike(v))) return { ok: false as const, status: 422, code: "invalid_recipient", message: "One or more recipient addresses are not valid." };
   287→  if (!input.subject?.trim() && !input.body?.trim()) {
   288→    return { ok: false as const, status: 422, code: "empty_message", message: "Add a subject or a message before sending." };
   289→  }
   290→  const atts = input.attachments ?? [];
   291→  if (atts.length > MAX_ATTACHMENTS) return { ok: false as const, status: 422, code: "too_many_attachments", message: `A message can carry at most ${MAX_ATTACHMENTS} attachments.` };
   292→  const totalKb = atts.reduce((n, a) => n + (a.sizeKb || 0), 0);
   293→  if (totalKb > MAX_TOTAL_KB) return { ok: false as const, status: 422, code: "attachment_too_large", message: "Attachments exceed the 10 MB message limit." };
   294→
   295→  const msg: MailMessage = {
   296→    id: `s-${Math.floor(Date.now() / 1000) % 100000}`,
   297→    folder: "sent",
   298→    fromId: SESSION_USER_ID,
   299→    to,
   300→    cc: cc.length ? cc : undefined,
   301→    subject: input.subject?.trim() || "(no subject)",
   302→    preview: (input.body ?? "").replace(/\s+/g, " ").trim().slice(0, 140),
   303→    body: input.body ?? "",
   304→    ts: Date.now(),
   305→    read: true,
   306→    sendStatus: "Sent",
   307→    attachments: atts.length
   308→      ? atts.map((a) => ({ name: a.name, sizeKb: a.sizeKb, type: (a.type?.toUpperCase() || "BIN") as never }))
   309→      : undefined,
   310→  };
   311→  store().mail.unshift(msg);
   312→  writeAudit({ actorId: SESSION_USER_ID, action: "Mail sent", resource: msg.subject, result: "Success", context: `${to.length} recipient(s)${cc.length ? `, ${cc.length} cc` : ""}${atts.length ? `, ${atts.length} attachment(s)` : ""}` });
   313→  return { ok: true as const, message: msg };
   314→}

   316→export function saveDraft(role: WorkspaceRole, input: Partial<MailMessage> & { draftId?: string }) {
   317→  const s = store();

   322→        to: input.to ?? existing.to,
   323→        cc: input.cc ?? existing.cc,

   347→export function retryMail(role: WorkspaceRole, id: string) {
   348→  const m = store().mail.find((x) => x.id === id);
   349→  if (!m || m.folder !== "sent") return { ok: false as const, status: 404, code: "not_found", message: "Message not found in Sent." };
   350→  if (m.sendStatus !== "Failed" && m.sendStatus !== "Pending") {
   351→    return { ok: false as const, status: 409, code: "not_retryable", message: `This message is already ${(m.sendStatus ?? "sent").toLowerCase()}.` };
   352→  }
   353→  m.sendStatus = "Sent";
   354→  m.sendDetail = undefined;
   355→  writeAudit({ actorId: SESSION_USER_ID, action: "Mail retry succeeded", resource: m.subject, result: "Success", context: "Delivery retried after failure" });
   356→  return { ok: true as const, message: m };
   357→}

   374→const SENSITIVE_DOC_ACTIONS = new Set(["download", "share"]);
   375→
   376→export function documentAction(
   377→  role: WorkspaceRole,
   378→  input: { action: "download" | "share" | "rename" | "move" | "upload"; docId?: string; name?: string; category?: string; toEmployeeId?: string; fileMeta?: { name: string; sizeKb: number; type: string } },
   379→) {
   380→  const s = store();
   381→
   382→  if (input.action === "upload") {
   383→    const f = input.fileMeta;
   384→    if (!f || !f.name?.trim()) return { ok: false as const, status: 422, code: "invalid_file", message: "Choose a file to upload." };
   385→    if (f.sizeKb > MAX_TOTAL_KB) return { ok: false as const, status: 422, code: "file_too_large", message: "Files above 10 MB must go through the enterprise storage vault." };
   386→    const doc: WorkspaceDoc = {
   387→      id: `doc-${Math.floor(Date.now() / 1000) % 100000}`,
   388→      name: f.name,
   389→      type: (f.type?.toUpperCase() || "DOCX") as WorkspaceDoc["type"],
   390→      category: "Personal",
   391→      ownerId: SESSION_USER_ID,
   392→      sizeKb: f.sizeKb,
   393→      modified: Date.now(),
   394→      sensitivity: "Internal",
   395→      description: "Uploaded from My Documents. File content remains in your enterprise storage; Qubit One holds the record.",
   396→    };
   397→    s.documents.unshift(doc);
   398→    writeAudit({ actorId: SESSION_USER_ID, action: "Document uploaded", resource: doc.name, result: "Success", context: "My Documents · metadata record only" });
   399→    return { ok: true as const, doc };
   400→  }
   401→
   402→  const doc = s.documents.find((d) => d.id === input.docId);
   403→  if (!doc) return { ok: false as const, status: 404, code: "not_found", message: "Document not found." };
   404→
   405→  if (SENSITIVE_DOC_ACTIONS.has(input.action) && doc.sensitivity === "Restricted") {
   406→    writeAudit({ actorId: SESSION_USER_ID, action: `Document ${input.action} denied`, resource: doc.name, result: "Blocked", severity: "Warning", context: "Restricted document — HR-managed access only" });
   407→    return { ok: false as const, status: 403, code: "restricted", message: "This document is restricted. Access is managed by HR — request access from the document owner." };
   408→  }
   409→
   410→  if (input.action === "download") {
   411→    writeAudit({ actorId: SESSION_USER_ID, action: "Document downloaded", resource: doc.name, result: "Success", context: `Sensitivity: ${doc.sensitivity}` });
   412→    return { ok: true as const, doc };
   413→  }
   414→
   415→  if (input.action === "share") {
   416→    if (!input.toEmployeeId || !employees.some((e) => e.id === input.toEmployeeId)) {
   417→      return { ok: false as const, status: 422, code: "invalid_recipient", message: "Choose a colleague to share with." };
   418→    }
   419→    if (doc.sensitivity === "Confidential" && role === "employee") {
   420→      return { ok: false as const, status: 403, code: "forbidden", message: "Confidential documents cannot be shared from this role." };
   421→    }
   422→    writeAudit({ actorId: SESSION_USER_ID, action: "Document shared", resource: doc.name, result: "Success", context: `Shared with ${employees.find((e) => e.id === input.toEmployeeId)?.name}` });
   423→    return { ok: true as const, doc };
   424→  }
   425→
   426→  if (doc.ownerId !== SESSION_USER_ID) {
   427→    return { ok: false as const, status: 403, code: "forbidden", message: "Only the owner can rename or move this document." };
   428→  }
   429→  if (input.action === "rename") {
   430→    const name = input.name?.trim();
   431→    if (!name || name.length < 3) return { ok: false as const, status: 422, code: "invalid_name", message: "Enter a document name (3+ characters)." };
   432→    doc.name = name;
   433→  } else if (input.action === "move") {
   434→    const cat = input.category as WorkspaceDoc["category"];
   435→    if (!["Personal", "Shared with me", "HR", "Company", "Workflow"].includes(cat)) {
   436→      return { ok: false as const, status: 422, code: "invalid_category", message: "Choose a valid category." };
   437→    }
   438→    doc.category = cat;
   439→  }
   440→  return { ok: true as const, doc };
   441→}

   450→const EDITABLE_PROFILE_FIELDS = new Set(["phone", "about", "skills"]);
   451→
   452→export function updateProfile(role: WorkspaceRole, patch: Record<string, unknown>) {
   453→  const keys = Object.keys(patch);
   454→  const illegal = keys.filter((k) => !EDITABLE_PROFILE_FIELDS.has(k));
   455→  if (illegal.length > 0) {
   456→    writeAudit({ actorId: SESSION_USER_ID, action: "Profile update denied", resource: illegal.join(", "), result: "Blocked", severity: "Warning", context: "Attempted edit of company-controlled fields" });
   457→    return { ok: false as const, status: 403, code: "company_controlled", message: `These fields are managed by authorized HR workflows: ${illegal.join(", ")}.` };
   458→  }
   459→  if (patch.phone !== undefined) {
   460→    const phone = String(patch.phone).trim();
   461→    if (!/^\+91\s\d{5}\s\d{5}$/.test(phone)) {
   462→      return { ok: false as const, status: 422, code: "invalid_phone", message: "Use the format +91 98200 12345." };
   463→    }
   464→  }
   465→  if (patch.about !== undefined && String(patch.about).length > 280) {
   466→    return { ok: false as const, status: 422, code: "invalid_about", message: "About must be 280 characters or fewer." };
   467→  }
   468→  if (patch.skills !== undefined && !Array.isArray(patch.skills)) {
   469→    return { ok: false as const, status: 422, code: "invalid_skills", message: "Skills must be a list." };
   470→  }
   471→  writeAudit({ actorId: SESSION_USER_ID, action: "Profile updated", resource: keys.join(", "), result: "Success", context: "Self-service fields only" });
   472→  return { ok: true as const, updated: keys };
   473→}

   476→// Meetings (Screens 44 · 45) — the meetings/calendar domain is
   477→// served from this store like every other workspace domain
   478→// (Phase 19 §6/§12: no enterprise fixture values imported into
   479→// client components).
   480→//
   481→// Scope model: "My Meetings" is a personal workspace surface —
   482→// exactly like listTasks — so the list is scoped to the session
   483→// identity by participation (organizer or invited participant)
   484→// for every role preview. Roles with governance authority
   485→// (administrator / dept-head) may CANCEL meetings they do not
   486→// organize; that permission is enforced here, never in the UI.
   487→//
   488→// Minutes flow: notes and action items are recorded AGAINST the
   489→// store's session maps (meetingNotes / meetingActions, keyed by
   490→// meeting id — mirroring mobile-server's meetingNotes/meetingActions
   491→// concept) for COMPLETED meetings, and only by the organizer.
   492→// materializeMeeting() merges the fixture record with those session
   493→// entries at read time, so every workspace reader — list, ?id=
   494→// detail, cancel response and the calendar drawer — sees the same
   495→// state without the store ever mutating the fixture-shaped record.
   496→// Mobile overlap (honest note): the Phase 18 mobile surface (screen
   497→// 289, mobile-server.ts) keeps its own session maps for its field
   498→// flow; the mobile-* files are out of scope here, so the workspace
   499→// store is the single authority for screens 44/45 and for
   500→// /api/workspace/meetings.
   501→// ============================================================
   502→
   503→/** Fixture record + session minutes merged into the wire shape. */
   504→function materializeMeeting(s: WorkspaceStore, m: Meeting): Meeting {
   505→  const notes = s.meetingNotes.get(m.id) ?? [];
   506→  const actions = s.meetingActions.get(m.id) ?? [];
   507→  if (notes.length === 0 && actions.length === 0) return m;
   508→  return {
   509→    ...m,
   510→    notes: notes.length ? [m.notes, ...notes.map((n) => n.text)].filter(Boolean).join("\n\n") : m.notes,
   511→    actionItems: actions.length ? [...(m.actionItems ?? []), ...actions.map((a) => ({ title: a.title, ownerId: a.ownerId, due: a.due }))] : m.actionItems,
   512→  };
   513→}

   526→export function listMeetings(role: WorkspaceRole): MeetingsScopePayload {
   527→  const s = store();

   529→  // (Same boundary as listTasks; meetings are personal workspace data.)

   531→    .filter((m) => m.organizerId === SESSION_USER_ID || m.participantIds.includes(SESSION_USER_ID))

   540→    events: [...s.events],
   541→    scope: "own",
   542→    note: scoped
   543→      ? "Scoped to your own participation — you see the meetings you organize or attend. Cancellation follows organizer permissions and is enforced server-side."
   544→      : null,
   545→    asOf: WORKSPACE_NOW_LABEL,
   546→    now: WORKSPACE_NOW,

   563→export function cancelMeeting(role: WorkspaceRole, input: { id: string; reason: string }) {
   564→  const s = store();
   565→  const m = s.meetings.find((x) => x.id === input.id);
   566→  if (!m) return { ok: false as const, status: 404, code: "not_found", message: "Meeting not found." };
   567→  if (m.status !== "Scheduled") {
   568→    return {
   569→      ok: false as const,
   570→      status: 409,
   571→      code: "invalid_state",
   572→      message: m.status === "Cancelled" ? "This meeting was already cancelled." : `A meeting that is ${m.status.toLowerCase()} cannot be cancelled.`,
   573→    };
   574→  }
   575→  const isOrganizer = m.organizerId === SESSION_USER_ID;
   576→  const governance = role === "administrator" || role === "dept-head";
   577→  if (!isOrganizer && !governance) {
   578→    writeAudit({ actorId: SESSION_USER_ID, action: "Meeting cancellation denied", resource: `${m.title} (${m.id})`, result: "Blocked", severity: "Warning", source: "Workspace", context: "Only the organizer — or an administrator / department head — can cancel a meeting" });
   579→    return { ok: false as const, status: 403, code: "forbidden", message: "Only the meeting organizer — or an administrator / department head — can cancel this meeting." };
   580→  }
   581→  const reason = input.reason?.trim() ?? "";
   582→  if (reason.length < 4) {
   583→    return { ok: false as const, status: 422, code: "reason_required", message: "A short reason is required when cancelling a meeting." };
   584→  }
   585→  m.status = "Cancelled";
   586→  writeAudit({
   587→    actorId: SESSION_USER_ID,
   588→    action: "Meeting cancelled",
   589→    resource: `${m.id} — ${m.title}`,
   590→    result: "Success",
   591→    severity: "Info",
   592→    source: "Workspace",
   593→    context: `${isOrganizer ? "Organizer" : "Governance"} cancellation — reason: ${reason}`,
   594→  });
   595→  return {
   596→    ok: true as const,
   597→    meeting: materializeMeeting(s, m),
   598→    message: `"${m.title}" was cancelled — ${m.participantIds.length} participant${m.participantIds.length === 1 ? "" : "s"} notified and the calendar updated.`,
   599→  };
   600→}

   602→// Notes and action items (the COMPLETED-meeting minutes flow) are recorded
   603→// into the store's session maps keyed by meeting id — the workspace-store
   604→// mirror of mobile-server's meetingNotes/meetingActions concept. Only the
   605→// organizer records minutes; the fixture-shaped record is never mutated.
   606→// Every read path materializes the map entries onto the meeting (see
   607→// materializeMeeting) so list, ?id= detail and the calendar drawer agree.
   608→export function saveMeetingNotes(role: WorkspaceRole, input: { id: string; text: string }) {
   609→  const s = store();
   610→  const m = s.meetings.find((x) => x.id === input.id);
   611→  if (!m) return { ok: false as const, status: 404, code: "not_found", message: "Meeting not found." };
   612→  if (m.organizerId !== SESSION_USER_ID) {
   613→    writeAudit({ actorId: SESSION_USER_ID, action: "Meeting note denied", resource: `${m.id} — ${m.title}`, result: "Blocked", severity: "Warning", source: "Workspace", context: "Only the organizer can record minutes" });
   614→    return { ok: false as const, status: 403, code: "forbidden", message: "Only the meeting organizer can record notes for this meeting." };
   615→  }
   616→  if (m.status !== "Completed") {
   617→    return { ok: false as const, status: 409, code: "invalid_state", message: `Minutes are recorded on completed meetings — this one is ${m.status.toLowerCase()}.` };
   618→  }
   619→  const text = input.text?.trim() ?? "";
   620→  if (text.length < 2) return { ok: false as const, status: 422, code: "invalid_note", message: "A note of at least 2 characters is required." };
   621→  const entry = { at: Date.now(), by: SESSION_USER_ID, text };
   622→  const existing = s.meetingNotes.get(m.id) ?? [];
   623→  s.meetingNotes.set(m.id, [...existing, entry]);
   624→  writeAudit({ actorId: SESSION_USER_ID, action: "Meeting notes saved", resource: `${m.id} — ${m.title}`, result: "Success", source: "Workspace", context: "Note appended to the meeting record" });
   625→  return { ok: true as const, meeting: materializeMeeting(s, m), message: "Note added to the meeting record." };
   626→}

   628→export function addMeetingAction(role: WorkspaceRole, input: { id: string; title: string; ownerId?: string; due?: number | null }) {
   629→  const s = store();
   630→  const m = s.meetings.find((x) => x.id === input.id);
   631→  if (!m) return { ok: false as const, status: 404, code: "not_found", message: "Meeting not found." };
   632→  if (m.organizerId !== SESSION_USER_ID) {
   633→    writeAudit({ actorId: SESSION_USER_ID, action: "Meeting action item denied", resource: `${m.id} — ${m.title}`, result: "Blocked", severity: "Warning", source: "Workspace", context: "Only the organizer can record action items" });
   634→    return { ok: false as const, status: 403, code: "forbidden", message: "Only the meeting organizer can add action items." };
   635→  }
   636→  if (m.status !== "Completed") {
   637→    return { ok: false as const, status: 409, code: "invalid_state", message: `Action items are recorded on completed meetings — this one is ${m.status.toLowerCase()}.` };
   638→  }
   639→  const title = input.title?.trim() ?? "";
   640→  if (title.length < 4) return { ok: false as const, status: 422, code: "invalid_title", message: "Describe the action item (4+ characters)." };
   641→  const ownerId = input.ownerId ?? SESSION_USER_ID;
   642→  if (!employees.some((e) => e.id === ownerId)) {
   643→    return { ok: false as const, status: 422, code: "invalid_owner", message: "Choose a valid owner." };
   644→  }
   645→  if (input.due !== null && input.due !== undefined && Number.isNaN(input.due)) {
   646→    return { ok: false as const, status: 422, code: "invalid_due", message: "Due date is invalid." };
   647→  }
   648→  const entry = { title, ownerId, due: input.due ?? null, at: Date.now() };
   649→  const existing = s.meetingActions.get(m.id) ?? [];
   650→  s.meetingActions.set(m.id, [...existing, entry]);
   651→  writeAudit({ actorId: SESSION_USER_ID, action: "Meeting action item added", resource: `${m.id} — ${title}`, result: "Success", source: "Workspace", context: `Owner: ${employees.find((e) => e.id === ownerId)?.name ?? ownerId}` });
   652→  return { ok: true as const, meeting: materializeMeeting(s, m), message: "Action item recorded on the meeting." };
   653→}

   655→// Calendar entries created from My Calendar (screen 44) — personal events
   656→// such as focus blocks and deadlines. Meeting invites WITH participants run
   657→// through the Meetings module; this records the entry on the session
   658→// calendar server-side so the screen holds no client-only business state.
   659→export function createCalendarEvent(role: WorkspaceRole, input: { title: string; type: CalendarEvent["type"]; start: number; end: number; location?: string }) {
   660→  const title = input.title?.trim() ?? "";
   661→  if (title.length < 3) return { ok: false as const, status: 422, code: "invalid_title", message: "Give the event a title (at least 3 characters)." };
   662→  const VALID_TYPES: CalendarEvent["type"][] = ["Meeting", "Deadline", "Company event", "Leave", "Focus time"];
   663→  if (!VALID_TYPES.includes(input.type)) {
   664→    return { ok: false as const, status: 422, code: "invalid_type", message: "Choose a valid event type." };
   665→  }
   666→  if (!Number.isFinite(input.start) || !Number.isFinite(input.end) || input.end <= input.start) {
   667→    return { ok: false as const, status: 422, code: "invalid_time", message: "The event end must be after its start." };
   668→  }
   669→  const ev: CalendarEvent = {
   670→    id: `ev-w${Math.floor(Date.now() / 1000) % 100000}`,
   671→    title,
   672→    type: input.type,
   673→    start: input.start,
   674→    end: input.end,
   675→    organizerId: SESSION_USER_ID,
   676→    location: input.location?.trim() || undefined,
   677→  };
   678→  store().events.push(ev);
   679→  writeAudit({ actorId: SESSION_USER_ID, action: "Calendar event created", resource: title, result: "Success", source: "Workspace", context: `${ev.type} added from My Calendar` });
   680→  return { ok: true as const, event: ev, message: `"${title}" was added to your calendar.` };
   681→}
