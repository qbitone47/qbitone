    44→const MONTHS = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
    45→const SENSITIVITY_LEVELS: Sensitivity[] = ["Public", "Internal", "Confidential", "Restricted"];

    48→/** Parse the "2 Sep 2025" style dates used across Qubit One data. */
    49→function parseDateStr(s: string): Date | null {
    50→  const m = /^(\d{1,2})\s+([A-Za-z]{3,})\s+(\d{4})$/.exec(s.trim());
    51→  if (m) {
    52→    const mi = MONTHS.findIndex((mm) => mm.toLowerCase() === m[2].slice(0, 3).toLowerCase());
    53→    if (mi >= 0) return new Date(Number(m[3]), mi, Number(m[1]));
    54→  }
    55→  const d = new Date(s);
    56→  return Number.isNaN(d.getTime()) ? null : d;
    57→}
    58→
    59→function fmtDate(d: Date): string {
    60→  return `${d.getDate()} ${MONTHS[d.getMonth()]} ${d.getFullYear()}`;
    61→}
    62→
    63→function addMonths(d: Date, n: number): Date {
    64→  const c = new Date(d);
    65→  c.setMonth(c.getMonth() + n);
    66→  return c;
    67→}
