    74→
    75→function roleCanSee(mod: NavModule, roleId: string): boolean {
    76→  if (roleId === "administrator") return true;
    77→  if (!mod.allowedRoles.includes("all") && !mod.allowedRoles.includes(roleId)) return false;
    78→  if (mod.parent) {
    79→    const parent = navModules.find((m) => m.id === mod.parent);
    80→    return parent ? roleCanSee(parent, roleId) : true;
    81→  }
    82→  return true;
    83→}

   131→      a.order = b.order;
   132→      b.order = tmp;

   154→  const previewing = previewRole !== "administrator";

   385→                      </DropdownMenu>

   387→                  </li>

   404→
   405→            <TabsContent value="details" className="px-4 pb-4 pt-3">

   500→                        {selected.allowedRoles.includes("all")
   501→                          ? "All employees"
   502→                          : selected.allowedRoles.map((r) => navRoles.find((x) => x.id === r)?.label ?? r).join(", ")}
   503→                      </KV>
   504→                    </dl>
   505→                  </div>
   506→                </>
   507→              )}
   508→            </TabsContent>
   509→
   510→            <TabsContent value="roles" className="px-4 pb-4 pt-3">
   511→              <p className="mb-3 text-xs text-muted-foreground">
   512→                Choose which roles see <span className="font-medium text-slate-700">{selected?.name}</span> in their sidebar.
   513→              </p>
   514→              <ul className="space-y-1">
   515→                {navRoles.map((r) => {
   516→                  const isAll = selected?.allowedRoles.includes("all");
   517→                  const checked = isAll || selected?.allowedRoles.includes(r.id);
   518→                  return (
   519→                    <li key={r.id} className="flex items-center justify-between gap-2 rounded-md px-2 py-2 hover:bg-slate-50">
   520→                      <label className="flex items-center gap-2.5 text-[13px] text-slate-700">
   521→                        <Checkbox
   522→                          checked={!!checked}
   523→                          disabled={!canEdit || isAll || r.id === "administrator"}
   524→                          onCheckedChange={(v) => {
   525→                            mutate((draft) => {
   526→                              const m = draft.find((x) => x.id === selected.id)!;
   527→                              if (v) m.allowedRoles = [...m.allowedRoles.filter((x) => x !== "all"), r.id];
   528→                              else m.allowedRoles = m.allowedRoles.filter((x) => x !== r.id);
   529→                            });
   530→                          }}
   531→                          aria-label={`${r.label} can see ${selected?.name}`}
   532→                        />
   533→                        {r.label}
   534→                      </label>
   535→                      {isAll && <span className="text-[11px] text-muted-foreground">All-roles module</span>}
   536→                    </li>
   537→                  );
   538→                })}
   539→              </ul>
   540→              <p className="mt-3 rounded-md bg-slate-50 px-3 py-2 text-[11px] text-muted-foreground">
   541→                Restricted modules must list explicit roles. Administrators always retain access for governance.
   542→              </p>
   543→            </TabsContent>
   544→
   545→            <TabsContent value="preview" className="px-4 pb-4 pt-3">
   546→              <Field label="Preview as role">
   547→                <Select value={previewTabRole} onValueChange={setPreviewTabRole}>
   548→                  <SelectTrigger><SelectValue /></SelectTrigger>
   549→                  <SelectContent>
   550→                    {navRoles.map((r) => (
   551→                      <SelectItem key={r.id} value={r.id}>{r.label}</SelectItem>
   552→                    ))}
   553→                  </SelectContent>
   554→                </Select>
   555→              </Field>
   556→              <div className="mt-3 rounded-lg bg-navy-900 p-3">
   557→                <p className="mb-2 px-1 text-[10px] font-semibold uppercase tracking-widest text-slate-500">Sidebar preview</p>
   558→                <ul className="space-y-0.5">
   559→                  {topLevel
   560→                    .filter((m) => roleCanSee(m, previewTabRole))
   561→                    .map((m) => (
   562→                      <li
   563→                        key={m.id}
   564→                        className="flex items-center gap-2.5 rounded-md bg-navy-800/60 px-2.5 py-1.5 text-[13px] text-slate-200"
   565→                      >
   566→                        <ModuleIcon name={m.icon} className="h-4 w-4 text-slate-400" />
   567→                        {m.name}
   568→                      </li>
   569→                    ))}
   570→                  {topLevel.filter((m) => roleCanSee(m, previewTabRole)).length === 0 && (
   571→                    <li className="px-2 py-4 text-center text-xs text-slate-500">No modules visible for this role.</li>
   572→                  )}
   573→                </ul>
   574→              </div>
   575→              <p className="mt-3 text-[11px] text-muted-foreground">
   576→                {topLevel.filter((m) => roleCanSee(m, previewTabRole)).length} of {topLevel.length} top-level modules visible to{" "}
   577→                {navRoles.find((r) => r.id === previewTabRole)?.label}.
   578→              </p>
   579→            </TabsContent>
   580→          </Tabs>
   581→        </SectionCard>
   582→      </div>
   583→
   584→      {/* Dialogs */}
   585→      {renameFor && (
   586→        <FormDialog
   587→          open
   588→          onOpenChange={(o) => !o && setRenameFor(null)}
   589→          title={`Rename — ${renameFor.name}`}
   590→          submitLabel="Rename"
   591→          onSubmit={() => {
   592→            const input = document.getElementById("rename-input") as HTMLInputElement | null;
   593→            const value = input?.value.trim();
   594→            if (value) mutate((draft) => { draft.find((m) => m.id === renameFor.id)!.name = value; });
   595→            toast({ title: "Module renamed", description: "Remember to save changes to apply navigation updates." });
   596→            setRenameFor(null);
   597→          }}
   598→        >
   599→          <Field label="Module name" required>
   600→            <Input id="rename-input" defaultValue={renameFor.name} autoFocus />
   601→          </Field>
   602→        </FormDialog>
   603→      )}
   604→
   605→      {reparentFor && (
   606→        <FormDialog
   607→          open
   608→          onOpenChange={(o) => !o && setReparentFor(null)}
   609→          title={`Set parent — ${reparentFor.name}`}
   610→          submitLabel="Move module"
   611→          onSubmit={() => {
   612→            toast({ title: "Module moved", description: "Parent updated. Save changes to apply." });
   613→            setReparentFor(null);
   614→          }}
   615→        >
   616→          <Field label="New parent" required>
   617→            <Select
   618→              defaultValue={reparentFor.parent ?? "none"}
   619→              onValueChange={(v) =>
   620→                mutate((draft) => {
   621→                  const m = draft.find((x) => x.id === reparentFor.id)!;
   622→                  m.parent = v === "none" ? null : v;
   623→                  m.order = 99;
   624→                })
   625→              }
   626→            >
   627→              <SelectTrigger><SelectValue /></SelectTrigger>
   628→              <SelectContent>
   629→                <SelectItem value="none">None (top level)</SelectItem>
   630→                {topLevel.filter((m) => m.id !== reparentFor.id).map((m) => (
   631→                  <SelectItem key={m.id} value={m.id}>{m.name}</SelectItem>
   632→                ))}
   633→              </SelectContent>
   634→            </Select>
   635→          </Field>
   636→        </FormDialog>
   637→      )}
   638→
   639→      {addOpen && (
   640→        <FormDialog
   641→          open
   642→          onOpenChange={(o) => !o && setAddOpen(null)}
   643→          title={addOpen.parent ? "Add Submodule" : "Add Top-level Module"}
   644→          submitLabel="Create module"
   645→          onSubmit={() => {
   646→            toast({ title: "Module created", description: "It appears in the tree once saved. Configure route and roles in Details." });
   647→            setAddOpen(null);
   648→          }}
   649→        >
   650→          <Field label="Module name" required>
   651→            <Input placeholder="e.g. Reports" />
   652→          </Field>
   653→          <div className="grid grid-cols-2 gap-3">
   654→            <Field label="Icon">
   655→              <Select defaultValue="folder">
   656→                <SelectTrigger><SelectValue /></SelectTrigger>
   657→                <SelectContent>
   658→                  {Object.keys(ICONS).slice(0, 12).map((k) => (
   659→                    <SelectItem key={k} value={k}>{k}</SelectItem>
   660→                  ))}
   661→                </SelectContent>
   662→              </Select>
   663→            </Field>
   664→            <Field label="Route" required>
   665→              <Input placeholder="/reports" className="font-mono text-xs" />
   666→            </Field>
   667→          </div>
   668→          {addOpen.parent && (
   669→            <p className="rounded-md bg-slate-50 px-3 py-2 text-[11px] text-muted-foreground">
   670→              Parent: <span className="font-medium text-slate-700">{modules.find((m) => m.id === addOpen.parent)?.name}</span>
   671→            </p>
   672→          )}
   673→        </FormDialog>
   674→      )}
   675→
   676→      <ConfirmDialog
   677→        open={!!deleteFor}
   678→        onOpenChange={(o) => !o && setDeleteFor(null)}
   679→        title={`Delete ${deleteFor?.name}?`}
   680→        destructive
   681→        confirmLabel="Delete module"
   682→        description="Users lose access to this navigation entry immediately after save. Routes and permissions are retained for restore."
   683→        onConfirm={() => {
   684→          mutate((draft) => draft.filter((m) => m.id !== deleteFor!.id));
   685→          toast({ title: "Module deleted", description: "Save changes to apply. Restore is available from audit history." });
   686→          setDeleteFor(null);
   687→        }}
   688→      />
   689→    </div>
   690→  );
   691→}
