    76→export function requireSalesAccess(
    77→  role: WorkspaceRole,
    78→  screen: string,
    79→): { ok: true; scope: SalesScope } | { ok: false; status: number; code: string; message: string } {
    80→  if (role === "administrator") return { ok: true, scope: "company" };
    81→  if (role === "dept-head") return { ok: true, scope: "team" };
    82→  if (role === "sales-exec") return { ok: true, scope: "own" };
    83→  writeAudit({
    84→    actorId: SESSION_USER_ID,
    85→    action: "Sales access denied",
    86→    resource: screen,
    87→    result: "Blocked",
    88→    severity: "Warning",
    89→    context: `Role boundary "${role}" holds no authorization for the Sales Operating System`,
    90→  });
    91→  return {
    92→    ok: false,
    93→    status: 403,
    94→    code: "sales_access_required",
    95→    message: "The Sales Operating System requires a sales role. You are previewing a boundary without sales authorization, and the server enforces this restriction.",
    96→  };
    97→}

   173→export function quoteSubtotal(q: Pick<SalesQuotation, "items">): number {
   174→  return q.items.reduce((sum, i) => sum + i.qty * i.unitPrice * (1 - i.discountPct / 100), 0);
   175→}
   176→export function quoteTax(q: Pick<SalesQuotation, "items" | "taxPct">): number {
   177→  return Math.round(quoteSubtotal(q) * (q.taxPct / 100));
   178→}
   179→export function quoteTotal(q: Pick<SalesQuotation, "items" | "taxPct">): number {
   180→  return quoteSubtotal(q) + quoteTax(q);
   181→}
   182→export function quoteOverallDiscountPct(q: Pick<SalesQuotation, "items">): number {
   183→  const gross = q.items.reduce((s, i) => s + i.qty * i.unitPrice, 0);
   184→  if (gross <= 0) return 0;
   185→  return ((gross - quoteSubtotal(q)) / gross) * 100;
   186→}

   309→function pipelinePayload(scope: SalesScope) {
   310→  const s = store();
   311→  const open = s.opportunities.filter((o) => canSee(scope, o.ownerId) && o.stage !== "Won" && o.stage !== "Lost");
   312→  const stages = ["New", "Qualification", "Discovery", "Proposal", "Negotiation", "Contracting"] as OpportunityStage[];
   313→  const columns = stages.map((st) => {
   314→    const cards = open.filter((o) => o.stage === st);
   315→    return {
   316→      stage: st,
   317→      count: cards.length,
   318→      value: cards.reduce((sum, o) => sum + o.valueCr, 0),
   319→      weighted: cards.reduce((sum, o) => sum + (o.valueCr * o.probability) / 100, 0),
   320→      cards: cards.map((o) => ({
   321→        id: o.id,
   322→        name: o.name,
   323→        accountId: o.accountId,
   324→        accountName: s.accounts.find((a) => a.id === o.accountId)?.name ?? "—",
   325→        ownerName: nameOf(o.ownerId),
   326→        ownerId: o.ownerId,
   327→        valueCr: o.valueCr,
   328→        probability: o.probability,
   329→        expectedClose: o.expectedClose,
   330→        nextAction: o.nextAction,
   331→        riskNote: o.riskNote,
   332→        isMine: o.ownerId === SESSION_USER_ID,
   333→      })),
   334→    };
   335→  });
   336→  const won = s.opportunities.filter((o) => canSee(scope, o.ownerId) && o.stage === "Won");
   337→  const lost = s.opportunities.filter((o) => canSee(scope, o.ownerId) && o.stage === "Lost");
   338→  const gross = open.reduce((sum, o) => sum + o.valueCr, 0);
   339→  return {
   340→    columns,
   341→    won: { count: won.length, value: won.reduce((s2, o) => s2 + o.valueCr, 0), cards: won.map((o) => ({ id: o.id, name: o.name, accountName: s.accounts.find((a) => a.id === o.accountId)?.name ?? "—", valueCr: o.valueCr, closedAt: o.closedAt, ownerName: nameOf(o.ownerId) })) },
   342→    lost: { count: lost.length, cards: lost.map((o) => ({ id: o.id, name: o.name, lossReason: o.lossReason ?? "—", ownerName: nameOf(o.ownerId) })) },
   343→    totals: { gross, weighted: open.reduce((sum, o) => sum + (o.valueCr * o.probability) / 100, 0), count: open.length },
   344→  };
   345→}

   353→function homePayload(scope: SalesScope) {
   354→  const s = store();
   355→  const mine = (o: { ownerId: string }) => canSee(scope, o.ownerId);
   356→  const myOpps = s.opportunities.filter((o) => mine(o) && o.stage !== "Won" && o.stage !== "Lost");
   357→  const now = Date.now();
   358→  const todays = s.followups.filter((f) => mine(f) && (f.state === "Overdue" || f.state === "Due Today"));
   359→  const upcoming = s.followups
   360→    .filter((f) => mine(f) && f.state === "Upcoming")
   361→    .sort((a, b) => a.dueAt - b.dueAt)
   362→    .slice(0, 4);
   363→  const meetings = s.meetings
   364→    .filter((m) => mine(m) && m.status === "Scheduled" && m.startAt >= now - 12 * 3_600_000)
   365→    .sort((a, b) => a.startAt - b.startAt)
   366→    .slice(0, 4);
   367→  const recentLeads = s.leads.filter((l) => mine(l)).sort((a, b) => b.createdAt - a.createdAt).slice(0, 4);
   368→  const recentActivity = s.activities.filter((a) => {
   369→    if (a.entity.kind === "opportunity") return s.opportunities.some((o) => o.id === a.entity.id && mine(o));
   370→    if (a.entity.kind === "lead") return s.leads.some((l) => l.id === a.entity.id && mine(l));
   371→    if (a.entity.kind === "quotation") return s.quotations.some((q) => q.id === a.entity.id && mine(q));
   372→    if (a.entity.kind === "order") return s.orders.some((o) => o.id === a.entity.id && mine(o));
   373→    return false;
   374→  }).slice(0, 6);
   375→  const pendingQuotes = s.quotations.filter((q) => mine(q) && ["Draft", "Pending Approval", "Sent", "Viewed", "Negotiation"].includes(q.status));
   376→  const target = SALES_TARGETS.find((t) => t.kind === "employee" && t.refId === SESSION_USER_ID) ?? null;
   377→  const teamTarget = scope !== "own" ? SALES_TARGETS.filter((t) => t.kind === "team") : [];
   378→  return {
   379→    scope,
   380→    myOpps: myOpps.slice(0, 5).map((o) => ({ id: o.id, name: o.name, accountName: s.accounts.find((a) => a.id === o.accountId)?.name ?? "—", stage: o.stage, valueCr: o.valueCr, probability: o.probability, expectedClose: o.expectedClose, riskNote: o.riskNote, nextAction: o.nextAction })),
   381→    pipeline: pipelinePayload(scope).totals,
   382→    followups: { today: todays.slice(0, 5), upcoming },
   383→    meetings,
   384→    recentLeads,
   385→    recentActivity,
   386→    pendingQuotations: pendingQuotes.slice(0, 4).map((q) => ({ id: q.id, number: q.number, status: q.status, accountId: q.accountId, accountName: s.accounts.find((a) => a.id === q.accountId)?.name ?? "—", total: quoteTotal(q), ownerName: nameOf(q.ownerId) })),
   387→    target: target ? { quotaCr: target.quotaCr, achievedCr: target.achievedCr, pipelineCr: target.pipelineCr, forecastCr: target.forecastCr } : null,
   388→    teamTargets: teamTarget,
   389→    counts: {
   390→      myOpenOpps: myOpps.length,
   391→      dueFollowups: todays.length,
   392→      unreadWhatsApp: s.conversations.filter((c) => mine(c) && c.unread > 0).length,
   393→      visitsPlanned: s.visits.filter((v) => mine(v) && v.status === "Planned").length,
   394→    },
   395→  };
   396→}

   398→function teamPayload(scope: SalesScope) {
   399→  const s = store();
   400→  if (scope === "own") {
   401→    return { ok: false as const, status: 403, code: "manager_required", message: "The team dashboard requires a manager boundary. Sales executives see their own workspace on Sales Home." };
   402→  }
   403→  const teams = SALES_TEAMS.map((t) => {
   404→    const target = SALES_TARGETS.find((r) => r.kind === "team" && r.refId === t.id)!;
   405→    const members = t.memberIds.map((m) => {
   406→      const empTarget = SALES_TARGETS.find((r) => r.kind === "employee" && r.refId === m);
   407→      const opps = s.opportunities.filter((o) => o.ownerId === m && o.stage !== "Won" && o.stage !== "Lost");
   408→      const overdue = s.followups.filter((f) => f.ownerId === m && f.state === "Overdue");
   409→      const lb = SALES_LEADERBOARD.find((l) => l.employeeId === m);
   410→      return {
   411→        employeeId: m,
   412→        name: nameOf(m),
   413→        quotaCr: empTarget?.quotaCr ?? 0,
   414→        achievedCr: empTarget?.achievedCr ?? 0,
   415→        pipelineCr: opps.reduce((sum, o) => sum + o.valueCr, 0),
   416→        openOpps: opps.length,
   417→        overdueFollowups: overdue.length,
   418→        conversionPct: lb?.conversionPct ?? 0,
   419→        attention:
   420→          overdue.length >= 2 ? "Overdue follow-ups piling up"
   421→          : (empTarget && empTarget.quotaCr > 0 && empTarget.achievedCr / empTarget.quotaCr < 0.2) ? "Achievement below pace"
   422→          : (lb && lb.followupCompletionPct < 70) ? "Follow-up completion needs coaching"
   423→          : null,
   424→      };
   425→    });
   426→    const teamOpps = s.opportunities.filter((o) => t.memberIds.includes(o.ownerId) && o.stage !== "Won" && o.stage !== "Lost");
   427→    const risks = teamOpps.filter((o) => o.riskNote);
   428→    return { id: t.id, name: t.name, region: t.region, head: nameOf(t.headId), target, members, risks: risks.map((o) => ({ id: o.id, name: o.name, riskNote: o.riskNote!, ownerName: nameOf(o.ownerId), valueCr: o.valueCr })), openOpps: teamOpps.length, quotations: s.quotations.filter((q) => t.memberIds.includes(q.ownerId) && ["Draft", "Pending Approval", "Sent", "Viewed", "Negotiation"].includes(q.status)).length };
   429→  });
   430→  const totals = {
   431→    quotaCr: teams.reduce((s2, t) => s2 + t.target.quotaCr, 0),
   432→    achievedCr: teams.reduce((s2, t) => s2 + t.target.achievedCr, 0),
   433→    pipelineCr: teams.reduce((s2, t) => s2 + t.target.pipelineCr, 0),
   434→    forecastCr: teams.reduce((s2, t) => s2 + t.target.forecastCr, 0),
   435→  };
   436→  return { ok: true as const, teams, totals, overdueTotal: s.followups.filter((f) => f.state === "Overdue").length };
   437→}

   671→      };

   692→        },
   693→      };
   694→    default:
   695→      return { ok: false, status: 400, code: "bad_section", message: "Unknown sales section." };
   696→  }

   717→function findDuplicates(input: { name?: string; email?: string; phone?: string; company?: string }) {
   718→  const s = store();
   719→  const matches: { kind: string; id: string; label: string; reason: string }[] = [];
   720→  for (const l of s.leads) {
   721→    if (input.email && l.email.toLowerCase() === input.email.toLowerCase()) matches.push({ kind: "lead", id: l.id, label: `${l.name} — ${l.contactPerson}`, reason: "Same email" });
   722→    else if (input.phone && l.phone.replace(/\D/g, "").endsWith(input.phone.replace(/\D/g, "").slice(-10))) matches.push({ kind: "lead", id: l.id, label: `${l.name} — ${l.contactPerson}`, reason: "Similar phone" });
   723→    else if (input.name && l.name.toLowerCase() === input.name.toLowerCase()) matches.push({ kind: "lead", id: l.id, label: `${l.name} — ${l.contactPerson}`, reason: "Same lead name" });
   724→  }
   725→  for (const c of s.contacts) {
   726→    if (input.email && c.email.toLowerCase() === input.email.toLowerCase()) matches.push({ kind: "contact", id: c.id, label: `${c.name} (${c.designation})`, reason: "Same email as existing contact" });
   727→    else if (input.phone && c.phone.replace(/\D/g, "").endsWith(input.phone.replace(/\D/g, "").slice(-10))) matches.push({ kind: "contact", id: c.id, label: `${c.name} (${c.designation})`, reason: "Similar phone to existing contact" });
   728→  }
   729→  for (const a of s.accounts) {
   730→    if (input.company && a.name.toLowerCase() === input.company.toLowerCase()) matches.push({ kind: "account", id: a.id, label: a.name, reason: "Account already exists" });
   731→  }
   732→  return matches;

   779→const num = (o: Record<string, unknown>, k: string): number | undefined => (typeof o[k] === "number" && Number.isFinite(o[k] as number) ? (o[k] as number) : undefined);
   780→const bool = (o: Record<string, unknown>, k: string): boolean | undefined => (typeof o[k] === "boolean" ? (o[k] as boolean) : undefined);
   781→
   782→export function salesPost(role: WorkspaceRole, action: Action): SalesActionResult {
   783→  const s = store();
   784→  const gate = requireSalesAccess(role, `Sales · ${action.type}`);
   785→  if (!gate.ok) return gate;
   786→  const { scope } = gate;
   787→  const audit = (args: { action: string; resource: string; severity?: "Info" | "Warning" | "High"; context: string; result?: "Success" | "Blocked" }) =>
   788→    writeAudit({ actorId: SESSION_USER_ID, action: args.action, resource: args.resource, result: args.result ?? "Success", severity: args.severity ?? "Info", source: "Sales OS", context: args.context });
   789→
   790→  const ownOnly = (ownerId: string | undefined): SalesActionResult | null => {
   791→    if (scope === "company") return null;
   792→    if (ownerId && canSee(scope, ownerId)) return null;
   793→    return { ok: false, status: 403, code: "scope_denied", message: scope === "own" ? "Only records you own can be changed from this boundary." : "Only records inside your team scope can be changed." };
   794→  };
   795→
   796→  switch (action.type) {
   797→    // ---------- leads ----------
   798→    case "lead.create": {
   799→      const i = action.input;
   800→      const name = str(i, "name");
   801→      const company = str(i, "company");
   802→      const email = str(i, "email");
   803→      const phone = str(i, "phone");
   804→      if (!name || !company || !email || !phone) return { ok: false, status: 422, code: "required", message: "Lead name, company, email and phone are required." };
   805→      if (!emailOk(email)) return { ok: false, status: 422, code: "invalid_email", message: "Enter a valid email address." };
   806→      if (!phoneOk(phone)) return { ok: false, status: 422, code: "invalid_phone", message: "Enter a valid phone number." };
   807→      if (!str(i, "ownerId")) return { ok: false, status: 422, code: "owner_required", message: "Choose an owner — leads are never created unowned." };
   808→      const matches = findDuplicates({ name, email, phone, company });
   809→      if (matches.length > 0 && !bool(i, "force")) {
   810→        return { ok: false, status: 409, code: "possible_duplicates", message: "Possible existing matches were found — review them before creating a duplicate.", matches };
   811→      }
   812→      const lead: SalesLead = {
   813→        id: nextId("lead"),
   814→        name, company,
   815→        contactPerson: str(i, "contactPerson") ?? name,
   816→        phone, email,
   817→        source: str(i, "source") ?? "Website",
   818→        industry: str(i, "industry") ?? "Other",
   819→        location: str(i, "location") ?? "—",
   820→        ownerId: str(i, "ownerId")!,
   821→        status: "New",
   822→        priority: (str(i, "priority") as LeadPriority) ?? "Warm",
   823→        score: num(i, "score") ?? 50,
   824→        estimatedValueCr: num(i, "estimatedValueCr") ?? 0,
   825→        expectedClose: str(i, "expectedClose"),
   826→        nextFollowUpAt: num(i, "nextFollowUpAt"),
   827→        createdAt: now(),
   828→        lastActivityAt: now(),
   829→        tags: Array.isArray(i.tags) ? (i.tags as string[]).slice(0, 6) : [],
   830→        notes: str(i, "notes"),
   831→      };
   832→      s.leads.unshift(lead);
   833→      addActivity({ at: now(), type: "Status change", actorId: SESSION_USER_ID, entity: { kind: "lead", id: lead.id }, summary: "Lead created", outcome: `Source ${lead.source}` });
   834→      if (lead.nextFollowUpAt) {
   835→        s.followups.unshift({ id: nextId("fu"), type: "Call", subject: `First follow-up — ${lead.name}`, ownerId: lead.ownerId, leadId: lead.id, dueAt: lead.nextFollowUpAt, state: "Upcoming", priority: lead.priority === "Hot" ? "High" : "Medium" });
   836→      }
   837→      audit({ action: "Lead created", resource: `${lead.name} — ${lead.company}`, context: `Owner ${nameOf(lead.ownerId)} · source ${lead.source}${lead.nextFollowUpAt ? " · follow-up scheduled" : ""}` });
   838→      return { ok: true, message: `Lead created and assigned to ${nameOf(lead.ownerId)}. An initial activity was recorded.`, lead: leadWithRefs(lead, scope) };
   839→    }
   840→
   841→    case "lead.assign": {
   842→      const lead = s.leads.find((l) => l.id === str(action.input, "id"));
   843→      if (!lead) return { ok: false, status: 404, code: "not_found", message: "Lead not found." };
   844→      if (!canSee(scope, lead.ownerId)) return { ok: false, status: 403, code: "scope_denied", message: "This lead is outside your authorized scope." };
   845→      if (scope === "own") return { ok: false, status: 403, code: "manager_required", message: "Reassignment is a manager action — ask your sales manager, or use Convert to move the lead forward." };
   846→      const to = str(action.input, "ownerId");
   847→      if (!to || !SALES_OWNERS.some((o) => o.id === to)) return { ok: false, status: 422, code: "owner_required", message: "Choose a valid owner from the sales team." };
   848→      if (to === lead.ownerId) return { ok: false, status: 422, code: "same_owner", message: "This lead already belongs to that owner." };
   849→      const from = lead.ownerId;
   850→      lead.ownerId = to;
   851→      lead.lastActivityAt = now();
   852→      s.followups.filter((f) => f.leadId === lead.id && f.state !== "Completed").forEach((f) => (f.ownerId = to));
   853→      addActivity({ at: now(), type: "Lead assigned", actorId: SESSION_USER_ID, entity: { kind: "lead", id: lead.id }, summary: `Ownership moved ${nameOf(from)} → ${nameOf(to)}` });
   854→      audit({ action: "Lead ownership changed", resource: `${lead.name} — ${lead.company}`, context: `${nameOf(from)} → ${nameOf(to)}; open follow-ups reassigned` });
   855→      return { ok: true, message: `Lead reassigned to ${nameOf(to)}. Open follow-ups moved with it.`, lead: leadWithRefs(lead, scope) };
   856→    }
   857→
   858→    case "lead.status": {
   859→      const lead = s.leads.find((l) => l.id === str(action.input, "id"));
   860→      if (!lead) return { ok: false, status: 404, code: "not_found", message: "Lead not found." };
   861→      if (scope === "own" && lead.ownerId !== SESSION_USER_ID) return { ok: false, status: 403, code: "scope_denied", message: "Only records you own can be changed from this boundary." };
   862→      const status = str(action.input, "status") as LeadStatus | undefined;
   863→      if (!status || !["Working", "Qualified", "Unqualified"].includes(status)) return { ok: false, status: 422, code: "invalid_status", message: "Status must be Working, Qualified or Unqualified — use Convert for the full lifecycle." };
   864→      if (status === "Unqualified" && !str(action.input, "reason")) return { ok: false, status: 422, code: "reason_required", message: "A disqualification reason is required — it is kept on the lead and in the audit trail." };
   865→      const prev = lead.status;
   866→      lead.status = status;
   867→      if (status === "Unqualified") lead.disqualifiedReason = str(action.input, "reason");
   868→      lead.lastActivityAt = now();
   869→      addActivity({ at: now(), type: "Status change", actorId: SESSION_USER_ID, entity: { kind: "lead", id: lead.id }, summary: `Lead ${prev} → ${status}`, outcome: status === "Unqualified" ? lead.disqualifiedReason : undefined });
   870→      audit({ action: `Lead marked ${status.toLowerCase()}`, resource: `${lead.name} — ${lead.company}`, context: `${prev} → ${status}${status === "Unqualified" ? ` · reason: ${lead.disqualifiedReason}` : ""}` });
   871→      return { ok: true, message: `Lead marked ${status}.`, lead: leadWithRefs(lead, scope) };
   872→    }
   873→
   874→    case "lead.note": {
   875→      const lead = s.leads.find((l) => l.id === str(action.input, "id"));
   876→      if (!lead) return { ok: false, status: 404, code: "not_found", message: "Lead not found." };
   877→      if (!canSee(scope, lead.ownerId)) return { ok: false, status: 403, code: "scope_denied", message: "This lead is outside your authorized scope." };
   878→      const body = str(action.input, "body");
   879→      if (!body || body.length < 3) return { ok: false, status: 422, code: "note_required", message: "Write the note before saving." };
   880→      lead.lastActivityAt = now();
   881→      addActivity({ at: now(), type: "Note", actorId: SESSION_USER_ID, entity: { kind: "lead", id: lead.id }, summary: body });
   882→      audit({ action: "Note added to lead", resource: `${lead.name} — ${lead.company}`, context: body.slice(0, 120) });
   883→      return { ok: true, message: "Note recorded on the lead timeline." };
   884→    }
   885→
   886→    case "lead.followup": {
   887→      const lead = s.leads.find((l) => l.id === str(action.input, "id"));
   888→      if (!lead) return { ok: false, status: 404, code: "not_found", message: "Lead not found." };
   889→      if (!canSee(scope, lead.ownerId)) return { ok: false, status: 403, code: "scope_denied", message: "This lead is outside your authorized scope." };
   890→      const dueAt = num(action.input, "dueAt");
   891→      const type = (str(action.input, "type") ?? "Call") as SalesFollowUp["type"];
   892→      if (!dueAt || dueAt < now() - 60_000) return { ok: false, status: 422, code: "invalid_date", message: "Schedule the follow-up for a future date and time." };
   893→      const fu: SalesFollowUp = { id: nextId("fu"), type, subject: str(action.input, "subject") ?? `Follow-up — ${lead.name}`, ownerId: lead.ownerId, leadId: lead.id, dueAt, state: dueAt < today0End() ? "Due Today" : "Upcoming", priority: (str(action.input, "priority") as "High" | "Medium" | "Low") ?? "Medium" };
   894→      s.followups.unshift(fu);
   895→      lead.nextFollowUpAt = dueAt;
   896→      addActivity({ at: now(), type: "Follow-up", actorId: SESSION_USER_ID, entity: { kind: "lead", id: lead.id }, summary: `${fu.type} follow-up scheduled — ${new Date(dueAt).toLocaleString("en-IN", { dateStyle: "medium", timeStyle: "short" })}` });
   897→      audit({ action: "Follow-up scheduled", resource: `${lead.name} — ${lead.company}`, context: `${fu.type} · due ${new Date(dueAt).toLocaleString("en-IN")}` });
   898→      return { ok: true, message: "Follow-up scheduled and visible in the Follow-up Queue." };
   899→    }
   900→
   901→    case "lead.convert": {
   902→      const lead = s.leads.find((l) => l.id === str(action.input, "id"));
   903→      if (!lead) return { ok: false, status: 404, code: "not_found", message: "Lead not found." };
   904→      if (!canSee(scope, lead.ownerId)) return { ok: false, status: 403, code: "scope_denied", message: "This lead is outside your authorized scope." };
   905→      if (lead.status === "Converted") return { ok: false, status: 409, code: "already_converted", message: "This lead is already converted — the history on the lead shows what was created." };
   906→      if (lead.status === "Unqualified") return { ok: false, status: 409, code: "unqualified", message: "Unqualified leads cannot be converted. Re-qualify the lead first." };
   907→      const i = action.input;
   908→      const accountId = str(i, "accountId");
   909→      const newAccountName = str(i, "newAccountName");
   910→      const contactId = str(i, "contactId");
   911→      if (!accountId && !newAccountName) return { ok: false, status: 422, code: "account_required", message: "Choose an existing account or provide a new account name — the review step prevents duplicates." };
   912→      if (!contactId && !newAccountName && !accountId) return { ok: false, status: 422, code: "contact_required", message: "Choose an existing contact or create one." };
   913→      let acc = accountId ? s.accounts.find((a) => a.id === accountId) : undefined;
   914→      if (!acc) {
   915→        const dup = findDuplicates({ company: newAccountName });
   916→        if (dup.some((m) => m.kind === "account")) return { ok: false, status: 409, code: "possible_duplicates", message: "An account with that name already exists — match it instead of creating a duplicate.", matches: dup };
   917→        acc = { id: nextId("acc"), name: newAccountName!, industry: str(i, "industry") ?? lead.industry, location: lead.location, region: "—", ownerId: str(i, "opportunityOwnerId") ?? lead.ownerId, status: "Customer", customerSince: new Date().toLocaleDateString("en-IN", { month: "short", year: "numeric" }), phone: lead.phone, website: "—", employees: 0 };
   918→        s.accounts.unshift(acc);
   919→        audit({ action: "Account created from lead conversion", resource: acc.name, context: `From lead ${lead.name}` });
   920→      }
   921→      let ct = contactId ? s.contacts.find((c) => c.id === contactId) : undefined;
   922→      if (!ct) {
   923→        const dup = findDuplicates({ email: lead.email, phone: lead.phone });
   924→        if (dup.some((m) => m.kind === "contact")) return { ok: false, status: 409, code: "possible_duplicates", message: "A contact with that email/phone already exists — match it instead of creating a duplicate.", matches: dup };
   925→        ct = { id: nextId("ct"), name: lead.contactPerson, designation: str(i, "designation") ?? "Contact", accountId: acc.id, phone: lead.phone, email: lead.email, ownerId: lead.ownerId, consent: { whatsapp: true, email: true, call: true }, preferred: "Email" };
   926→        s.contacts.unshift(ct);
   927→      }
   928→      const oppId = bool(i, "createOpportunity") === false ? undefined : nextId("opp");
   929→      let opp: SalesOpportunity | undefined;
   930→      if (oppId) {
   931→        const valueCr = num(i, "valueCr") ?? lead.estimatedValueCr;
   932→        if (!valueCr || valueCr <= 0) return { ok: false, status: 422, code: "value_required", message: "Set an opportunity value before converting." };
   933→        opp = {
   934→          id: oppId,
   935→          name: str(i, "opportunityName") ?? `${acc.name} — new business`,
   936→          accountId: acc.id,
   937→          contactId: ct.id,
   938→          ownerId: str(i, "opportunityOwnerId") ?? lead.ownerId,
   939→          stage: (str(i, "stage") as OpportunityStage) ?? "Qualification",
   940→          probability: num(i, "probability") ?? 20,
   941→          valueCr,
   942→          expectedClose: str(i, "expectedClose") ?? "TBD",
   943→          source: lead.source,
   944→          products: [],
   945→          nextAction: str(i, "nextAction") ?? "First follow-up after conversion",
   946→          nextFollowUpAt: lead.nextFollowUpAt,
   947→          createdAt: now(),
   948→          lastStageChangeAt: now(),
   949→        };
   950→        s.opportunities.unshift(opp);
   951→      }
   952→      lead.status = "Converted";
   953→      lead.convertedTo = { contactId: ct.id, accountId: acc.id, opportunityId: opp?.id, at: now() };
   954→      lead.lastActivityAt = now();
   955→      addActivity({ at: now(), type: "Conversion", actorId: SESSION_USER_ID, entity: { kind: "lead", id: lead.id }, summary: `Lead converted — contact, account${opp ? " and opportunity" : ""} created`, outcome: [ct.name, acc.name, opp?.name].filter(Boolean).join(" · ") });
   956→      audit({ action: "Lead converted", resource: `${lead.name} — ${lead.company}`, context: `Contact ${ct.name} · Account ${acc.name}${opp ? ` · Opportunity ${opp.name}` : " · no opportunity created"}` });
   957→      return {
   958→        ok: true,
   959→        message: `Conversion complete — ${ct.name} · ${acc.name}${opp ? ` · ${opp.name}` : ""}. The conversion is recorded on the lead timeline and in the audit trail.`,
   960→        contactId: ct.id,
   961→        accountId: acc.id,
   962→        opportunityId: opp?.id,
   963→      };
   964→    }
   965→
   966→    // ---------- opportunities ----------
   967→    case "opportunity.stage": {
   968→      const opp = s.opportunities.find((o) => o.id === str(action.input, "id"));
   969→      if (!opp) return { ok: false, status: 404, code: "not_found", message: "Opportunity not found." };
   970→      if (scope === "own" && opp.ownerId !== SESSION_USER_ID) return { ok: false, status: 403, code: "scope_denied", message: "Only records you own can be changed from this boundary." };
   971→      const to = str(action.input, "stage") as OpportunityStage | undefined;
   972→      if (!to) return { ok: false, status: 422, code: "stage_required", message: "Choose the target stage." };
   973→      const flow: OpportunityStage[] = ["New", "Qualification", "Discovery", "Proposal", "Negotiation", "Contracting", "Won"];
   974→      const prev = opp.stage;
   975→      if (prev === "Won" || prev === "Lost") return { ok: false, status: 409, code: "closed", message: `This opportunity is closed as ${prev}. Closed records are final — create a new opportunity for follow-on business.` };
   976→      if (to === "Won") {
   977→        if (prev !== "Contracting") return { ok: false, status: 409, code: "invalid_transition", message: "Only opportunities in Contracting can be marked Won — move the stage first." };
   978→        if (!opp.quotationId || s.quotations.find((q) => q.id === opp.quotationId)?.status !== "Accepted") {
   979→          return { ok: false, status: 409, code: "quotation_required", message: "Business rule: an opportunity needs an Accepted quotation before it can be Won." };
   980→        }
   981→      }
   982→      if (to === "Lost") {
   983→        if (!["Proposal", "Negotiation", "Contracting", "Discovery", "Qualification", "New"].includes(prev)) return { ok: false, status: 409, code: "invalid_transition", message: "This stage cannot be marked Lost." };
   984→        if (!str(action.input, "lossReason")) return { ok: false, status: 422, code: "reason_required", message: "A loss reason is required — it feeds the win/loss report." };
   985→      }
   986→      if (to !== "Won" && to !== "Lost") {
   987→        const fromIdx = flow.indexOf(prev);
   988→        const toIdx = flow.indexOf(to);
   989→        if (toIdx === -1 || toIdx < fromIdx) return { ok: false, status: 409, code: "invalid_transition", message: `Stage cannot move backwards (${prev} → ${to}). Closed stages are excluded from regression.` };
   990→        if (toIdx - fromIdx > 2) return { ok: false, status: 409, code: "skip_transition", message: "Stages move max two steps at a time — update through intermediate stages." };
   991→        const cfg = s.settings.stages.find((st) => st.name === to);
   992→        if (cfg?.requiresQuotation && !opp.quotationId) return { ok: false, status: 409, code: "quotation_required", message: `The ${to} stage expects a linked quotation — create one from this opportunity first.` };
   993→      }
   994→      opp.stage = to;
   995→      opp.probability = to === "Won" ? 100 : to === "Lost" ? 0 : (s.settings.stages.find((st) => st.name === to)?.probability ?? opp.probability);
   996→      opp.lastStageChangeAt = now();
   997→      if (to === "Won") opp.closedAt = now();
   998→      if (to === "Lost") { opp.lossReason = str(action.input, "lossReason"); opp.closedAt = now(); }
   999→      addActivity({ at: now(), type: "Stage change", actorId: SESSION_USER_ID, entity: { kind: "opportunity", id: opp.id }, summary: `Stage moved ${prev} → ${to}`, outcome: to === "Lost" ? opp.lossReason : to === "Won" ? "Closed Won" : undefined });
  1000→      audit({ action: "Opportunity stage changed", resource: `${opp.name} (${nameOf(opp.ownerId)})`, severity: to === "Won" || to === "Lost" ? "High" : "Info", context: `${prev} → ${to}${to === "Lost" ? ` · reason: ${opp.lossReason}` : ""}` });
  1001→      return { ok: true, message: `Stage updated ${prev} → ${to}. The change is audited.`, opportunity: oppWithRefs(opp) };
  1002→    }
  1003→
  1004→    case "opportunity.update": {
  1005→      const opp = s.opportunities.find((o) => o.id === str(action.input, "id"));
  1006→      if (!opp) return { ok: false, status: 404, code: "not_found", message: "Opportunity not found." };
  1007→      if (scope === "own" && opp.ownerId !== SESSION_USER_ID) return { ok: false, status: 403, code: "scope_denied", message: "Only records you own can be changed from this boundary." };
  1008→      const changes: string[] = [];
  1009→      const valueCr = num(action.input, "valueCr");
  1010→      if (valueCr !== undefined) {
  1011→        if (valueCr <= 0) return { ok: false, status: 422, code: "invalid_value", message: "Value must be greater than zero." };
  1012→        if (valueCr !== opp.valueCr) { changes.push(`value ₹${opp.valueCr} Cr → ₹${valueCr} Cr`); opp.valueCr = valueCr; }
  1013→      }
  1014→      const expectedClose = str(action.input, "expectedClose");
  1015→      if (expectedClose && expectedClose !== opp.expectedClose) { changes.push(`close date ${opp.expectedClose} → ${expectedClose}`); opp.expectedClose = expectedClose; }
  1016→      const probability = num(action.input, "probability");
  1017→      if (probability !== undefined) {
  1018→        if (probability < 0 || probability > 100) return { ok: false, status: 422, code: "invalid_probability", message: "Probability is a percentage between 0 and 100." };
  1019→        if (probability !== opp.probability) { changes.push(`probability ${opp.probability}% → ${probability}%`); opp.probability = probability; }
  1020→      }
  1021→      const nextAction = str(action.input, "nextAction");
  1022→      if (nextAction !== undefined && nextAction !== opp.nextAction) { changes.push("next action"); opp.nextAction = nextAction; }
  1023→      if (changes.length === 0) return { ok: false, status: 422, code: "no_changes", message: "Nothing changed — update at least one field." };
  1024→      addActivity({ at: now(), type: "Status change", actorId: SESSION_USER_ID, entity: { kind: "opportunity", id: opp.id }, summary: `Updated: ${changes.join(" · ")}` });
  1025→      audit({ action: "Opportunity updated", resource: `${opp.name} (${nameOf(opp.ownerId)})`, context: changes.join(" · ") });
  1026→      return { ok: true, message: "Opportunity updated — the change is recorded on the timeline and audited.", opportunity: oppWithRefs(opp) };
  1027→    }
  1028→
  1029→    // ---------- follow-ups ----------
  1030→    case "followup.complete": {
  1031→      const fu = s.followups.find((f) => f.id === str(action.input, "id"));
  1032→      if (!fu) return { ok: false, status: 404, code: "not_found", message: "Follow-up not found." };
  1033→      if (scope === "own" && fu.ownerId !== SESSION_USER_ID) return { ok: false, status: 403, code: "scope_denied", message: "Only your own follow-ups can be completed from this boundary." };
  1034→      if (fu.state === "Completed") return { ok: false, status: 409, code: "already_done", message: "This follow-up is already completed." };
  1035→      if (fu.state === "Cancelled") return { ok: false, status: 409, code: "cancelled", message: "Cancelled follow-ups cannot be completed — reschedule it first." };
  1036→      const outcome = str(action.input, "outcome");
  1037→      if (!outcome || outcome.length < 3) return { ok: false, status: 422, code: "outcome_required", message: "Record the outcome — completed follow-ups carry it into the customer timeline." };
  1038→      fu.state = "Completed";
  1039→      fu.outcome = outcome;
  1040→      fu.completedAt = now();
  1041→      const entity = fu.leadId ? { kind: "lead" as const, id: fu.leadId } : fu.accountId ? { kind: "account" as const, id: fu.accountId } : { kind: "opportunity" as const, id: fu.opportunityId ?? "" };
  1042→      addActivity({ at: now(), type: "Follow-up", actorId: SESSION_USER_ID, entity, summary: `${fu.type} follow-up completed — ${fu.subject}`, outcome });
  1043→      audit({ action: "Follow-up completed", resource: fu.subject, context: outcome.slice(0, 120) });
  1044→      return { ok: true, message: "Follow-up completed — outcome recorded in the customer timeline." };
  1045→    }
  1046→
  1047→    case "followup.reschedule": {
  1048→      const fu = s.followups.find((f) => f.id === str(action.input, "id"));

  1595→        source: "WhatsApp Campaign",

  1614→    }

  1616→    // ---------- settings & targets ----------
  1617→    case "settings.update": {
  1618→      if (role !== "administrator") {
  1619→        audit({ action: "Sales settings change denied", resource: "Sales settings", result: "Blocked", severity: "Warning", context: `Role boundary "${role}" holds no configuration authority` });
  1620→        return { ok: false, status: 403, code: "admin_required", message: "Sales configuration is restricted to the administrator boundary — changes are validated and audited." };
  1621→      }
  1622→      const area = str(action.input, "area");
  1623→      const value = action.input.value;
  1624→      if (!area) return { ok: false, status: 422, code: "area_required", message: "Name the configuration area you are changing." };
  1625→      const st = s.settings as unknown as Record<string, unknown>;
  1626→      if (!(area in st)) return { ok: false, status: 422, code: "unknown_area", message: "That configuration area does not exist." };
  1627→      if (area === "stages" && Array.isArray(value)) {
  1628→        // stages cannot be removed while opportunities reference them
  1629→        const names = (value as { name: OpportunityStage }[]).map((v) => v.name);
  1630→        const inUse = s.opportunities.filter((o) => !["Won", "Lost"].includes(o.stage)).map((o) => o.stage);
  1631→        const missing = inUse.filter((stg) => !names.includes(stg));
  1632→        if (missing.length > 0) return { ok: false, status: 409, code: "stage_in_use", message: `Cannot remove stage(s) ${[...new Set(missing)].join(", ")} — open opportunities still use them. Re-map those opportunities first.` };
  1633→      }
  1634→      const before = JSON.stringify(st[area]);
  1635→      st[area] = structuredClone(value);
  1636→      const after = JSON.stringify(st[area]);
  1637→      if (before === after) return { ok: false, status: 422, code: "no_changes", message: "Nothing changed in that area." };
  1638→      audit({ action: "Sales settings updated", resource: `Sales settings · ${area}`, severity: "High", context: `Change validated and applied; previous value recoverable from the audit trail` });
  1639→      return { ok: true, message: `${area} updated — the change is audited and recoverable.`, settings: s.settings };
  1640→    }
  1641→
  1642→    case "target.update": {
  1643→      if (role !== "administrator") return { ok: false, status: 403, code: "admin_required", message: "Targets are configured by the administrator boundary — managers see team performance, not quota editing." };
  1644→      const id = str(action.input, "id");
  1645→      const quotaCr = num(action.input, "quotaCr");
  1646→      const t = SALES_TARGETS.find((x) => x.id === id);
  1647→      if (!t) return { ok: false, status: 404, code: "not_found", message: "Target row not found." };
  1648→      if (quotaCr === undefined || quotaCr < 0) return { ok: false, status: 422, code: "invalid_quota", message: "Enter a non-negative quota." };

  1652→      return { ok: true, message: `${t.label} quota updated to ₹${quotaCr} Cr — change audited.` };
  1653→    }
  1654→
  1655→    default:
  1656→      return { ok: false, status: 400, code: "unknown_action", message: "Unsupported sales action." };
  1657→  }
