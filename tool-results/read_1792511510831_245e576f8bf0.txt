   163→export function decideApproval(
   164→  role: WorkspaceRole,
   165→  input: { id: string; decision: "Approved" | "Rejected" | "Changes requested"; note?: string; stepUpPassword?: string },
   166→) {
   167→  const s = store();
   168→  const req = s.approvals.find((a) => a.id === input.id);
   169→  if (!req) return { ok: false as const, status: 404, code: "not_found", message: "Approval request not found." };
   170→  if (req.status !== "Pending") {
   171→    return { ok: false as const, status: 409, code: "already_decided", message: `This request was already ${req.status.toLowerCase()} by another approver.` };
   172→  }
   173→
   174→  // Server-side authorization: employee boundaries hold no approval authority.
   175→  if (role === "employee" || role === "sales-exec") {
   176→    writeAudit({ actorId: SESSION_USER_ID, action: "Approval decision denied", resource: req.title, result: "Blocked", severity: "Warning", context: "Role holds no approval authority for this request type" });
   177→    return { ok: false as const, status: 403, code: "forbidden", message: "Your role does not hold approval authority for this request." };
   178→  }
   179→
   180→  // Step-up authentication for sensitive approvals (server-enforced).
   181→  if (req.stepUpRequired) {
   182→    if (!input.stepUpPassword || input.stepUpPassword.length < 8) {
   183→      return { ok: false as const, status: 401, code: "step_up_required", message: "Reauthentication is required for this approval." };
   184→    }
   185→  }
   186→
   187→  // High-value decisions require a note for the audit trail.
   188→  if (input.decision === "Rejected" && (!input.note || input.note.trim().length < 4)) {
   189→    return { ok: false as const, status: 422, code: "note_required", message: "A reason is required when rejecting a request." };
   190→  }
   191→
   192→  req.status = input.decision;
   193→  req.decidedAt = Date.now();
   194→  req.decidedBy = SESSION_USER_ID;
   195→  req.decisionNote = input.note?.trim() || undefined;
   196→  req.history = [
   197→    ...(req.history ?? []),
   198→    { at: Date.now(), actor: "Sagar Singh", event: input.decision + (input.note ? ` — ${input.note.trim()}` : "") },
   199→  ];
   200→
   201→  writeAudit({
   202→    actorId: SESSION_USER_ID,
   203→    action: `Approval ${input.decision.toLowerCase()}`,
   204→    resource: `${req.kind} · ${req.title}`,
   205→    result: "Success",
   206→    severity: req.stepUpRequired ? "High" : "Info",
   207→    source: "Web · Mumbai",
   208→    context: req.stepUpRequired
   209→      ? "Step-up authentication verified before the decision"
   210→      : `Routed via Approval Matrix (${req.rule})`,
   211→  });
   212→
   213→  return { ok: true as const, approval: req };
   214→}

   224→export function createTask(role: WorkspaceRole, input: { title: string; detail?: string; priority?: Task["priority"]; due?: number | null; /** Phase 20: origin of the task (defaults to Manual for the quick action) */ source?: Task["source"]; related?: Task["related"] }) {
   225→  const title = input.title?.trim() ?? "";
   226→  if (title.length < 4) return { ok: false as const, status: 422, code: "invalid_title", message: "Task title must be at least 4 characters." };
   227→  if (title.length > 140) return { ok: false as const, status: 422, code: "invalid_title", message: "Task title must be 140 characters or fewer." };
   228→  if (input.due !== null && input.due !== undefined && Number.isNaN(input.due)) {
   229→    return { ok: false as const, status: 422, code: "invalid_due", message: "Due date is invalid." };
   230→  }
   231→
   232→  const t: Task = {
   233→    id: `task-${Math.floor(Date.now() / 1000) % 100000}`,
   234→    title,
   235→    detail: input.detail?.trim() || "",
   236→    status: "Not Started",
   237→    priority: input.priority ?? "Medium",
   238→    due: input.due ?? null,
   239→    source: input.source ?? "Manual",
   240→    ownerId: SESSION_USER_ID,
   241→    assigneeId: SESSION_USER_ID,
   242→    related: input.related,
   243→  };
   244→  store().tasks.unshift(t);
   245→  writeAudit({ actorId: SESSION_USER_ID, action: "Task created", resource: t.title, result: "Success", context: input.source ? `Created from ${input.source}` : "Created from Employee Home quick action" });

   251→  "In Progress": ["Waiting", "Blocked", "Completed", "Cancelled"],
   252→  Blocked: ["In Progress", "Waiting", "Cancelled"],

   258→export function updateTask(role: WorkspaceRole, input: { id: string; status: Task["status"] }) {
   259→  const t = store().tasks.find((x) => x.id === input.id);
   260→  if (!t) return { ok: false as const, status: 404, code: "not_found", message: "Task not found." };
   261→  // Authorization: only the assignee transitions their own task.
   262→  if (t.assigneeId !== SESSION_USER_ID) {
   263→    writeAudit({ actorId: SESSION_USER_ID, action: "Task update denied", resource: t.title, result: "Blocked", severity: "Warning", context: "Caller is not the task assignee" });
   264→    return { ok: false as const, status: 403, code: "forbidden", message: "Only the task assignee can update this task." };
   265→  }
   266→  const allowed = TASK_TRANSITIONS[t.status] ?? [];
   267→  if (!allowed.includes(input.status)) {
   268→    return { ok: false as const, status: 409, code: "invalid_transition", message: `A task cannot move from ${t.status} to ${input.status}.` };
   269→  }
   270→  t.status = input.status;
   271→  if (input.status === "Completed") t.completedAt = Date.now();
   272→  writeAudit({ actorId: SESSION_USER_ID, action: "Task status changed", resource: t.title, result: "Success", context: `${t.status === "Completed" ? "Marked completed" : `Moved to ${input.status}`}` });
   273→  return { ok: true as const, task: t };
   274→}

   282→export function listMail(role: WorkspaceRole, params: { folder?: string; q?: string; unreadOnly?: boolean; label?: string; limit?: number }) {
   283→  const s = store();
   284→  const folder = params.folder ?? "inbox";
   285→  let items = folder === "all" ? [...s.mail] : s.mail.filter((m) => m.folder === folder);
   286→  // Authorization scope: the session mailbox only (demo: single-user store).
   287→  const q = params.q?.trim().toLowerCase();
   288→  if (q) {
   289→    items = items.filter((m) => {
   290→      const fromName = m.fromId ? employees.find((e) => e.id === m.fromId)?.name ?? "" : m.fromExternal?.name ?? "";
   291→      const fromEmail = m.fromId ? employees.find((e) => e.id === m.fromId)?.email ?? "" : m.fromExternal?.email ?? "";
   292→      return [m.subject, m.preview, m.body, fromName, fromEmail, ...(m.to ?? [])].join(" ").toLowerCase().includes(q);
   293→    });
   294→  }
   295→  if (params.unreadOnly) items = items.filter((m) => !m.read);
   296→  if (params.label) items = items.filter((m) => m.labels?.includes(params.label as never));
   297→  items.sort((a, b) => b.ts - a.ts);
   298→  const limit = params.limit && params.limit > 0 ? Math.min(params.limit, 200) : 100;
   299→  return { items: items.slice(0, limit), total: items.length, scopeNote: "Search is scoped to your authorized mailbox server-side." };
   300→}

   302→export function sendMail(role: WorkspaceRole, input: { to: string[]; cc?: string[]; bcc?: string[]; subject: string; body: string; attachments?: { name: string; sizeKb: number; type: string }[] }) {
   303→  const to = (input.to ?? []).map((v) => v.trim()).filter(Boolean);
   304→  const cc = (input.cc ?? []).map((v) => v.trim()).filter(Boolean);
   305→  if (to.length === 0) return { ok: false as const, status: 422, code: "no_recipients", message: "Add at least one recipient." };
   306→  if (to.some((v) => !emailLike(v))) return { ok: false as const, status: 422, code: "invalid_recipient", message: "One or more recipient addresses are not valid." };
   307→  if (!input.subject?.trim() && !input.body?.trim()) {
   308→    return { ok: false as const, status: 422, code: "empty_message", message: "Add a subject or a message before sending." };
   309→  }
   310→  const atts = input.attachments ?? [];
   311→  if (atts.length > MAX_ATTACHMENTS) return { ok: false as const, status: 422, code: "too_many_attachments", message: `A message can carry at most ${MAX_ATTACHMENTS} attachments.` };
   312→  const totalKb = atts.reduce((n, a) => n + (a.sizeKb || 0), 0);
   313→  if (totalKb > MAX_TOTAL_KB) return { ok: false as const, status: 422, code: "attachment_too_large", message: "Attachments exceed the 10 MB message limit." };
   314→
   315→  const msg: MailMessage = {
   316→    id: `s-${Math.floor(Date.now() / 1000) % 100000}`,
   317→    folder: "sent",
   318→    fromId: SESSION_USER_ID,
   319→    to,
   320→    cc: cc.length ? cc : undefined,
   321→    subject: input.subject?.trim() || "(no subject)",
   322→    preview: (input.body ?? "").replace(/\s+/g, " ").trim().slice(0, 140),
   323→    body: input.body ?? "",
   324→    ts: Date.now(),
   325→    read: true,
   326→    sendStatus: "Sent",
   327→    attachments: atts.length
   328→      ? atts.map((a) => ({ name: a.name, sizeKb: a.sizeKb, type: (a.type?.toUpperCase() || "BIN") as never }))
   329→      : undefined,
   330→  };
   331→  store().mail.unshift(msg);
   332→  writeAudit({ actorId: SESSION_USER_ID, action: "Mail sent", resource: msg.subject, result: "Success", context: `${to.length} recipient(s)${cc.length ? `, ${cc.length} cc` : ""}${atts.length ? `, ${atts.length} attachment(s)` : ""}` });
   333→  return { ok: true as const, message: msg };
   334→}

   367→export function retryMail(role: WorkspaceRole, id: string) {
   368→  const m = store().mail.find((x) => x.id === id);
   369→  if (!m || m.folder !== "sent") return { ok: false as const, status: 404, code: "not_found", message: "Message not found in Sent." };
   370→  if (m.sendStatus !== "Failed" && m.sendStatus !== "Pending") {
   371→    return { ok: false as const, status: 409, code: "not_retryable", message: `This message is already ${(m.sendStatus ?? "sent").toLowerCase()}.` };
   372→  }
   373→  m.sendStatus = "Sent";
   374→  m.sendDetail = undefined;
   375→  writeAudit({ actorId: SESSION_USER_ID, action: "Mail retry succeeded", resource: m.subject, result: "Success", context: "Delivery retried after failure" });
   376→  return { ok: true as const, message: m };
   377→}

   394→const SENSITIVE_DOC_ACTIONS = new Set(["download", "share"]);

   396→export function documentAction(
   397→  role: WorkspaceRole,
   398→  input: { action: "download" | "share" | "rename" | "move" | "upload"; docId?: string; name?: string; category?: string; toEmployeeId?: string; fileMeta?: { name: string; sizeKb: number; type: string } },
   399→) {
   400→  const s = store();
   401→
   402→  if (input.action === "upload") {
   403→    const f = input.fileMeta;
   404→    if (!f || !f.name?.trim()) return { ok: false as const, status: 422, code: "invalid_file", message: "Choose a file to upload." };
   405→    if (f.sizeKb > MAX_TOTAL_KB) return { ok: false as const, status: 422, code: "file_too_large", message: "Files above 10 MB must go through the enterprise storage vault." };
   406→    const doc: WorkspaceDoc = {
   407→      id: `doc-${Math.floor(Date.now() / 1000) % 100000}`,
   408→      name: f.name,
   409→      type: (f.type?.toUpperCase() || "DOCX") as WorkspaceDoc["type"],
   410→      category: "Personal",
   411→      ownerId: SESSION_USER_ID,
   412→      sizeKb: f.sizeKb,
   413→      modified: Date.now(),
   414→      sensitivity: "Internal",
   415→      description: "Uploaded from My Documents. File content remains in your enterprise storage; Qubit One holds the record.",
   416→    };
   417→    s.documents.unshift(doc);
   418→    writeAudit({ actorId: SESSION_USER_ID, action: "Document uploaded", resource: doc.name, result: "Success", context: "My Documents · metadata record only" });
   419→    return { ok: true as const, doc };
   420→  }
   421→
   422→  const doc = s.documents.find((d) => d.id === input.docId);
   423→  if (!doc) return { ok: false as const, status: 404, code: "not_found", message: "Document not found." };
   424→
   425→  if (SENSITIVE_DOC_ACTIONS.has(input.action) && doc.sensitivity === "Restricted") {
   426→    writeAudit({ actorId: SESSION_USER_ID, action: `Document ${input.action} denied`, resource: doc.name, result: "Blocked", severity: "Warning", context: "Restricted document — HR-managed access only" });
   427→    return { ok: false as const, status: 403, code: "restricted", message: "This document is restricted. Access is managed by HR — request access from the document owner." };
   428→  }
   429→
   430→  if (input.action === "download") {
   431→    writeAudit({ actorId: SESSION_USER_ID, action: "Document downloaded", resource: doc.name, result: "Success", context: `Sensitivity: ${doc.sensitivity}` });
   432→    return { ok: true as const, doc };
   433→  }
   434→
   435→  if (input.action === "share") {
   436→    if (!input.toEmployeeId || !employees.some((e) => e.id === input.toEmployeeId)) {
   437→      return { ok: false as const, status: 422, code: "invalid_recipient", message: "Choose a colleague to share with." };
   438→    }
   439→    if (doc.sensitivity === "Confidential" && role === "employee") {
   440→      return { ok: false as const, status: 403, code: "forbidden", message: "Confidential documents cannot be shared from this role." };
   441→    }
   442→    writeAudit({ actorId: SESSION_USER_ID, action: "Document shared", resource: doc.name, result: "Success", context: `Shared with ${employees.find((e) => e.id === input.toEmployeeId)?.name}` });
   443→    return { ok: true as const, doc };
   444→  }
   445→
   446→  if (doc.ownerId !== SESSION_USER_ID) {
   447→    return { ok: false as const, status: 403, code: "forbidden", message: "Only the owner can rename or move this document." };
   448→  }

   456→      return { ok: false as const, status: 422, code: "invalid_category", message: "Choose a valid category." };
   457→    }
   458→    doc.category = cat;
   459→  }
   460→  return { ok: true as const, doc };
   461→}

   497→// served from this store like every other workspace domain
   498→// (Phase 19 §6/§12: no enterprise fixture values imported into
   499→// client components).
   500→//
   501→// Scope model: "My Meetings" is a personal workspace surface —
   502→// exactly like listTasks — so the list is scoped to the session
   503→// identity by participation (organizer or invited participant)
   504→// for every role preview. Roles with governance authority
   505→// (administrator / dept-head) may CANCEL meetings they do not
   506→// organize; that permission is enforced here, never in the UI.
   507→//
   508→// Minutes flow: notes and action items are recorded AGAINST the
   509→// store's session maps (meetingNotes / meetingActions, keyed by
   510→// meeting id — mirroring mobile-server's meetingNotes/meetingActions
   511→// concept) for COMPLETED meetings, and only by the organizer.
   512→// materializeMeeting() merges the fixture record with those session
   513→// entries at read time, so every workspace reader — list, ?id=
   514→// detail, cancel response and the calendar drawer — sees the same
   515→// state without the store ever mutating the fixture-shaped record.
   516→//
   517→// Phase 20 completion of the chain:
   518→//   · createMeeting — full meeting creation with participants; every
   519→//     invited participant gets a notification in the SAME store that
   520→//     seeds the notification center (no second notification system).
   521→//   · cancelMeeting — now actually notifies participants (the message
   522→//     always claimed it; the store makes it true).
   523→//   · addMeetingAction(asTask) / promoteMeetingAction — meeting action
   524→//     items become tasks in the ONE centralized task store via the
   525→//     existing createTask, and promotion is tracked so an item can
   526→//     never become two tasks.
   527→// ============================================================

   529→/** Session display name (demo identity is fixed). */
   530→const sessionName = employees.find((e) => e.id === SESSION_USER_ID)?.name ?? SESSION_USER_ID;
   531→
   532→/** Fixture record + session minutes merged into the wire shape.
   533→ *  Phase 20: also stamps `promoted` onto action items from the
   534→ *  meetingActionsPromoted map so detail readers can hide the
   535→ *  "Add to My Tasks" affordance for items already promoted. */
   536→function materializeMeeting(s: WorkspaceStore, m: Meeting): Meeting {
   537→  const notes = s.meetingNotes.get(m.id) ?? [];
   538→  const actions = s.meetingActions.get(m.id) ?? [];
   539→  const promoted = s.meetingActionsPromoted.get(m.id);
   540→  if (notes.length === 0 && actions.length === 0 && (!promoted || promoted.size === 0)) return m;
   541→  const all: NonNullable<Meeting["actionItems"]> = [
   542→    ...(m.actionItems ?? []).map((a) => ({ ...a })),
   543→    ...actions.map((a) => ({ title: a.title, ownerId: a.ownerId, due: a.due })),
   544→  ];
   545→  for (const i of promoted ?? []) {
   546→    if (all[i]) all[i] = { ...all[i], promoted: true };
   547→  }
   548→  return {
   549→    ...m,
   550→    notes: notes.length ? [m.notes, ...notes.map((n) => n.text)].filter(Boolean).join("\n\n") : m.notes,
   551→    actionItems: all.length ? all : m.actionItems,
   552→  };
   553→}

   590→/**
   591→ * Phase 20 — participant notifications. Appends one WorkspaceNotification
   592→ * per participant EXCLUDING the acting session user, into the SAME store
   593→ * that seeds the notification center (single notification system for
   594→ * desktop and mobile — no second engine). Idempotent by id:
   595→ * ntf-mtg-<meetingId>-<kind>-<participantId>.
   596→ * ts uses the fixed demo clock (WORKSPACE_NOW) so dynamic notifications
   597→ * render "just now" against the workspace timeline every other surface
   598→ * anchors to — consistent with relTime's contract in workspace-data.

   600→function notifyMeetingParticipants(meeting: Meeting, kind: "created" | "cancelled", actorName: string): number {
   601→  const s = store();
   602→  let notified = 0;
   603→  for (const pid of meeting.participantIds) {
   604→    if (pid === SESSION_USER_ID) continue; // the acting organizer/manager never notifies themselves
   605→    const id = `ntf-mtg-${meeting.id}-${kind}-${pid}`;
   606→    if (s.notifications.some((n) => n.id === id)) continue; // idempotent
   607→    const when = `${fmtDate(meeting.start)} · ${fmtTime(meeting.start)}–${fmtTime(meeting.end)}`;
   608→    const notification: WorkspaceNotification = {
   609→      id,
   610→      title: kind === "created" ? `Meeting invite — "${meeting.title}"` : `"${meeting.title}" was cancelled`,
   611→      body:
   612→        kind === "created"
   613→          ? `${actorName} invited you. ${when} · ${meeting.platform}${meeting.location ? ` · ${meeting.location}` : ""}.`
   614→          : `${actorName} cancelled the ${when} slot (${meeting.platform}). Your calendar is updated.`,
   615→      category: "Meetings",
   616→      ts: WORKSPACE_NOW,
   617→      unread: true,
   618→      priority: kind === "cancelled" ? "High" : "Medium",
   619→      source: "Meetings",
   620→      link: { screen: "my-meetings", params: { meetingId: meeting.id } },
   621→    };
   622→    s.notifications.push(notification);
   623→    notified += 1;
   624→  }
   625→  return notified;
   626→}

   655→export function cancelMeeting(role: WorkspaceRole, input: { id: string; reason: string }) {
   656→  const s = store();
   657→  const m = s.meetings.find((x) => x.id === input.id);
   658→  if (!m) return { ok: false as const, status: 404, code: "not_found", message: "Meeting not found." };
   659→  if (m.status !== "Scheduled") {
   660→    return {
   661→      ok: false as const,
   662→      status: 409,
   663→      code: "invalid_state",
   664→      message: m.status === "Cancelled" ? "This meeting was already cancelled." : `A meeting that is ${m.status.toLowerCase()} cannot be cancelled.`,
   665→    };
   666→  }
   667→  const isOrganizer = m.organizerId === SESSION_USER_ID;
   668→  const governance = role === "administrator" || role === "dept-head";
   669→  if (!isOrganizer && !governance) {
   670→    writeAudit({ actorId: SESSION_USER_ID, action: "Meeting cancellation denied", resource: `${m.title} (${m.id})`, result: "Blocked", severity: "Warning", source: "Workspace", context: "Only the organizer — or an administrator / department head — can cancel a meeting" });
   671→    return { ok: false as const, status: 403, code: "forbidden", message: "Only the meeting organizer — or an administrator / department head — can cancel this meeting." };
   672→  }
   673→  const reason = input.reason?.trim() ?? "";
   674→  if (reason.length < 4) {
   675→    return { ok: false as const, status: 422, code: "reason_required", message: "A short reason is required when cancelling a meeting." };
   676→  }
   677→  m.status = "Cancelled";
   678→  // Phase 20 §5 — a cancelled meeting leaves the Calendar: remove its linked
   679→  // event(s) from the same store so the two surfaces never disagree.
   680→  const before = s.events.length;
   681→  s.events = s.events.filter((e) => e.meetingId !== m.id);
   682→  const removedEvents = before - s.events.length;
   683→  const notified = notifyMeetingParticipants(m, "cancelled", sessionName);
   684→  writeAudit({
   685→    actorId: SESSION_USER_ID,
   686→    action: "Meeting cancelled",
   687→    resource: `${m.id} — ${m.title}`,
   688→    result: "Success",
   689→    severity: "Info",
   690→    source: "Workspace",
   691→    context: `${isOrganizer ? "Organizer" : "Governance"} cancellation — reason: ${reason} · ${notified} participant${notified === 1 ? "" : "s"} notified · ${removedEvents} calendar event${removedEvents === 1 ? "" : "s"} removed`,
   692→  });
   693→  return {
   694→    ok: true as const,
   695→    meeting: materializeMeeting(s, m),
   696→    message: `"${m.title}" was cancelled — ${notified} participant${notified === 1 ? "" : "s"} notified and the calendar updated.`,
   697→  };
   698→}

   726→export function addMeetingAction(role: WorkspaceRole, input: { id: string; title: string; ownerId?: string; due?: number | null; /** Phase 20: also create a task in the ONE task store */ asTask?: boolean }) {
   727→  const s = store();
   728→  const m = s.meetings.find((x) => x.id === input.id);
   729→  if (!m) return { ok: false as const, status: 404, code: "not_found", message: "Meeting not found." };
   730→  if (m.organizerId !== SESSION_USER_ID) {
   731→    writeAudit({ actorId: SESSION_USER_ID, action: "Meeting action item denied", resource: `${m.id} — ${m.title}`, result: "Blocked", severity: "Warning", source: "Workspace", context: "Only the organizer can record action items" });
   732→    return { ok: false as const, status: 403, code: "forbidden", message: "Only the meeting organizer can add action items." };
   733→  }
   734→  if (m.status !== "Completed") {
   735→    return { ok: false as const, status: 409, code: "invalid_state", message: `Action items are recorded on completed meetings — this one is ${m.status.toLowerCase()}.` };
   736→  }
   737→  const title = input.title?.trim() ?? "";
   738→  if (title.length < 4) return { ok: false as const, status: 422, code: "invalid_title", message: "Describe the action item (4+ characters)." };
   739→  const ownerId = input.ownerId ?? SESSION_USER_ID;
   740→  if (!employees.some((e) => e.id === ownerId)) {
   741→    return { ok: false as const, status: 422, code: "invalid_owner", message: "Choose a valid owner." };
   742→  }
   743→  if (input.due !== null && input.due !== undefined && Number.isNaN(input.due)) {
   744→    return { ok: false as const, status: 422, code: "invalid_due", message: "Due date is invalid." };
   745→  }
   746→
   747→  // Phase 20 — asTask: create the task FIRST (via the existing createTask,
   748→  // the ONE centralized task store) so a failed task creation leaves the
   749→  // meeting record untouched. The item's index in the MATERIALIZED array
   750→  // is fixture items first, session items after — compute it pre-insert.
   751→  let taskId: string | undefined;
   752→  if (input.asTask) {
   753→    const taskRes = createTask(role, {
   754→      title,
   755→      detail: `Meeting action item from ${m.id} — "${m.title}".`,
   756→      due: input.due ?? null,
   757→      source: "Meeting action item",
   758→      related: { label: m.title, kind: "Meeting", ref: m.id },
   759→    });
   760→    if (!taskRes.ok) return taskRes;
   761→    taskId = taskRes.task.id;
   762→  }
   763→
   764→  const existing = s.meetingActions.get(m.id) ?? [];
   765→  const index = (m.actionItems?.length ?? 0) + existing.length;
   766→  s.meetingActions.set(m.id, [...existing, { title, ownerId, due: input.due ?? null, at: Date.now() }]);
   767→  if (taskId) {
   768→    const promoted = s.meetingActionsPromoted.get(m.id) ?? new Set<number>();
   769→    promoted.add(index);
   770→    s.meetingActionsPromoted.set(m.id, promoted);
   771→  }
   772→
   773→  // ONE audit event carries both facts when a task is also created.
   774→  if (taskId) {
   775→    writeAudit({ actorId: SESSION_USER_ID, action: "Meeting action item recorded", resource: `${m.id} — ${title}`, result: "Success", source: "Workspace", context: `Owner: ${employees.find((e) => e.id === ownerId)?.name ?? ownerId} · task created (${taskId})` });
   776→    return { ok: true as const, meeting: materializeMeeting(s, m), taskId, message: "Action item recorded on the meeting — task added to My Tasks." };
   777→  }
   778→  writeAudit({ actorId: SESSION_USER_ID, action: "Meeting action item added", resource: `${m.id} — ${title}`, result: "Success", source: "Workspace", context: `Owner: ${employees.find((e) => e.id === ownerId)?.name ?? ownerId}` });
   779→  return { ok: true as const, meeting: materializeMeeting(s, m), message: "Action item recorded on the meeting." };
   780→}

   782→/**
   783→ * Phase 20 — promote an already-recorded action item (by index in the
   784→ * MATERIALIZED actionItems array) into a task via the existing createTask.
   785→ * The organizer OR any participant may promote into their own task queue
   786→ * (the task is always created for the session user — "Add to My Tasks");
   787→ * promotion is tracked per meeting/index so an item can never become two
   788→ * tasks (409 on repeat).
   789→ */
   790→export function promoteMeetingAction(role: WorkspaceRole, input: { id: string; index: number }) {
   791→  const s = store();
   792→  const m = s.meetings.find((x) => x.id === input.id);
   793→  if (!m) return { ok: false as const, status: 404, code: "not_found", message: "Meeting not found." };
   794→  const participating = m.organizerId === SESSION_USER_ID || m.participantIds.includes(SESSION_USER_ID);
   795→  if (!participating) {
   796→    writeAudit({ actorId: SESSION_USER_ID, action: "Meeting action promotion denied", resource: `${m.id} — ${m.title}`, result: "Blocked", severity: "Warning", source: "Workspace", context: "Caller is neither the organizer nor a participant" });
   797→    return { ok: false as const, status: 403, code: "forbidden", message: "You are not a participant of this meeting." };
   798→  }
   799→  const item = materializeMeeting(s, m).actionItems?.[input.index];
   800→  if (!item) return { ok: false as const, status: 404, code: "not_found", message: "Action item not found." };
   801→  const promoted = s.meetingActionsPromoted.get(m.id) ?? new Set<number>();
   802→  if (promoted.has(input.index)) {
   803→    return { ok: false as const, status: 409, code: "already_promoted", message: `"${item.title}" is already in My Tasks.` };
   804→  }
   805→  const taskRes = createTask(role, {
   806→    title: item.title,
   807→    detail: `Promoted from meeting action item — ${m.id} "${m.title}".`,
   808→    due: item.due,
   809→    source: "Meeting action item",
   810→    related: { label: m.title, kind: "Meeting", ref: m.id },
   811→  });
   812→  if (!taskRes.ok) return taskRes;
   813→  promoted.add(input.index);
   814→  s.meetingActionsPromoted.set(m.id, promoted);
   815→  writeAudit({ actorId: SESSION_USER_ID, action: "Meeting action item promoted to task", resource: `${m.id} — ${item.title}`, result: "Success", source: "Workspace", context: `Task ${taskRes.task.id} created in My Tasks` });
   816→  return { ok: true as const, task: taskRes.task, message: `"${item.title}" was added to My Tasks.` };
   817→}

   819→/**
   820→ * Phase 20 — full meeting creation with participants. Any role may create
   821→ * their own meeting (the session user is always the organizer and a
   822→ * participant). Every invited participant receives a notification in the
   823→ * SAME store that seeds the notification center, and the creation is
   824→ * audited with participant count and time.
   825→ */
   826→export function createMeeting(role: WorkspaceRole, input: { title: string; agenda?: string; start: number; end: number; location?: string; platform?: string; participantIds: string[] }) {
   827→  const s = store();
   828→  const title = input.title?.trim() ?? "";
   829→  if (!title) return { ok: false as const, status: 422, code: "invalid_title", message: "Give the meeting a title." };
   830→  if (title.length < 4) return { ok: false as const, status: 422, code: "invalid_title", message: "Meeting title must be at least 4 characters." };
   831→  if (title.length > 140) return { ok: false as const, status: 422, code: "invalid_title", message: "Keep the meeting title within 140 characters." };
   832→  if (!Number.isFinite(input.start) || !Number.isFinite(input.end)) {
   833→    return { ok: false as const, status: 400, code: "invalid_time", message: "Start and end must be epoch milliseconds." };
   834→  }
   835→  if (input.end <= input.start) {
   836→    return { ok: false as const, status: 400, code: "invalid_time", message: "The meeting end must be after its start." };
   837→  }
   838→  if (input.location !== undefined && (typeof input.location !== "string" || input.location.trim().length > 120)) {
   839→    return { ok: false as const, status: 422, code: "invalid_location", message: "Location must be 120 characters or fewer." };
   840→  }
   841→  const VALID_PLATFORMS: Meeting["platform"][] = ["Qubit Meet", "Zoom", "Microsoft Teams", "Google Meet", "Room", "In person"];
   842→  if (input.platform !== undefined && !VALID_PLATFORMS.includes(input.platform as Meeting["platform"])) {
   843→    return { ok: false as const, status: 422, code: "invalid_platform", message: "Choose a valid platform (Qubit Meet, Zoom, Microsoft Teams, Google Meet, Room or In person)." };
   844→  }
   845→  if (!Array.isArray(input.participantIds)) {
   846→    return { ok: false as const, status: 400, code: "invalid_participants", message: "participantIds must be a list of employee ids." };
   847→  }
   848→  const cleaned = Array.from(new Set(input.participantIds.map((v) => String(v ?? "").trim()).filter(Boolean)));
   849→  const unknown = cleaned.filter((pid) => !employees.some((e) => e.id === pid));
   850→  if (unknown.length > 0) {
   851→    return { ok: false as const, status: 400, code: "invalid_participants", message: `Unknown participant${unknown.length === 1 ? "" : "s"}: ${unknown.join(", ")}.` };
   852→  }
   853→  const participantIds = Array.from(new Set([SESSION_USER_ID, ...cleaned])); // organizer joins and duplicates collapse
   854→  if (participantIds.length > 12) {
   855→    return { ok: false as const, status: 422, code: "too_many_participants", message: "A meeting can have at most 12 participants." };
   856→  }
   857→
   858→  const platform = (input.platform as Meeting["platform"]) ?? "Qubit Meet";
   859→  const seq = s.meetings.filter((x) => /^mtg-9\d\d$/.test(x.id)).length; // dynamic ids start at mtg-900 — never collides with fixture mtg-3xx
   860→  const meeting: Meeting = {
   861→    id: `mtg-${900 + seq}`,
   862→    title,
   863→    start: input.start,
   864→    end: input.end,
   865→    organizerId: SESSION_USER_ID,
   866→    participantIds,
   867→    location: input.location?.trim() || platform,
   868→    platform,
   869→    status: "Scheduled",
   870→    agenda: input.agenda?.trim() || undefined,
   871→  };
   872→  s.meetings.push(meeting);
   873→  // Phase 20 §5 — the meeting appears in the Calendar (screen 44): a calendar
   874→  // event of type "Meeting" is created in the SAME store the calendar reads,
   875→  // linked by meetingId so detail navigation and cancellation stay consistent.
   876→  const event: CalendarEvent = {
   877→    id: `ev-${900 + seq}`,
   878→    title,
   879→    type: "Meeting",
   880→    start: meeting.start,
   881→    end: meeting.end,
   882→    organizerId: SESSION_USER_ID,
   883→    meetingId: meeting.id,
   884→    location: meeting.location,
   885→  };
   886→  s.events.push(event);
   887→  const notified = notifyMeetingParticipants(meeting, "created", sessionName);
   888→  writeAudit({
   889→    actorId: SESSION_USER_ID,
   890→    action: "Meeting scheduled",
   891→    resource: `${meeting.id} — ${title}`,
   892→    result: "Success",
   893→    severity: "Info",
   894→    source: "Workspace",
   895→    context: `${participantIds.length} participant${participantIds.length === 1 ? "" : "s"} (${notified} notified) · ${fmtDate(meeting.start)} ${fmtTime(meeting.start)}–${fmtTime(meeting.end)} IST`,
   896→  });
   897→  return {
   898→    ok: true as const,
   899→    meeting: materializeMeeting(s, meeting),
   900→    message: `"${title}" scheduled with ${participantIds.length} participant${participantIds.length === 1 ? "" : "s"} — ${notified} notified.`,
   901→  };
   902→}

   904→// Calendar entries created from My Calendar (screen 44) — personal events
   905→// such as focus blocks and deadlines. Meeting invites WITH participants run
   906→// through the Meetings module; this records the entry on the session
   907→// calendar server-side so the screen holds no client-only business state.
   908→export function createCalendarEvent(role: WorkspaceRole, input: { title: string; type: CalendarEvent["type"]; start: number; end: number; location?: string }) {
   909→  const title = input.title?.trim() ?? "";
   910→  if (title.length < 3) return { ok: false as const, status: 422, code: "invalid_title", message: "Give the event a title (at least 3 characters)." };
   911→  const VALID_TYPES: CalendarEvent["type"][] = ["Meeting", "Deadline", "Company event", "Leave", "Focus time"];
   912→  if (!VALID_TYPES.includes(input.type)) {
   913→    return { ok: false as const, status: 422, code: "invalid_type", message: "Choose a valid event type." };
   914→  }
   915→  if (!Number.isFinite(input.start) || !Number.isFinite(input.end) || input.end <= input.start) {
   916→    return { ok: false as const, status: 422, code: "invalid_time", message: "The event end must be after its start." };
   917→  }
   918→  const ev: CalendarEvent = {
   919→    id: `ev-w${Math.floor(Date.now() / 1000) % 100000}`,
   920→    title,
   921→    type: input.type,
   922→    start: input.start,
   923→    end: input.end,
   924→    organizerId: SESSION_USER_ID,
   925→    location: input.location?.trim() || undefined,
   926→  };
   927→  store().events.push(ev);
   928→  writeAudit({ actorId: SESSION_USER_ID, action: "Calendar event created", resource: title, result: "Success", source: "Workspace", context: `${ev.type} added from My Calendar` });
   929→  return { ok: true as const, event: ev, message: `"${title}" was added to your calendar.` };
   930→}
