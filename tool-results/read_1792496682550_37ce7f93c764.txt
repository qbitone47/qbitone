    81→  // routing authority is the Approval Matrix (screen 09); the "employee"

    88→  }
    89→  const queue = s.approvals.filter((a) => a.status === "Pending");
    90→  const history = s.approvals.filter((a) => a.status !== "Pending");
    91→  return { queue, history, note: null };
    92→}

   136→export function decideApproval(
   137→  role: WorkspaceRole,
   138→  input: { id: string; decision: "Approved" | "Rejected" | "Changes requested"; note?: string; stepUpPassword?: string },
   139→) {
   140→  const s = store();
   141→  const req = s.approvals.find((a) => a.id === input.id);
   142→  if (!req) return { ok: false as const, status: 404, code: "not_found", message: "Approval request not found." };
   143→  if (req.status !== "Pending") {
   144→    return { ok: false as const, status: 409, code: "already_decided", message: `This request was already ${req.status.toLowerCase()} by another approver.` };
   145→  }
   146→
   147→  // Server-side authorization: employee boundaries hold no approval authority.
   148→  if (role === "employee" || role === "sales-exec") {
   149→    writeAudit({ actorId: SESSION_USER_ID, action: "Approval decision denied", resource: req.title, result: "Blocked", severity: "Warning", context: "Role holds no approval authority for this request type" });
   150→    return { ok: false as const, status: 403, code: "forbidden", message: "Your role does not hold approval authority for this request." };
   151→  }
   152→
   153→  // Step-up authentication for sensitive approvals (server-enforced).
   154→  if (req.stepUpRequired) {
   155→    if (!input.stepUpPassword || input.stepUpPassword.length < 8) {
   156→      return { ok: false as const, status: 401, code: "step_up_required", message: "Reauthentication is required for this approval." };
   157→    }
   158→  }
   159→
   160→  // High-value decisions require a note for the audit trail.
   161→  if (input.decision === "Rejected" && (!input.note || input.note.trim().length < 4)) {
   162→    return { ok: false as const, status: 422, code: "note_required", message: "A reason is required when rejecting a request." };
   163→  }
   164→
   165→  req.status = input.decision;
   166→  req.decidedAt = Date.now();
   167→  req.decidedBy = SESSION_USER_ID;
   168→  req.decisionNote = input.note?.trim() || undefined;
   169→  req.history = [
   170→    ...(req.history ?? []),
   171→    { at: Date.now(), actor: "Sagar Singh", event: input.decision + (input.note ? ` — ${input.note.trim()}` : "") },
   172→  ];
   173→
   174→  writeAudit({
   175→    actorId: SESSION_USER_ID,
   176→    action: `Approval ${input.decision.toLowerCase()}`,
   177→    resource: `${req.kind} · ${req.title}`,
   178→    result: "Success",
   179→    severity: req.stepUpRequired ? "High" : "Info",
   180→    source: "Web · Mumbai",
   181→    context: req.stepUpRequired
   182→      ? "Step-up authentication verified before the decision"
   183→      : `Routed via Approval Matrix (${req.rule})`,
   184→  });
   185→
   186→  return { ok: true as const, approval: req };
   187→}

   197→export function createTask(role: WorkspaceRole, input: { title: string; detail?: string; priority?: Task["priority"]; due?: number | null }) {
   198→  const title = input.title?.trim() ?? "";
   199→  if (title.length < 4) return { ok: false as const, status: 422, code: "invalid_title", message: "Task title must be at least 4 characters." };
   200→  if (title.length > 140) return { ok: false as const, status: 422, code: "invalid_title", message: "Task title must be 140 characters or fewer." };
   201→  if (input.due !== null && input.due !== undefined && Number.isNaN(input.due)) {
   202→    return { ok: false as const, status: 422, code: "invalid_due", message: "Due date is invalid." };
   203→  }
   204→
   205→  const t: Task = {
   206→    id: `task-${Math.floor(Date.now() / 1000) % 100000}`,
   207→    title,
   208→    detail: input.detail?.trim() || "",
   209→    status: "Not Started",
   210→    priority: input.priority ?? "Medium",
   211→    due: input.due ?? null,
   212→    source: "Manual",
   213→    ownerId: SESSION_USER_ID,
   214→    assigneeId: SESSION_USER_ID,
   215→  };
   216→  store().tasks.unshift(t);
   217→  writeAudit({ actorId: SESSION_USER_ID, action: "Task created", resource: t.title, result: "Success", context: "Created from Employee Home quick action" });
   218→  return { ok: true as const, task: t };
   219→}

   221→const TASK_TRANSITIONS: Record<string, Task["status"][]> = {
   222→  "Not Started": ["In Progress", "Cancelled"],
   223→  "In Progress": ["Waiting", "Blocked", "Completed", "Cancelled"],
   224→  Blocked: ["In Progress", "Waiting", "Cancelled"],
   225→  Waiting: ["In Progress", "Blocked", "Cancelled"],
   226→  Completed: ["In Progress"],
   227→  Cancelled: [],
   228→};

   230→export function updateTask(role: WorkspaceRole, input: { id: string; status: Task["status"] }) {
   231→  const t = store().tasks.find((x) => x.id === input.id);
   232→  if (!t) return { ok: false as const, status: 404, code: "not_found", message: "Task not found." };
   233→  // Authorization: only the assignee transitions their own task.
   234→  if (t.assigneeId !== SESSION_USER_ID) {
   235→    writeAudit({ actorId: SESSION_USER_ID, action: "Task update denied", resource: t.title, result: "Blocked", severity: "Warning", context: "Caller is not the task assignee" });
   236→    return { ok: false as const, status: 403, code: "forbidden", message: "Only the task assignee can update this task." };
   237→  }
   238→  const allowed = TASK_TRANSITIONS[t.status] ?? [];
   239→  if (!allowed.includes(input.status)) {
   240→    return { ok: false as const, status: 409, code: "invalid_transition", message: `A task cannot move from ${t.status} to ${input.status}.` };
   241→  }
   242→  t.status = input.status;
   243→  if (input.status === "Completed") t.completedAt = Date.now();
   244→  writeAudit({ actorId: SESSION_USER_ID, action: "Task status changed", resource: t.title, result: "Success", context: `${t.status === "Completed" ? "Marked completed" : `Moved to ${input.status}`}` });
   245→  return { ok: true as const, task: t };
   246→}

   254→export function listMail(role: WorkspaceRole, params: { folder?: string; q?: string; unreadOnly?: boolean; label?: string; limit?: number }) {
   255→  const s = store();
   256→  const folder = params.folder ?? "inbox";
   257→  let items = folder === "all" ? [...s.mail] : s.mail.filter((m) => m.folder === folder);
   258→  // Authorization scope: the session mailbox only (demo: single-user store).
   259→  const q = params.q?.trim().toLowerCase();
   260→  if (q) {
   261→    items = items.filter((m) => {
   262→      const fromName = m.fromId ? employees.find((e) => e.id === m.fromId)?.name ?? "" : m.fromExternal?.name ?? "";
   263→      const fromEmail = m.fromId ? employees.find((e) => e.id === m.fromId)?.email ?? "" : m.fromExternal?.email ?? "";
   264→      return [m.subject, m.preview, m.body, fromName, fromEmail, ...(m.to ?? [])].join(" ").toLowerCase().includes(q);
   265→    });
   266→  }
   267→  if (params.unreadOnly) items = items.filter((m) => !m.read);
   268→  if (params.label) items = items.filter((m) => m.labels?.includes(params.label as never));
   269→  items.sort((a, b) => b.ts - a.ts);
   270→  const limit = params.limit && params.limit > 0 ? Math.min(params.limit, 200) : 100;
   271→  return { items: items.slice(0, limit), total: items.length, scopeNote: "Search is scoped to your authorized mailbox server-side." };
   272→}

   274→export function sendMail(role: WorkspaceRole, input: { to: string[]; cc?: string[]; bcc?: string[]; subject: string; body: string; attachments?: { name: string; sizeKb: number; type: string }[] }) {
   275→  const to = (input.to ?? []).map((v) => v.trim()).filter(Boolean);
   276→  const cc = (input.cc ?? []).map((v) => v.trim()).filter(Boolean);
   277→  if (to.length === 0) return { ok: false as const, status: 422, code: "no_recipients", message: "Add at least one recipient." };
   278→  if (to.some((v) => !emailLike(v))) return { ok: false as const, status: 422, code: "invalid_recipient", message: "One or more recipient addresses are not valid." };
   279→  if (!input.subject?.trim() && !input.body?.trim()) {
   280→    return { ok: false as const, status: 422, code: "empty_message", message: "Add a subject or a message before sending." };
   281→  }
   282→  const atts = input.attachments ?? [];
   283→  if (atts.length > MAX_ATTACHMENTS) return { ok: false as const, status: 422, code: "too_many_attachments", message: `A message can carry at most ${MAX_ATTACHMENTS} attachments.` };
   284→  const totalKb = atts.reduce((n, a) => n + (a.sizeKb || 0), 0);
   285→  if (totalKb > MAX_TOTAL_KB) return { ok: false as const, status: 422, code: "attachment_too_large", message: "Attachments exceed the 10 MB message limit." };
   286→
   287→  const msg: MailMessage = {
   288→    id: `s-${Math.floor(Date.now() / 1000) % 100000}`,
   289→    folder: "sent",
   290→    fromId: SESSION_USER_ID,
   291→    to,
   292→    cc: cc.length ? cc : undefined,
   293→    subject: input.subject?.trim() || "(no subject)",
   294→    preview: (input.body ?? "").replace(/\s+/g, " ").trim().slice(0, 140),
   295→    body: input.body ?? "",
   296→    ts: Date.now(),
   297→    read: true,
   298→    sendStatus: "Sent",
   299→    attachments: atts.length
   300→      ? atts.map((a) => ({ name: a.name, sizeKb: a.sizeKb, type: (a.type?.toUpperCase() || "BIN") as never }))
   301→      : undefined,
   302→  };
   303→  store().mail.unshift(msg);
   304→  writeAudit({ actorId: SESSION_USER_ID, action: "Mail sent", resource: msg.subject, result: "Success", context: `${to.length} recipient(s)${cc.length ? `, ${cc.length} cc` : ""}${atts.length ? `, ${atts.length} attachment(s)` : ""}` });
   305→  return { ok: true as const, message: msg };
   306→}

   308→export function saveDraft(role: WorkspaceRole, input: Partial<MailMessage> & { draftId?: string }) {
   309→  const s = store();
   310→  if (input.draftId) {
   311→    const existing = s.mail.find((m) => m.id === input.draftId && m.folder === "drafts");
   312→    if (existing) {
   313→      Object.assign(existing, {
   314→        to: input.to ?? existing.to,
   315→        cc: input.cc ?? existing.cc,
   316→        subject: input.subject ?? existing.subject,
   317→        body: input.body ?? existing.body,
   318→        preview: (input.body ?? existing.body).replace(/\s+/g, " ").trim().slice(0, 140),
   319→        ts: Date.now(),
   320→      });
   321→      return { ok: true as const, draft: existing };
   322→    }
   323→  }
   324→  const draft: MailMessage = {
   325→    id: `d-${Math.floor(Date.now() / 1000) % 100000}`,
   326→    folder: "drafts",
   327→    fromId: SESSION_USER_ID,
   328→    to: input.to ?? [],
   329→    subject: input.subject ?? "",
   330→    preview: (input.body ?? "").replace(/\s+/g, " ").trim().slice(0, 140),
   331→    body: input.body ?? "",
   332→    ts: Date.now(),
   333→    read: true,
   334→  };
   335→  s.mail.unshift(draft);
   336→  return { ok: true as const, draft };
   337→}

   339→export function retryMail(role: WorkspaceRole, id: string) {
   340→  const m = store().mail.find((x) => x.id === id);
   341→  if (!m || m.folder !== "sent") return { ok: false as const, status: 404, code: "not_found", message: "Message not found in Sent." };
   342→  if (m.sendStatus !== "Failed" && m.sendStatus !== "Pending") {
   343→    return { ok: false as const, status: 409, code: "not_retryable", message: `This message is already ${(m.sendStatus ?? "sent").toLowerCase()}.` };
   344→  }
   345→  m.sendStatus = "Sent";
   346→  m.sendDetail = undefined;
   347→  writeAudit({ actorId: SESSION_USER_ID, action: "Mail retry succeeded", resource: m.subject, result: "Success", context: "Delivery retried after failure" });
   348→  return { ok: true as const, message: m };
   349→}

   366→const SENSITIVE_DOC_ACTIONS = new Set(["download", "share"]);
   367→
   368→export function documentAction(
   369→  role: WorkspaceRole,
   370→  input: { action: "download" | "share" | "rename" | "move" | "upload"; docId?: string; name?: string; category?: string; toEmployeeId?: string; fileMeta?: { name: string; sizeKb: number; type: string } },
   371→) {
   372→  const s = store();
   373→
   374→  if (input.action === "upload") {
   375→    const f = input.fileMeta;
   376→    if (!f || !f.name?.trim()) return { ok: false as const, status: 422, code: "invalid_file", message: "Choose a file to upload." };
   377→    if (f.sizeKb > MAX_TOTAL_KB) return { ok: false as const, status: 422, code: "file_too_large", message: "Files above 10 MB must go through the enterprise storage vault." };
   378→    const doc: WorkspaceDoc = {
   379→      id: `doc-${Math.floor(Date.now() / 1000) % 100000}`,
   380→      name: f.name,
   381→      type: (f.type?.toUpperCase() || "DOCX") as WorkspaceDoc["type"],
   382→      category: "Personal",
   383→      ownerId: SESSION_USER_ID,
   384→      sizeKb: f.sizeKb,
   385→      modified: Date.now(),
   386→      sensitivity: "Internal",
   387→      description: "Uploaded from My Documents. File content remains in your enterprise storage; Qubit One holds the record.",
   388→    };
   389→    s.documents.unshift(doc);
   390→    writeAudit({ actorId: SESSION_USER_ID, action: "Document uploaded", resource: doc.name, result: "Success", context: "My Documents · metadata record only" });
   391→    return { ok: true as const, doc };
   392→  }
   393→
   394→  const doc = s.documents.find((d) => d.id === input.docId);
   395→  if (!doc) return { ok: false as const, status: 404, code: "not_found", message: "Document not found." };
   396→
   397→  if (SENSITIVE_DOC_ACTIONS.has(input.action) && doc.sensitivity === "Restricted") {
   398→    writeAudit({ actorId: SESSION_USER_ID, action: `Document ${input.action} denied`, resource: doc.name, result: "Blocked", severity: "Warning", context: "Restricted document — HR-managed access only" });
   399→    return { ok: false as const, status: 403, code: "restricted", message: "This document is restricted. Access is managed by HR — request access from the document owner." };
   400→  }
   401→
   402→  if (input.action === "download") {
   403→    writeAudit({ actorId: SESSION_USER_ID, action: "Document downloaded", resource: doc.name, result: "Success", context: `Sensitivity: ${doc.sensitivity}` });
   404→    return { ok: true as const, doc };
   405→  }
   406→
   407→  if (input.action === "share") {
   408→    if (!input.toEmployeeId || !employees.some((e) => e.id === input.toEmployeeId)) {
   409→      return { ok: false as const, status: 422, code: "invalid_recipient", message: "Choose a colleague to share with." };
   410→    }
   411→    if (doc.sensitivity === "Confidential" && role === "employee") {
   412→      return { ok: false as const, status: 403, code: "forbidden", message: "Confidential documents cannot be shared from this role." };
   413→    }
   414→    writeAudit({ actorId: SESSION_USER_ID, action: "Document shared", resource: doc.name, result: "Success", context: `Shared with ${employees.find((e) => e.id === input.toEmployeeId)?.name}` });
   415→    return { ok: true as const, doc };
   416→  }
   417→
   418→  if (doc.ownerId !== SESSION_USER_ID) {
   419→    return { ok: false as const, status: 403, code: "forbidden", message: "Only the owner can rename or move this document." };
   420→  }
   421→  if (input.action === "rename") {
   422→    const name = input.name?.trim();
   423→    if (!name || name.length < 3) return { ok: false as const, status: 422, code: "invalid_name", message: "Enter a document name (3+ characters)." };
   424→    doc.name = name;
   425→  } else if (input.action === "move") {
   426→    const cat = input.category as WorkspaceDoc["category"];
   427→    if (!["Personal", "Shared with me", "HR", "Company", "Workflow"].includes(cat)) {
   428→      return { ok: false as const, status: 422, code: "invalid_category", message: "Choose a valid category." };
   429→    }
   430→    doc.category = cat;
   431→  }
   432→  return { ok: true as const, doc };
   433→}

   442→const EDITABLE_PROFILE_FIELDS = new Set(["phone", "about", "skills"]);
   443→
   444→export function updateProfile(role: WorkspaceRole, patch: Record<string, unknown>) {
   445→  const keys = Object.keys(patch);
   446→  const illegal = keys.filter((k) => !EDITABLE_PROFILE_FIELDS.has(k));
   447→  if (illegal.length > 0) {
   448→    writeAudit({ actorId: SESSION_USER_ID, action: "Profile update denied", resource: illegal.join(", "), result: "Blocked", severity: "Warning", context: "Attempted edit of company-controlled fields" });
   449→    return { ok: false as const, status: 403, code: "company_controlled", message: `These fields are managed by authorized HR workflows: ${illegal.join(", ")}.` };
   450→  }
   451→  if (patch.phone !== undefined) {
   452→    const phone = String(patch.phone).trim();
   453→    if (!/^\+91\s\d{5}\s\d{5}$/.test(phone)) {
   454→      return { ok: false as const, status: 422, code: "invalid_phone", message: "Use the format +91 98200 12345." };
   455→    }
   456→  }
   457→  if (patch.about !== undefined && String(patch.about).length > 280) {
   458→    return { ok: false as const, status: 422, code: "invalid_about", message: "About must be 280 characters or fewer." };
   459→  }
   460→  if (patch.skills !== undefined && !Array.isArray(patch.skills)) {
   461→    return { ok: false as const, status: 422, code: "invalid_skills", message: "Skills must be a list." };
   462→  }
   463→  writeAudit({ actorId: SESSION_USER_ID, action: "Profile updated", resource: keys.join(", "), result: "Success", context: "Self-service fields only" });
   464→  return { ok: true as const, updated: keys };
   465→}

   541→export function cancelMeeting(role: WorkspaceRole, input: { id: string; reason: string }) {
   542→  const s = store();
   543→  const m = s.meetings.find((x) => x.id === input.id);
   544→  if (!m) return { ok: false as const, status: 404, code: "not_found", message: "Meeting not found." };
   545→  if (m.status !== "Scheduled") {
   546→    return {
   547→      ok: false as const,
   548→      status: 409,
   549→      code: "invalid_state",
   550→      message: m.status === "Cancelled" ? "This meeting was already cancelled." : `A meeting that is ${m.status.toLowerCase()} cannot be cancelled.`,
   551→    };
   552→  }
   553→  const isOrganizer = m.organizerId === SESSION_USER_ID;
   554→  const governance = role === "administrator" || role === "dept-head";
   555→  if (!isOrganizer && !governance) {
   556→    writeAudit({ actorId: SESSION_USER_ID, action: "Meeting cancellation denied", resource: `${m.title} (${m.id})`, result: "Blocked", severity: "Warning", source: "Workspace", context: "Only the organizer — or an administrator / department head — can cancel a meeting" });
   557→    return { ok: false as const, status: 403, code: "forbidden", message: "Only the meeting organizer — or an administrator / department head — can cancel this meeting." };
   558→  }
   559→  const reason = input.reason?.trim() ?? "";
   560→  if (reason.length < 4) {
   561→    return { ok: false as const, status: 422, code: "reason_required", message: "A short reason is required when cancelling a meeting." };
   562→  }
   563→  m.status = "Cancelled";
   564→  writeAudit({
   565→    actorId: SESSION_USER_ID,
   566→    action: "Meeting cancelled",
   567→    resource: `${m.title} (${m.id})`,
   568→    result: "Success",
   569→    severity: "Info",
   570→    source: "Workspace",
   571→    context: `${isOrganizer ? "Organizer" : "Governance"} cancellation — reason: ${reason}`,
   572→  });
   573→  return {
   574→    ok: true as const,
   575→    meeting: m,
   576→    message: `"${m.title}" was cancelled — ${m.participantIds.length} participant${m.participantIds.length === 1 ? "" : "s"} notified and the calendar updated.`,
   577→  };
   578→}

   625→// Calendar entries created from My Calendar (screen 44) — personal events
   626→// such as focus blocks and deadlines. Meeting invites WITH participants run
   627→// through the Meetings module; this records the entry on the session
   628→// calendar server-side so the screen holds no client-only business state.
   629→export function createCalendarEvent(role: WorkspaceRole, input: { title: string; type: CalendarEvent["type"]; start: number; end: number; location?: string }) {
   630→  const title = input.title?.trim() ?? "";
   631→  if (title.length < 3) return { ok: false as const, status: 422, code: "invalid_title", message: "Give the event a title (at least 3 characters)." };
   632→  const VALID_TYPES: CalendarEvent["type"][] = ["Meeting", "Deadline", "Company event", "Leave", "Focus time"];
   633→  if (!VALID_TYPES.includes(input.type)) {
   634→    return { ok: false as const, status: 422, code: "invalid_type", message: "Choose a valid event type." };
   635→  }
   636→  if (!Number.isFinite(input.start) || !Number.isFinite(input.end) || input.end <= input.start) {
   637→    return { ok: false as const, status: 422, code: "invalid_time", message: "The event end must be after its start." };
   638→  }
   639→  const ev: CalendarEvent = {
   640→    id: `ev-w${Math.floor(Date.now() / 1000) % 100000}`,
   641→    title,
   642→    type: input.type,
   643→    start: input.start,
   644→    end: input.end,
   645→    organizerId: SESSION_USER_ID,
   646→    location: input.location?.trim() || undefined,
   647→  };
   648→  store().events.push(ev);
   649→  writeAudit({ actorId: SESSION_USER_ID, action: "Calendar event created", resource: title, result: "Success", source: "Workspace", context: `${ev.type} added from My Calendar` });
   650→  return { ok: true as const, event: ev, message: `"${title}" was added to your calendar.` };
   651→}
