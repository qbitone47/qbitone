    43→interface MeetingRow {
    44→  id: string;
    45→  title: string;
    46→  accountId: string;
    47→  contactId?: string;
    48→  opportunityId?: string;
    49→  ownerId: string;
    50→  ownerName?: string;
    51→  startAt: number;
    52→  durationMin: number;
    53→  location: string;
    54→  isLink?: boolean;
    55→  status: "Scheduled" | "Completed" | "Cancelled";
    56→  agenda?: string;
    57→  notes?: string;
    58→  account?: { id: string; name: string } | null;
    59→  contact?: { id: string; name: string; designation?: string } | null;
    60→  opportunity?: { id: string; name: string } | null;
    61→}
    62→
    63→interface MeetingsPayload {
    64→  rows: MeetingRow[];
    65→  scope: "company" | "team" | "own" | null;
    66→}
    67→
    68→interface FormRefsPayload {
    69→  owners: readonly { id: string; name: string; title: string }[];
    70→  accounts: { id: string; name: string; ownerId: string }[];
    71→  contacts: { id: string; name: string; accountId: string }[];
    72→  opportunities: { id: string; name: string; accountId: string }[];
    73→  leads: { id: string; name: string; company: string }[];
    74→}
    75→
    76→// ---------- datetime-local helpers ----------
    77→
    78→const pad2 = (n: number) => String(n).padStart(2, "0");
    79→
    80→const toLocalInput = (ts: number) => {
    81→  const d = new Date(ts);
    82→  return `${d.getFullYear()}-${pad2(d.getMonth() + 1)}-${pad2(d.getDate())}T${pad2(d.getHours())}:${pad2(d.getMinutes())}`;
    83→};
    84→
    85→const fromLocalInput = (v: string): number | undefined => {
    86→  if (!v) return undefined;
    87→  const t = new Date(v).getTime();
    88→  return Number.isFinite(t) ? t : undefined;
    89→};
    90→
    91→const NONE = "none";
    92→const DURATIONS = ["30", "45", "60", "90"];

    95→
    96→interface MeetingActions {
    97→  onJoin: (m: MeetingRow) => void;
    98→  onReschedule: (m: MeetingRow) => void;
    99→  onCancel: (m: MeetingRow) => void;
   100→  onNotes: (m: MeetingRow) => void;
   101→}
   102→
   103→function MeetingItem({ m, actions }: { m: MeetingRow; actions: MeetingActions }) {
   104→  const cancelled = m.status === "Cancelled";
   105→  const done = m.status === "Completed";
   106→  const scheduled = m.status === "Scheduled";
   107→  const d = new Date(m.startAt);
   108→
   109→  return (
   110→    <li
   111→      className={cn(
   112→        "flex flex-col gap-3 rounded-lg border px-3.5 py-3 sm:flex-row sm:items-start",
   113→        cancelled ? "border-slate-200/70 bg-slate-50/40" : "border-slate-200/80 bg-white",
   114→      )}
   115→    >
   116→      <div
   117→        className={cn(
   118→          "flex h-10 w-10 shrink-0 flex-col items-center justify-center rounded-lg",
   119→          scheduled ? "bg-blue-50 text-blue-700" : "bg-slate-100 text-slate-500",
   120→        )}
   121→        aria-hidden
   122→      >
   123→        <span className="text-[10px] font-semibold uppercase leading-none">{d.toLocaleDateString("en-IN", { month: "short" })}</span>
   124→        <span className="text-[15px] font-bold leading-tight tabular-nums">{d.getDate()}</span>
   125→      </div>
   126→      <div className="min-w-0 flex-1">
   127→        <div className="flex flex-wrap items-center gap-2">
   128→          <p className={cn("min-w-0 flex-1 basis-48 text-[13.5px] font-medium", cancelled ? "text-slate-500" : "text-slate-800")}>
   129→            {m.title}
   130→          </p>
   131→          <StatusBadge status={m.status} size="sm" />
   132→        </div>
   133→        <p className="mt-1 flex flex-wrap items-center gap-x-1.5 gap-y-0.5 text-xs text-slate-400">
   134→          <span>{fmtDateTime(m.startAt)}</span>
   135→          {m.account && (
   136→            <>
   137→              <span aria-hidden>·</span>
   138→              <EntityLink screen="account-detail" id={m.account.id} label={m.account.name} kind="account" />
   139→            </>
   140→          )}
   141→          {m.contact && (
   142→            <>
   143→              <span aria-hidden>·</span>
   144→              <span>{m.contact.name}</span>
   145→            </>
   146→          )}
   147→          {m.opportunity && (
   148→            <>
   149→              <span aria-hidden>·</span>
   150→              <EntityLink screen="opportunity-detail" id={m.opportunity.id} label={m.opportunity.name} kind="opportunity" />
   151→            </>
   152→          )}
   153→          <span aria-hidden>·</span>
   154→          <span>{m.ownerName ?? m.ownerId}</span>
   155→          <span aria-hidden>·</span>
   156→          <span className="tabular-nums">{m.durationMin} min</span>
   157→          <span aria-hidden>·</span>
   158→          <span className="inline-flex items-center gap-1">
   159→            {m.location}
   160→            {m.isLink && <Link2 className="h-3 w-3 shrink-0" aria-label="Meeting link" />}
   161→          </span>
   162→        </p>
   163→        {scheduled && m.agenda && (
   164→          <p className="mt-1.5 text-xs leading-relaxed text-slate-500">
   165→            <span className="font-medium text-slate-600">Agenda:</span> {m.agenda}
   166→          </p>
   167→        )}
   168→        {done && m.notes && (
   169→          <p className="mt-1.5 text-xs leading-relaxed text-slate-500">
   170→            <span className="font-medium text-slate-600">Notes:</span> {m.notes}
   171→          </p>
   172→        )}
   173→        {cancelled && m.notes && (
   174→          <p className="mt-1.5 text-xs leading-relaxed text-slate-500">
   175→            <span className="font-medium text-slate-600">Reason note:</span> {m.notes}
   176→          </p>
   177→        )}
   178→      </div>
   179→      <div className="flex flex-wrap items-center gap-1.5 sm:shrink-0 sm:justify-end">
   180→        {scheduled && m.isLink && (
   181→          <Button size="sm" variant="outline" className="h-7 gap-1.5 px-2.5 text-xs" onClick={() => actions.onJoin(m)}>
   182→            <Video className="h-3 w-3" aria-hidden /> Join
   183→          </Button>
   184→        )}
   185→        {scheduled && (
   186→          <Button size="sm" variant="outline" className="h-7 gap-1.5 px-2.5 text-xs" onClick={() => actions.onReschedule(m)}>
   187→            <CalendarClock className="h-3 w-3" aria-hidden /> Reschedule
   188→          </Button>
   189→        )}
   190→        {scheduled && (
   191→          <Button
   192→            size="sm"
   193→            variant="outline"
   194→            className="h-7 gap-1.5 px-2.5 text-xs text-red-600 hover:bg-red-50 hover:text-red-700"
   195→            onClick={() => actions.onCancel(m)}
   196→          >
   197→            <XCircle className="h-3 w-3" aria-hidden /> Cancel
   198→          </Button>
   199→        )}
   200→        <Button size="sm" variant="outline" className="h-7 gap-1.5 px-2.5 text-xs" onClick={() => actions.onNotes(m)}>
   201→          <StickyNote className="h-3 w-3" aria-hidden /> Notes
   202→        </Button>
   203→      </div>
   204→    </li>
   205→  );
   206→}

   208→function GroupEmpty({ title, message }: { title: string; message: string }) {
   209→  return <EmptyState icon={<CalendarDays className="h-5 w-5" />} title={title} message={message} className="py-7" />;
   210→}

   214→const EMPTY_SCHED = {
   215→  title: "",
   216→  accountId: "",
   217→  contactId: NONE,
   218→  opportunityId: NONE,
   219→  startAt: "",
   220→  durationMin: "30",
   221→  location: "",
   222→  agenda: "",
   223→};
   224→
   225→export function CustomerMeetingsScreen() {
   226→  const { role } = useRole();
   227→  const { data, loading, error, forbidden, scope, reload } = useSalesSection<MeetingsPayload>("meetings");
   228→  const refs = useSalesSection<FormRefsPayload>("form-refs");
   229→
   230→  const combinedLoading = loading || refs.loading;
   231→  const combinedError = error ?? refs.error;
   232→  const retryAll = () => {
   233→    reload();
   234→    refs.reload();
   235→  };
   236→
   237→  // Schedule meeting dialog
   238→  const [schedOpen, setSchedOpen] = React.useState(false);
   239→  const [sched, setSched] = React.useState(EMPTY_SCHED);
   240→  const [schedError, setSchedError] = React.useState<string | null>(null);
   241→  const [submitting, setSubmitting] = React.useState(false);
   242→
   243→  // Reschedule dialog
   244→  const [reschedTarget, setReschedTarget] = React.useState<MeetingRow | null>(null);
   245→  const [reschedInput, setReschedInput] = React.useState("");
   246→  const [reschedError, setReschedError] = React.useState<string | null>(null);
   247→
   248→  // Cancel confirm
   249→  const [cancelTarget, setCancelTarget] = React.useState<MeetingRow | null>(null);
   250→  const [cancelError, setCancelError] = React.useState<string | null>(null);
   251→  const [cancelling, setCancelling] = React.useState(false);
   252→
   253→  // Notes dialog — allowed on any state; outcome lands on the customer timeline
   254→  const [notesTarget, setNotesTarget] = React.useState<MeetingRow | null>(null);
   255→  const [notesText, setNotesText] = React.useState("");
   256→  const [notesError, setNotesError] = React.useState<string | null>(null);
   257→
   258→  const rows = data?.rows ?? [];
   259→
   260→  const bounds = React.useMemo(() => {
   261→    const s = new Date();
   262→    s.setHours(0, 0, 0, 0);
   263→    const e = new Date();
   264→    e.setHours(23, 59, 59, 999);
   265→    return { t0: s.getTime(), t1: e.getTime() };
   266→  }, []);
   267→
   268→  const scheduled = rows.filter((m) => m.status === "Scheduled");
   269→  const today = scheduled.filter((m) => m.startAt >= bounds.t0 && m.startAt <= bounds.t1).sort((a, b) => a.startAt - b.startAt);
   270→  const upcoming = scheduled.filter((m) => m.startAt > bounds.t1).sort((a, b) => a.startAt - b.startAt);
   271→  const past = rows
   272→    .filter((m) => m.status === "Completed" || m.status === "Cancelled")
   273→    .sort((a, b) => b.startAt - a.startAt);
   274→
   275→  const accountOptions = refs.data?.accounts ?? [];
   276→  const contactOptions = sched.accountId ? (refs.data?.contacts ?? []).filter((c) => c.accountId === sched.accountId) : [];
   277→  const oppOptions = sched.accountId ? (refs.data?.opportunities ?? []).filter((o) => o.accountId === sched.accountId) : [];
   278→
   279→  const openSchedule = () => {
   280→    setSched(EMPTY_SCHED);
   281→    setSchedError(null);
   282→    setSchedOpen(true);
   283→  };
   284→
   285→  const submitSchedule = async () => {
   286→    const startAt = fromLocalInput(sched.startAt);
   287→    if (!sched.title.trim() || !sched.accountId || !startAt) {
   288→      setSchedError("Title, customer and date/time are required.");
   289→      return;
   290→    }
   291→    setSubmitting(true);
   292→    setSchedError(null);
   293→    try {
   294→      const res = await salesPost(
   295→        "meeting.create",
   296→        {
   297→          title: sched.title.trim(),
   298→          accountId: sched.accountId,
   299→          contactId: sched.contactId === NONE ? undefined : sched.contactId,
   300→          opportunityId: sched.opportunityId === NONE ? undefined : sched.opportunityId,
   301→          startAt,
   302→          durationMin: Number(sched.durationMin),
   303→          location: sched.location.trim() || undefined,
   304→          agenda: sched.agenda.trim() || undefined,
   305→        },
   306→        role,
   307→      );
   308→      toast({ title: res.message ?? "Meeting created — it also appears in the Meetings module." });
   309→      setSchedOpen(false);
   310→      setSched(EMPTY_SCHED);
   311→      reload();
   312→    } catch (err) {
   313→      setSchedError(err instanceof Error ? err.message : "The meeting could not be created.");
   314→    } finally {
   315→      setSubmitting(false);
   316→    }
   317→  };
   318→
   319→  const openReschedule = (m: MeetingRow) => {
   320→    setReschedInput(toLocalInput(Math.max(m.startAt, Date.now() + 3_600_000)));
   321→    setReschedError(null);
   322→    setReschedTarget(m);
   323→  };
   324→
   325→  const submitReschedule = async () => {
   326→    if (!reschedTarget) return;
   327→    const startAt = fromLocalInput(reschedInput);
   328→    if (!startAt) {
   329→      setReschedError("Pick a date and time.");
   330→      return;
   331→    }
   332→    setSubmitting(true);
   333→    setReschedError(null);
   334→    try {
   335→      const res = await salesPost("meeting.reschedule", { id: reschedTarget.id, startAt }, role);
   336→      toast({ title: res.message ?? "Meeting rescheduled." });
   337→      setReschedTarget(null);
   338→      reload();
   339→    } catch (err) {
   340→      setReschedError(err instanceof Error ? err.message : "The meeting could not be rescheduled.");
   341→    } finally {
   342→      setSubmitting(false);
   343→    }
   344→  };
   345→
   346→  const submitCancel = async () => {
   347→    if (!cancelTarget) return;
   348→    setCancelling(true);
   349→    setCancelError(null);
   350→    try {
   351→      const res = await salesPost("meeting.cancel", { id: cancelTarget.id }, role);
   352→      toast({ title: res.message ?? "Meeting cancelled." });
   353→      setCancelTarget(null);
   354→      reload();
   355→    } catch (err) {
   356→      setCancelError(err instanceof Error ? err.message : "The meeting could not be cancelled.");
   357→    } finally {
   358→      setCancelling(false);
   359→    }
   360→  };
   361→
   362→  const openNotes = (m: MeetingRow) => {
   363→    setNotesText(m.notes ?? "");
   364→    setNotesError(null);
   365→    setNotesTarget(m);
   366→  };
   367→
   368→  const submitNotes = async () => {
   369→    if (!notesTarget) return;
   370→    if (!notesText.trim()) {
   371→      setNotesError("Write the meeting notes first.");
   372→      return;
   373→    }
   374→    setSubmitting(true);
   375→    setNotesError(null);
   376→    try {
   377→      const res = await salesPost("meeting.notes", { id: notesTarget.id, notes: notesText.trim() }, role);
   378→      toast({ title: res.message ?? "Notes saved to the meeting and the customer timeline." });
   379→      setNotesTarget(null);
   380→      reload();
   381→    } catch (err) {
   382→      setNotesError(err instanceof Error ? err.message : "The notes could not be saved.");
   383→    } finally {
   384→      setSubmitting(false);
   385→    }
   386→  };
   387→
   388→  if (forbidden) {
   389→    return (
   390→      <SalesGate screenName="Customer Meetings">
   391→        <></>
   392→      </SalesGate>
   393→    );
   394→  }
   395→
   396→  const actions: MeetingActions = {
   397→    onJoin: () =>
   398→      toast({ title: "The meeting link opens in the enterprise Meetings module (demo)" }),
   399→    onReschedule: openReschedule,
   400→    onCancel: (m) => {
   401→      setCancelError(null);
   402→      setCancelTarget(m);
   403→    },
   404→    onNotes: openNotes,
   405→  };
   406→
   407→  return (
   408→    <div>
   409→      <PageHeader
   410→        crumbs={[{ label: "Sales" }, { label: "Customer Meetings" }]}
   411→        title="Customer Meetings"
   412→        description="The sales meeting workspace — connected to the enterprise Meetings & Collaboration module, not a second calendar."
   413→        actions={
   414→          <Button size="sm" className="h-9 gap-1.5" onClick={openSchedule}>
   415→            <Plus className="h-3.5 w-3.5" aria-hidden /> Schedule meeting
   416→          </Button>
   417→        }
   418→      />
   419→
   420→      {combinedError && <SalesErrorBanner message={combinedError} onRetry={retryAll} />}
   421→      {combinedLoading && <SalesSkeleton rows={4} />}
   422→
   423→      {data && !combinedLoading && (
   424→        <div className="space-y-4">
   425→          <div className="qubit-card space-y-1 p-4">
   426→            <p className="text-xs tabular-nums text-muted-foreground" role="status">
   427→              {rows.length} meeting{rows.length === 1 ? "" : "s"} in scope · {scheduled.length} scheduled
   428→            </p>
   429→            <ScopeNote scope={scope} what="meetings" />
   430→          </div>
   431→
   432→          {/* Today */}
   433→          <SectionCard title={`Today (${today.length})`} description="Customer meetings scheduled for today, earliest first.">
   434→            {today.length === 0 ? (
   435→              <GroupEmpty
   436→                title="No meetings today"
   437→                message="The calendar is clear — schedule one from a customer, contact or opportunity."
   438→              />
   439→            ) : (
   440→              <ul className="space-y-2.5">
   441→                {today.map((m) => (
   442→                  <MeetingItem key={m.id} m={m} actions={actions} />
   443→                ))}
   444→              </ul>
   445→            )}
   446→          </SectionCard>
   447→
   448→          {/* Upcoming */}
   449→          <SectionCard title={`Upcoming (${upcoming.length})`} description="Confirmed meetings after today, soonest first.">
   450→            {upcoming.length === 0 ? (
   451→              <GroupEmpty title="Nothing scheduled ahead" message="New meetings start here or from any account, contact or opportunity record." />
   452→            ) : (
   453→              <ul className="space-y-2.5">
   454→                {upcoming.map((m) => (
   455→                  <MeetingItem key={m.id} m={m} actions={actions} />
   456→                ))}
   457→              </ul>
   458→            )}
   459→          </SectionCard>
   460→
   461→          {/* Past & cancelled — history with honesty */}
   462→          <SectionCard
   463→            title={`Past & cancelled (${past.length})`}
   464→            description="Completed meetings keep their notes on the customer timeline; cancelled ones stay visible with their reason."
   465→          >
   466→            {past.length === 0 ? (
   467→              <GroupEmpty title="No meeting history yet" message="Completed and cancelled meetings are kept here for the record." />
   468→            ) : (
   469→              <ul className="space-y-2.5">
   470→                {past.map((m) => (
   471→                  <MeetingItem key={m.id} m={m} actions={actions} />
   472→                ))}
   473→              </ul>
   474→            )}
   475→          </SectionCard>
   476→
   477→          <p className="mt-4 flex items-start gap-1.5 text-[11px] leading-relaxed text-slate-400">
   478→            <BellRing className="mt-0.5 h-3 w-3 shrink-0" aria-hidden />
   479→            <span>Reminders ride the enterprise notification rules (screen 10) — no separate reminder engine.</span>
   480→          </p>
   481→        </div>
   482→      )}
   483→
   484→      {!combinedLoading && !data && !combinedError && (
   485→        <EmptyState
   486→          icon={<CalendarDays />}
   487→          title="Customer Meetings could not load"
   488→          message="The sales service returned no data. Retry, or check the service status."
   489→          action={
   490→            <Button size="sm" onClick={reload}>
   491→              Retry
   492→            </Button>
   493→          }
   494→        />
   495→      )}
   496→
   497→      {/* Schedule meeting — one calendar; creation is audited server-side */}
   498→      <FormDialog
   499→        open={schedOpen}
   500→        onOpenChange={(o) => {
   501→          if (!o) setSchedOpen(false);
   502→        }}
   503→        title="Schedule meeting"
   504→        description="One calendar — the meeting also appears in the enterprise Meetings & Collaboration module."
   505→        onSubmit={submitSchedule}
   506→        submitLabel="Schedule meeting"
   507→        submitting={submitting}
   508→        wide
   509→      >
   510→        <Field label="Title" required>
   511→          <Input
   512→            required
   513→            value={sched.title}
   514→            onChange={(e) => setSched((s) => ({ ...s, title: e.target.value }))}
   515→            placeholder="e.g. Acme Phase 2 — kickoff planning"
   516→            aria-invalid={Boolean(schedError)}
   517→          />
   518→        </Field>
   519→        <div className="grid gap-4 sm:grid-cols-2">
   520→          <Field label="Customer account" required>
   521→            <Select
   522→              value={sched.accountId}
   523→              onValueChange={(v) => setSched((s) => ({ ...s, accountId: v, contactId: NONE, opportunityId: NONE }))}
   524→            >
   525→              <SelectTrigger className="w-full" aria-label="Customer account">
   526→                <SelectValue placeholder="Choose account" />
   527→              </SelectTrigger>
   528→              <SelectContent>
   529→                {accountOptions.length === 0 && (
   530→                  <SelectItem value="__empty" disabled>
   531→                    No accounts available
   532→                  </SelectItem>
   533→                )}
   534→                {accountOptions.map((a) => (
   535→                  <SelectItem key={a.id} value={a.id}>
   536→                    {a.name}
   537→                  </SelectItem>
   538→                ))}
   539→              </SelectContent>
   540→            </Select>
   541→          </Field>
   542→          <Field label="Contact" hint="Contacts filter to the chosen account.">
   543→            <Select
   544→              value={sched.contactId}
   545→              onValueChange={(v) => setSched((s) => ({ ...s, contactId: v }))}
   546→              disabled={!sched.accountId}
   547→            >
   548→              <SelectTrigger className="w-full" aria-label="Contact">
   549→                <SelectValue placeholder={sched.accountId ? "Optional" : "Choose an account first"} />
   550→              </SelectTrigger>
   551→              <SelectContent>
   552→                <SelectItem value={NONE}>No contact</SelectItem>
   553→                {contactOptions.length === 0 && sched.accountId && (
   554→                  <SelectItem value="__empty-ct" disabled>
   555→                    No contacts on file for this account
   556→                  </SelectItem>
   557→                )}
   558→                {contactOptions.map((c) => (
   559→                  <SelectItem key={c.id} value={c.id}>
   560→                    {c.name}
   561→                  </SelectItem>
   562→                ))}
   563→              </SelectContent>
   564→            </Select>
   565→          </Field>
   566→        </div>
   567→        <Field label="Opportunity" hint="Optional — links the meeting to a deal on the chosen account.">
   568→          <Select
   569→            value={sched.opportunityId}
   570→            onValueChange={(v) => setSched((s) => ({ ...s, opportunityId: v }))}
   571→            disabled={!sched.accountId}
   572→          >
   573→            <SelectTrigger className="w-full" aria-label="Opportunity">
   574→              <SelectValue placeholder={sched.accountId ? "Optional" : "Choose an account first"} />
   575→            </SelectTrigger>
   576→            <SelectContent>
   577→              <SelectItem value={NONE}>Not linked to a deal</SelectItem>
   578→              {oppOptions.length === 0 && sched.accountId && (
   579→                <SelectItem value="__empty-op" disabled>
   580→                  No opportunities on file for this account
   581→                </SelectItem>
   582→              )}
   583→              {oppOptions.map((o) => (
   584→                <SelectItem key={o.id} value={o.id}>
   585→                  {o.name}
   586→                </SelectItem>
   587→              ))}
   588→            </SelectContent>
   589→          </Select>
   590→        </Field>
   591→        <div className="grid gap-4 sm:grid-cols-2">
   592→          <Field label="Date & time" required>
   593→            <Input
   594→              type="datetime-local"
   595→              required
   596→              value={sched.startAt}
   597→              onChange={(e) => setSched((s) => ({ ...s, startAt: e.target.value }))}
   598→              aria-invalid={Boolean(schedError)}
   599→            />
   600→          </Field>
   601→          <Field label="Duration" required>
   602→            <Select value={sched.durationMin} onValueChange={(v) => setSched((s) => ({ ...s, durationMin: v }))}>
   603→              <SelectTrigger className="w-full" aria-label="Duration">
   604→                <SelectValue />
   605→              </SelectTrigger>
   606→              <SelectContent>
   607→                {DURATIONS.map((d) => (
   608→                  <SelectItem key={d} value={d}>
   609→                    {d} minutes
   610→                  </SelectItem>
   611→                ))}
   612→              </SelectContent>
   613→            </Select>
   614→          </Field>
   615→        </div>
   616→        <Field label="Location" hint="Leave empty for a Teams meeting — the link lives in the enterprise Meetings module.">
   617→          <Input
   618→            value={sched.location}
   619→            onChange={(e) => setSched((s) => ({ ...s, location: e.target.value }))}
   620→            placeholder="Teams meeting"
   621→          />
   622→        </Field>
   623→        <Field label="Agenda">
   624→          <Textarea
   625→            rows={2}
   626→            value={sched.agenda}
   627→            onChange={(e) => setSched((s) => ({ ...s, agenda: e.target.value }))}
   628→            placeholder="What the meeting needs to decide or produce…"
   629→          />
   630→        </Field>
   631→        {schedError && (
   632→          <p role="alert" className="text-xs font-medium text-red-600">
   633→            {schedError}
   634→          </p>
   635→        )}
   636→      </FormDialog>
   637→
   638→      {/* Reschedule — only Scheduled meetings; server rejects past times */}
   639→      <FormDialog
   640→        open={Boolean(reschedTarget)}
   641→        onOpenChange={(o) => {
   642→          if (!o) setReschedTarget(null);
   643→        }}
   644→        title="Reschedule meeting"
   645→        description={
   646→          reschedTarget
   647→            ? `"${reschedTarget.title}" — currently ${fmtDateTime(reschedTarget.startAt)}. The customer sees the new slot through the notification architecture.`
   648→            : undefined
   649→        }
   650→        onSubmit={submitReschedule}
   651→        submitLabel="Reschedule"
   652→        submitting={submitting}
   653→      >
   654→        <Field label="New date & time" required>
   655→          <Input
   656→            type="datetime-local"
   657→            required
   658→            value={reschedInput}
   659→            onChange={(e) => setReschedInput(e.target.value)}
   660→            aria-invalid={Boolean(reschedError)}
   661→          />
   662→        </Field>
   663→        {reschedError && (
   664→          <p role="alert" className="text-xs font-medium text-red-600">
   665→            {reschedError}
   666→          </p>
   667→        )}
   668→      </FormDialog>
   669→
   670→      {/* Cancel — destructive; confirmed; customer notified server-side */}
   671→      <ConfirmDialog
   672→        open={Boolean(cancelTarget)}
   673→        onOpenChange={(o) => {
   674→          if (!o) setCancelTarget(null);
   675→        }}
   676→        title="Cancel this meeting?"
   677→        description={
   678→          cancelTarget
   679→            ? `"${cancelTarget.title}" on ${fmtDateTime(cancelTarget.startAt)} will be cancelled. The customer is notified through the enterprise notification architecture — this cannot be undone.`
   680→            : ""
   681→        }
   682→        confirmLabel={cancelling ? "Cancelling…" : "Cancel meeting"}
   683→        destructive
   684→        onConfirm={submitCancel}
   685→      >
   686→        {cancelError && (
   687→          <p role="alert" className="mb-3 text-xs font-medium text-red-600">
   688→            {cancelError}
   689→          </p>
   690→        )}
   691→      </ConfirmDialog>
   692→
   693→      {/* Notes — allowed on any state; saved to the meeting and the customer timeline */}
   694→      <FormDialog
   695→        open={Boolean(notesTarget)}
   696→        onOpenChange={(o) => {
   697→          if (!o) setNotesTarget(null);
   698→        }}
   699→        title="Meeting notes"
   700→        description={
   701→          notesTarget
   702→            ? `"${notesTarget.title}" — notes are saved to the meeting and written to the customer timeline.`
   703→            : undefined
   704→        }
   705→        onSubmit={submitNotes}
   706→        submitLabel="Save notes"
   707→        submitting={submitting}
   708→      >
   709→        <Field label="Notes" required hint="Decisions, commitments, follow-ups — what the next person touching this account needs to know.">
   710→          <Textarea
   711→            required
   712→            rows={4}
   713→            value={notesText}
   714→            onChange={(e) => setNotesText(e.target.value)}
   715→            placeholder="e.g. Divya asked for a phased commercial structure; revision due this week."
   716→            aria-invalid={Boolean(notesError)}
   717→          />
   718→        </Field>
   719→        {notesError && (
   720→          <p role="alert" className="text-xs font-medium text-red-600">
   721→            {notesError}
   722→          </p>
   723→        )}
   724→      </FormDialog>
   725→    </div>
   726→  );
