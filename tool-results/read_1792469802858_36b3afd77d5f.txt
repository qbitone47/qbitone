    16→// ---------- Demo clock ----------
    17→export const WORKSPACE_NOW = Date.parse("2026-11-13T09:42:00+05:30");
    18→export const WORKSPACE_NOW_LABEL = "Friday, 13 November 2026";
    19→const MIN = 60_000;
    20→const HOUR = 3_600_000;
    21→const DAY = 86_400_000;
    22→
    23→/** Build an IST timestamp: dayOffset from today (0 = today), fixed wall-clock time. */
    24→const at = (dayOffset: number, hour: number, minute = 0): number =>
    25→  WORKSPACE_NOW - (9 * HOUR + 42 * MIN) + dayOffset * DAY + hour * HOUR + minute * MIN;
    26→
    27→export const ist = (ts: number) => new Date(ts + 330 * 60_000); // shift to IST, read via UTC getters
    28→
    29→export const fmtDateTime = (ts: number): string => {
    30→  const d = ist(ts);
    31→  const date = `${String(d.getUTCDate()).padStart(2, "0")} ${d.toLocaleDateString("en-IN", { month: "short", timeZone: "UTC" })} ${d.getUTCFullYear()}`;
    32→  const time = `${String(d.getUTCHours()).padStart(2, "0")}:${String(d.getUTCMinutes()).padStart(2, "0")}`;
    33→  return `${date}, ${time}`;
    34→};
    35→
    36→const MONTHS = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
    37→
    38→export const fmtDate = (ts: number): string => {
    39→  const d = ist(ts);
    40→  return `${String(d.getUTCDate()).padStart(2, "0")} ${MONTHS[d.getUTCMonth()]} ${d.getUTCFullYear()}`;
    41→};
    42→
    43→export const fmtTime = (ts: number): string => {
    44→  const d = ist(ts);
    45→  return `${String(d.getUTCHours()).padStart(2, "0")}:${String(d.getUTCMinutes()).padStart(2, "0")}`;
    46→};
    47→
    48→/** Stable relative label against the fixed demo clock. */
    49→export const relTime = (ts: number): string => {
    50→  const diff = WORKSPACE_NOW - ts;
    51→  if (diff < 0) {
    52→    const ahead = -diff;
    53→    if (ahead < HOUR) return `in ${Math.max(1, Math.round(ahead / MIN))} min`;
    54→    if (ahead < DAY) return `in ${Math.round(ahead / HOUR)} h`;
    55→    return `in ${Math.round(ahead / DAY)} d`;
    56→  }
    57→  if (diff < MIN) return "just now";
    58→  if (diff < HOUR) return `${Math.round(diff / MIN)} min ago`;
    59→  if (diff < DAY) return `${Math.round(diff / HOUR)} h ago`;
    60→  const days = Math.round(diff / DAY);
    61→  if (days === 1) return "yesterday";
    62→  if (days < 7) return `${days} days ago`;
    63→  return fmtDate(ts);
    64→};

   789→/** Older archive messages — generated deterministically so Mail Search
   790→ *  operates at a believable enterprise scale (spec §13). */
   791→function mulberry32(seed: number) {
   792→  return function () {
   793→    let t = (seed += 0x6d2b79f5);
   794→    t = Math.imul(t ^ (t >>> 15), t | 1);
   795→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
   796→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   797→  };
   798→}
