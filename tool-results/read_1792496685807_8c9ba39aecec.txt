   136→   136→export function decideApproval(
   137→   137→  role: WorkspaceRole,
   138→   138→  input: { id: string; decision: "Approved" | "Rejected" | "Changes requested"; note?: string; stepUpPassword?: string },
   139→   139→) {
   140→   140→  const s = store();
   141→   141→  const req = s.approvals.find((a) => a.id === input.id);
   142→   142→  if (!req) return { ok: false as const, status: 404, code: "not_found", message: "Approval request not found." };
   143→   143→  if (req.status !== "Pending") {
   144→   144→    return { ok: false as const, status: 409, code: "already_decided", message: `This request was already ${req.status.toLowerCase()} by another approver.` };
   145→   145→  }
   146→   146→
   147→   147→  // Server-side authorization: employee boundaries hold no approval authority.
   148→   148→  if (role === "employee" || role === "sales-exec") {
   149→   149→    writeAudit({ actorId: SESSION_USER_ID, action: "Approval decision denied", resource: req.title, result: "Blocked", severity: "Warning", context: "Role holds no approval authority for this request type" });
   150→   150→    return { ok: false as const, status: 403, code: "forbidden", message: "Your role does not hold approval authority for this request." };
   151→   151→  }
   152→   152→
   153→   153→  // Step-up authentication for sensitive approvals (server-enforced).
   154→   154→  if (req.stepUpRequired) {
   155→   155→    if (!input.stepUpPassword || input.stepUpPassword.length < 8) {
   156→   156→      return { ok: false as const, status: 401, code: "step_up_required", message: "Reauthentication is required for this approval." };
   157→   157→    }
   158→   158→  }
   159→   159→
   160→   160→  // High-value decisions require a note for the audit trail.
   161→   161→  if (input.decision === "Rejected" && (!input.note || input.note.trim().length < 4)) {
   162→   162→    return { ok: false as const, status: 422, code: "note_required", message: "A reason is required when rejecting a request." };
   163→   163→  }
   164→   164→
   165→   165→  req.status = input.decision;
   166→   166→  req.decidedAt = Date.now();
   167→   167→  req.decidedBy = SESSION_USER_ID;
   168→   168→  req.decisionNote = input.note?.trim() || undefined;
   169→   169→  req.history = [
   170→   170→    ...(req.history ?? []),
   171→   171→    { at: Date.now(), actor: "Sagar Singh", event: input.decision + (input.note ? ` — ${input.note.trim()}` : "") },
   172→   172→  ];
   173→   173→
   174→   174→  writeAudit({
   175→   175→    actorId: SESSION_USER_ID,
   176→   176→    action: `Approval ${input.decision.toLowerCase()}`,
   177→   177→    resource: `${req.kind} · ${req.title}`,
   178→   178→    result: "Success",
   179→   179→    severity: req.stepUpRequired ? "High" : "Info",
   180→   180→    source: "Web · Mumbai",
   181→   181→    context: req.stepUpRequired
   182→   182→      ? "Step-up authentication verified before the decision"
   183→   183→      : `Routed via Approval Matrix (${req.rule})`,
   184→   184→  });
   185→   185→
   186→   186→  return { ok: true as const, approval: req };
   187→   187→}

   230→   230→export function updateTask(role: WorkspaceRole, input: { id: string; status: Task["status"] }) {
   231→   231→  const t = store().tasks.find((x) => x.id === input.id);
   232→   232→  if (!t) return { ok: false as const, status: 404, code: "not_found", message: "Task not found." };
   233→   233→  // Authorization: only the assignee transitions their own task.
   234→   234→  if (t.assigneeId !== SESSION_USER_ID) {
   235→   235→    writeAudit({ actorId: SESSION_USER_ID, action: "Task update denied", resource: t.title, result: "Blocked", severity: "Warning", context: "Caller is not the task assignee" });
   236→   236→    return { ok: false as const, status: 403, code: "forbidden", message: "Only the task assignee can update this task." };
   237→   237→  }
   238→   238→  const allowed = TASK_TRANSITIONS[t.status] ?? [];
   239→   239→  if (!allowed.includes(input.status)) {
   240→   240→    return { ok: false as const, status: 409, code: "invalid_transition", message: `A task cannot move from ${t.status} to ${input.status}.` };
   241→   241→  }
   242→   242→  t.status = input.status;
   243→   243→  if (input.status === "Completed") t.completedAt = Date.now();
   244→   244→  writeAudit({ actorId: SESSION_USER_ID, action: "Task status changed", resource: t.title, result: "Success", context: `${t.status === "Completed" ? "Marked completed" : `Moved to ${input.status}`}` });
   245→   245→  return { ok: true as const, task: t };
   246→   246→}

   254→   254→export function listMail(role: WorkspaceRole, params: { folder?: string; q?: string; unreadOnly?: boolean; label?: string; limit?: number }) {
   255→   255→  const s = store();
   256→   256→  const folder = params.folder ?? "inbox";
   257→   257→  let items = folder === "all" ? [...s.mail] : s.mail.filter((m) => m.folder === folder);
   258→   258→  // Authorization scope: the session mailbox only (demo: single-user store).
   259→   259→  const q = params.q?.trim().toLowerCase();
   260→   260→  if (q) {
   261→   261→    items = items.filter((m) => {
   262→   262→      const fromName = m.fromId ? employees.find((e) => e.id === m.fromId)?.name ?? "" : m.fromExternal?.name ?? "";
   263→   263→      const fromEmail = m.fromId ? employees.find((e) => e.id === m.fromId)?.email ?? "" : m.fromExternal?.email ?? "";
   264→   264→      return [m.subject, m.preview, m.body, fromName, fromEmail, ...(m.to ?? [])].join(" ").toLowerCase().includes(q);
   265→   265→    });
   266→   266→  }
   267→   267→  if (params.unreadOnly) items = items.filter((m) => !m.read);
   268→   268→  if (params.label) items = items.filter((m) => m.labels?.includes(params.label as never));
   269→   269→  items.sort((a, b) => b.ts - a.ts);
   270→   270→  const limit = params.limit && params.limit > 0 ? Math.min(params.limit, 200) : 100;
   271→   271→  return { items: items.slice(0, limit), total: items.length, scopeNote: "Search is scoped to your authorized mailbox server-side." };
   272→   272→}

   274→   274→export function sendMail(role: WorkspaceRole, input: { to: string[]; cc?: string[]; bcc?: string[]; subject: string; body: string; attachments?: { name: string; sizeKb: number; type: string }[] }) {
   275→   275→  const to = (input.to ?? []).map((v) => v.trim()).filter(Boolean);
   276→   276→  const cc = (input.cc ?? []).map((v) => v.trim()).filter(Boolean);
   277→   277→  if (to.length === 0) return { ok: false as const, status: 422, code: "no_recipients", message: "Add at least one recipient." };
   278→   278→  if (to.some((v) => !emailLike(v))) return { ok: false as const, status: 422, code: "invalid_recipient", message: "One or more recipient addresses are not valid." };
   279→   279→  if (!input.subject?.trim() && !input.body?.trim()) {
   280→   280→    return { ok: false as const, status: 422, code: "empty_message", message: "Add a subject or a message before sending." };
   281→   281→  }
   282→   282→  const atts = input.attachments ?? [];
   283→   283→  if (atts.length > MAX_ATTACHMENTS) return { ok: false as const, status: 422, code: "too_many_attachments", message: `A message can carry at most ${MAX_ATTACHMENTS} attachments.` };
   284→   284→  const totalKb = atts.reduce((n, a) => n + (a.sizeKb || 0), 0);
   285→   285→  if (totalKb > MAX_TOTAL_KB) return { ok: false as const, status: 422, code: "attachment_too_large", message: "Attachments exceed the 10 MB message limit." };
   286→   286→
   287→   287→  const msg: MailMessage = {
   288→   288→    id: `s-${Math.floor(Date.now() / 1000) % 100000}`,
   289→   289→    folder: "sent",
   290→   290→    fromId: SESSION_USER_ID,
   291→   291→    to,
   292→   292→    cc: cc.length ? cc : undefined,
   293→   293→    subject: input.subject?.trim() || "(no subject)",
   294→   294→    preview: (input.body ?? "").replace(/\s+/g, " ").trim().slice(0, 140),
   295→   295→    body: input.body ?? "",
   296→   296→    ts: Date.now(),
   297→   297→    read: true,
   298→   298→    sendStatus: "Sent",
   299→   299→    attachments: atts.length
   300→   300→      ? atts.map((a) => ({ name: a.name, sizeKb: a.sizeKb, type: (a.type?.toUpperCase() || "BIN") as never }))
   301→   301→      : undefined,
   302→   302→  };
   303→   303→  store().mail.unshift(msg);
   304→   304→  writeAudit({ actorId: SESSION_USER_ID, action: "Mail sent", resource: msg.subject, result: "Success", context: `${to.length} recipient(s)${cc.length ? `, ${cc.length} cc` : ""}${atts.length ? `, ${atts.length} attachment(s)` : ""}` });
   305→   305→  return { ok: true as const, message: msg };
   306→   306→}

   340→   340→  const m = store().mail.find((x) => x.id === id);
   341→   341→  if (!m || m.folder !== "sent") return { ok: false as const, status: 404, code: "not_found", message: "Message not found in Sent." };

   344→   344→  }

   346→   346→  m.sendDetail = undefined;

   541→   541→export function cancelMeeting(role: WorkspaceRole, input: { id: string; reason: string }) {
   542→   542→  const s = store();
   543→   543→  const m = s.meetings.find((x) => x.id === input.id);

   551→   551→    };
   552→   552→  }
   553→   553→  const isOrganizer = m.organizerId === SESSION_USER_ID;
   554→   554→  const governance = role === "administrator" || role === "dept-head";
   555→   555→  if (!isOrganizer && !governance) {

   559→   559→  const reason = input.reason?.trim() ?? "";
   560→   560→  if (reason.length < 4) {
   561→   561→    return { ok: false as const, status: 422, code: "reason_required", message: "A short reason is required when cancelling a meeting." };
   562→   562→  }
   563→   563→  m.status = "Cancelled";
   564→   564→  writeAudit({

   567→   567→    resource: `${m.title} (${m.id})`,
   568→   568→    result: "Success",
   569→   569→    severity: "Info",
   570→   570→    source: "Workspace",
   571→   571→    context: `${isOrganizer ? "Organizer" : "Governance"} cancellation — reason: ${reason}`,
   572→   572→  });
   573→   573→  return {
   574→   574→    ok: true as const,
   575→   575→    meeting: m,
   576→   576→    message: `"${m.title}" was cancelled — ${m.participantIds.length} participant${m.participantIds.length === 1 ? "" : "s"} notified and the calendar updated.`,
   577→   577→  };
