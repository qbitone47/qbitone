   144→   144→export function decideApproval(
   145→   145→  role: WorkspaceRole,
   146→   146→  input: { id: string; decision: "Approved" | "Rejected" | "Changes requested"; note?: string; stepUpPassword?: string },
   147→   147→) {
   148→   148→  const s = store();
   149→   149→  const req = s.approvals.find((a) => a.id === input.id);
   150→   150→  if (!req) return { ok: false as const, status: 404, code: "not_found", message: "Approval request not found." };
   151→   151→  if (req.status !== "Pending") {
   152→   152→    return { ok: false as const, status: 409, code: "already_decided", message: `This request was already ${req.status.toLowerCase()} by another approver.` };
   153→   153→  }
   154→   154→
   155→   155→  // Server-side authorization: employee boundaries hold no approval authority.
   156→   156→  if (role === "employee" || role === "sales-exec") {
   157→   157→    writeAudit({ actorId: SESSION_USER_ID, action: "Approval decision denied", resource: req.title, result: "Blocked", severity: "Warning", context: "Role holds no approval authority for this request type" });
   158→   158→    return { ok: false as const, status: 403, code: "forbidden", message: "Your role does not hold approval authority for this request." };
   159→   159→  }
   160→   160→
   161→   161→  // Step-up authentication for sensitive approvals (server-enforced).
   162→   162→  if (req.stepUpRequired) {
   163→   163→    if (!input.stepUpPassword || input.stepUpPassword.length < 8) {
   164→   164→      return { ok: false as const, status: 401, code: "step_up_required", message: "Reauthentication is required for this approval." };
   165→   165→    }
   166→   166→  }
   167→   167→
   168→   168→  // High-value decisions require a note for the audit trail.
   169→   169→  if (input.decision === "Rejected" && (!input.note || input.note.trim().length < 4)) {
   170→   170→    return { ok: false as const, status: 422, code: "note_required", message: "A reason is required when rejecting a request." };
   171→   171→  }
   172→   172→
   173→   173→  req.status = input.decision;
   174→   174→  req.decidedAt = Date.now();
   175→   175→  req.decidedBy = SESSION_USER_ID;
   176→   176→  req.decisionNote = input.note?.trim() || undefined;
   177→   177→  req.history = [
   178→   178→    ...(req.history ?? []),
   179→   179→    { at: Date.now(), actor: "Sagar Singh", event: input.decision + (input.note ? ` — ${input.note.trim()}` : "") },
   180→   180→  ];
   181→   181→
   182→   182→  writeAudit({
   183→   183→    actorId: SESSION_USER_ID,
   184→   184→    action: `Approval ${input.decision.toLowerCase()}`,
   185→   185→    resource: `${req.kind} · ${req.title}`,
   186→   186→    result: "Success",
   187→   187→    severity: req.stepUpRequired ? "High" : "Info",
   188→   188→    source: "Web · Mumbai",
   189→   189→    context: req.stepUpRequired
   190→   190→      ? "Step-up authentication verified before the decision"
   191→   191→      : `Routed via Approval Matrix (${req.rule})`,
   192→   192→  });
   193→   193→
   194→   194→  return { ok: true as const, approval: req };
   195→   195→}

   238→   238→export function updateTask(role: WorkspaceRole, input: { id: string; status: Task["status"] }) {
   239→   239→  const t = store().tasks.find((x) => x.id === input.id);
   240→   240→  if (!t) return { ok: false as const, status: 404, code: "not_found", message: "Task not found." };
   241→   241→  // Authorization: only the assignee transitions their own task.
   242→   242→  if (t.assigneeId !== SESSION_USER_ID) {
   243→   243→    writeAudit({ actorId: SESSION_USER_ID, action: "Task update denied", resource: t.title, result: "Blocked", severity: "Warning", context: "Caller is not the task assignee" });
   244→   244→    return { ok: false as const, status: 403, code: "forbidden", message: "Only the task assignee can update this task." };
   245→   245→  }
   246→   246→  const allowed = TASK_TRANSITIONS[t.status] ?? [];
   247→   247→  if (!allowed.includes(input.status)) {
   248→   248→    return { ok: false as const, status: 409, code: "invalid_transition", message: `A task cannot move from ${t.status} to ${input.status}.` };
   249→   249→  }
   250→   250→  t.status = input.status;
   251→   251→  if (input.status === "Completed") t.completedAt = Date.now();
   252→   252→  writeAudit({ actorId: SESSION_USER_ID, action: "Task status changed", resource: t.title, result: "Success", context: `${t.status === "Completed" ? "Marked completed" : `Moved to ${input.status}`}` });
   253→   253→  return { ok: true as const, task: t };
   254→   254→}

   262→   262→export function listMail(role: WorkspaceRole, params: { folder?: string; q?: string; unreadOnly?: boolean; label?: string; limit?: number }) {
   263→   263→  const s = store();
   264→   264→  const folder = params.folder ?? "inbox";
   265→   265→  let items = folder === "all" ? [...s.mail] : s.mail.filter((m) => m.folder === folder);
   266→   266→  // Authorization scope: the session mailbox only (demo: single-user store).
   267→   267→  const q = params.q?.trim().toLowerCase();
   268→   268→  if (q) {
   269→   269→    items = items.filter((m) => {
   270→   270→      const fromName = m.fromId ? employees.find((e) => e.id === m.fromId)?.name ?? "" : m.fromExternal?.name ?? "";
   271→   271→      const fromEmail = m.fromId ? employees.find((e) => e.id === m.fromId)?.email ?? "" : m.fromExternal?.email ?? "";
   272→   272→      return [m.subject, m.preview, m.body, fromName, fromEmail, ...(m.to ?? [])].join(" ").toLowerCase().includes(q);
   273→   273→    });
   274→   274→  }
   275→   275→  if (params.unreadOnly) items = items.filter((m) => !m.read);
   276→   276→  if (params.label) items = items.filter((m) => m.labels?.includes(params.label as never));
   277→   277→  items.sort((a, b) => b.ts - a.ts);
   278→   278→  const limit = params.limit && params.limit > 0 ? Math.min(params.limit, 200) : 100;
   279→   279→  return { items: items.slice(0, limit), total: items.length, scopeNote: "Search is scoped to your authorized mailbox server-side." };
   280→   280→}

   282→   282→export function sendMail(role: WorkspaceRole, input: { to: string[]; cc?: string[]; bcc?: string[]; subject: string; body: string; attachments?: { name: string; sizeKb: number; type: string }[] }) {
   283→   283→  const to = (input.to ?? []).map((v) => v.trim()).filter(Boolean);
   284→   284→  const cc = (input.cc ?? []).map((v) => v.trim()).filter(Boolean);
   285→   285→  if (to.length === 0) return { ok: false as const, status: 422, code: "no_recipients", message: "Add at least one recipient." };
   286→   286→  if (to.some((v) => !emailLike(v))) return { ok: false as const, status: 422, code: "invalid_recipient", message: "One or more recipient addresses are not valid." };
   287→   287→  if (!input.subject?.trim() && !input.body?.trim()) {
   288→   288→    return { ok: false as const, status: 422, code: "empty_message", message: "Add a subject or a message before sending." };
   289→   289→  }
   290→   290→  const atts = input.attachments ?? [];
   291→   291→  if (atts.length > MAX_ATTACHMENTS) return { ok: false as const, status: 422, code: "too_many_attachments", message: `A message can carry at most ${MAX_ATTACHMENTS} attachments.` };
   292→   292→  const totalKb = atts.reduce((n, a) => n + (a.sizeKb || 0), 0);
   293→   293→  if (totalKb > MAX_TOTAL_KB) return { ok: false as const, status: 422, code: "attachment_too_large", message: "Attachments exceed the 10 MB message limit." };
   294→   294→
   295→   295→  const msg: MailMessage = {
   296→   296→    id: `s-${Math.floor(Date.now() / 1000) % 100000}`,
   297→   297→    folder: "sent",
   298→   298→    fromId: SESSION_USER_ID,
   299→   299→    to,
   300→   300→    cc: cc.length ? cc : undefined,
   301→   301→    subject: input.subject?.trim() || "(no subject)",
   302→   302→    preview: (input.body ?? "").replace(/\s+/g, " ").trim().slice(0, 140),
   303→   303→    body: input.body ?? "",
   304→   304→    ts: Date.now(),
   305→   305→    read: true,
   306→   306→    sendStatus: "Sent",
   307→   307→    attachments: atts.length
   308→   308→      ? atts.map((a) => ({ name: a.name, sizeKb: a.sizeKb, type: (a.type?.toUpperCase() || "BIN") as never }))
   309→   309→      : undefined,
   310→   310→  };
   311→   311→  store().mail.unshift(msg);
   312→   312→  writeAudit({ actorId: SESSION_USER_ID, action: "Mail sent", resource: msg.subject, result: "Success", context: `${to.length} recipient(s)${cc.length ? `, ${cc.length} cc` : ""}${atts.length ? `, ${atts.length} attachment(s)` : ""}` });
   313→   313→  return { ok: true as const, message: msg };
   314→   314→}

   503→   503→/** Fixture record + session minutes merged into the wire shape. */
   504→   504→function materializeMeeting(s: WorkspaceStore, m: Meeting): Meeting {
   505→   505→  const notes = s.meetingNotes.get(m.id) ?? [];
   506→   506→  const actions = s.meetingActions.get(m.id) ?? [];
   507→   507→  if (notes.length === 0 && actions.length === 0) return m;
   508→   508→  return {
   509→   509→    ...m,
   510→   510→    notes: notes.length ? [m.notes, ...notes.map((n) => n.text)].filter(Boolean).join("\n\n") : m.notes,
   511→   511→    actionItems: actions.length ? [...(m.actionItems ?? []), ...actions.map((a) => ({ title: a.title, ownerId: a.ownerId, due: a.due }))] : m.actionItems,
   512→   512→  };
   513→   513→}

   563→   563→export function cancelMeeting(role: WorkspaceRole, input: { id: string; reason: string }) {
   564→   564→  const s = store();
   565→   565→  const m = s.meetings.find((x) => x.id === input.id);
   566→   566→  if (!m) return { ok: false as const, status: 404, code: "not_found", message: "Meeting not found." };

   574→   574→  }

   576→   576→  const governance = role === "administrator" || role === "dept-head";
   577→   577→  if (!isOrganizer && !governance) {

   582→   582→  if (reason.length < 4) {

   594→   594→  });
   595→   595→  return {
   596→   596→    ok: true as const,
   597→   597→    meeting: materializeMeeting(s, m),
   598→   598→    message: `"${m.title}" was cancelled — ${m.participantIds.length} participant${m.participantIds.length === 1 ? "" : "s"} notified and the calendar updated.`,
