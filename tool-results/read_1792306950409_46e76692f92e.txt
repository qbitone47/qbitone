    39→// ============================================================
    40→// Screen 88 · Quotation Detail — the document lifecycle view.
    41→// Items, terms, approval trail and versions, plus state-aware
    42→// actions: every mutation calls salesPost and the server's own
    43→// message is toasted verbatim; 4xx errors (locked, invalid_state,
    44→// reason_required, step_up_required…) surface inline as well.
    45→// ============================================================
    46→
    47→interface QuoteItem {
    48→  id: string;
    49→  productId: string;
    50→  name: string;
    51→  qty: number;
    52→  unitPrice: number;
    53→  discountPct: number;
    54→}
    55→
    56→interface QuoteVersion {
    57→  at: number;
    58→  by: string;
    59→  note: string;
    60→}
    61→
    62→interface QuotationRecord {
    63→  id: string;
    64→  number: string;
    65→  accountId: string;
    66→  ownerId: string;
    67→  ownerName: string;
    68→  status: string;
    69→  createdAt: number;
    70→  validUntil: string;
    71→  items: QuoteItem[];
    72→  taxPct: number;
    73→  paymentTerms: string;
    74→  deliveryTerms: string;
    75→  notes?: string;
    76→  approvalId?: string;
    77→  approvalNote?: string;
    78→  sentAt?: number;
    79→  viewedAt?: number;
    80→  decidedAt?: number;
    81→  versions: QuoteVersion[];
    82→  subtotal: number;
    83→  taxAmount: number;
    84→  total: number;
    85→  overallDiscountPct: number;
    86→  account: { id: string; name: string } | null;
    87→  contact: { id: string; name: string; designation: string } | null;
    88→  opportunity: { id: string; name: string } | null;
    89→  order: { id: string; number: string } | null;
    90→}
    91→
    92→interface QuotationPayload {
    93→  quotation: QuotationRecord;
    94→  scope: "company" | "team" | "own";
    95→}
    96→
    97→/** Same math as the server (sales-server.ts) — display only; the server owns pricing. */
    98→const lineTotal = (it: QuoteItem) => Math.round(it.qty * it.unitPrice * (1 - it.discountPct / 100));
    99→
   100→const CANCELLABLE = ["Draft", "Pending Approval", "Approved", "Sent", "Viewed", "Negotiation"];
   101→
   102→const STATUS_EXPLAIN: Record<string, string> = {
   103→  Draft: "Draft — edit freely, then submit for approval. Discount and value rules are checked on the server.",
   104→  "Pending Approval": "Awaiting approval — routed by the Approval Matrix.",
   105→  Approved: "Approved and ready to send to the customer.",
   106→  Sent: "Sent — waiting for the customer to open the document.",
   107→  Viewed: "The customer has opened the quotation — follow up while it is fresh.",
   108→  Negotiation: "In commercial negotiation — record outcomes on the opportunity, or revise the document when terms change.",
   109→  Accepted: "Accepted by the customer — create the sales order to raise the invoice in Finance and request fulfillment from Inventory.",
   110→  Rejected: "Rejected — the approval note and version trail below record what happens next.",
   111→  Expired: "Expired unaccepted — revise to open a fresh version if the customer re-engages.",
   112→  Cancelled: "Cancelled — the reason is recorded in the notes and version history.",
   113→};
   114→
   115→const msgOf = (err: unknown): string =>
   116→  err instanceof SalesApiError ? err.message : "The sales service is unreachable right now.";
   117→
   118→export function QuotationDetailScreen() {
   119→  const nav = useQubitNav();
   120→  const { role } = useRole();
   121→  const quotationId = nav.params.quotationId;
   122→  const { data, loading, error, forbidden, reload } = useSalesSection<QuotationPayload>("quotation", { id: quotationId });
   123→
   124→  const q = data?.quotation ?? null;
   125→  const canDecide = role === "administrator" || role === "dept-head";
   126→
   127→  // ---------- header action state ----------
   128→  const [busy, setBusy] = React.useState<string | null>(null);
   129→  const [actionError, setActionError] = React.useState<string | null>(null);
   130→
   131→  // ---------- step-up approval (total > ₹1 Cr) ----------
   132→  const [stepUpOpen, setStepUpOpen] = React.useState(false);
   133→
   134→  // ---------- reject dialog ----------
   135→  const [rejectOpen, setRejectOpen] = React.useState(false);
   136→  const [rejectNote, setRejectNote] = React.useState("");
   137→  const [rejectError, setRejectError] = React.useState<string | null>(null);
   138→  const [rejectBusy, setRejectBusy] = React.useState(false);
   139→
   140→  // ---------- cancel dialog (required reason) ----------
   141→  const [cancelOpen, setCancelOpen] = React.useState(false);
   142→  const [cancelReason, setCancelReason] = React.useState("");
   143→  const [cancelError, setCancelError] = React.useState<string | null>(null);
   144→  const [cancelBusy, setCancelBusy] = React.useState(false);
   145→  // Radix auto-closes AlertDialog on action click; swallow that close once when
   146→  // client validation fails so the inline error stays reviewable.
   147→  const stayOpenRef = React.useRef(false);
   148→
   149→  const runAction = async (
   150→    type: string,
   151→    input: Record<string, unknown>,
   152→    okTitle: string,
   153→    onOk?: (res: { message?: string; quotation?: { id: string }; order?: { id: string } }) => void,
   154→  ) => {
   155→    if (!q) return;
   156→    setBusy(type);
   157→    setActionError(null);
   158→    try {
   159→      const res = await salesPost<{ ok: boolean; message?: string; quotation?: { id: string }; order?: { id: string } }>(type, input, role);
   160→      toast({ title: okTitle, description: res.message });
   161→      onOk?.(res);
   162→    } catch (err) {
   163→      setActionError(msgOf(err));
   164→      toast({ variant: "destructive", title: okTitle + " failed", description: msgOf(err) });
   165→    } finally {
   166→      setBusy(null);
   167→    }
   168→  };
   169→
   170→  const approve = async (stepUpPassword?: string) => {
   171→    if (!q) return;
   172→    setBusy("quotation.decide");
   173→    setActionError(null);
   174→    try {
   175→      const res = await salesPost<{ ok: boolean; message?: string }>(
   176→        "quotation.decide",
   177→        stepUpPassword ? { id: q.id, decision: "Approved", stepUpPassword } : { id: q.id, decision: "Approved" },
   178→        role,
   179→      );
   180→      toast({ title: "Approval recorded", description: res.message });
   181→      setStepUpOpen(false);
   182→      reload();
   183→    } catch (err) {
   184→      if (err instanceof SalesApiError && err.status === 401 && err.code === "step_up_required") {
   185→        // Decisions above ₹1 Cr require reauthentication — open the shared StepUpDialog and resubmit with the password.
   186→        setStepUpOpen(true);
   187→        toast({ title: "Reauthentication required", description: err.message });
   188→      } else {
   189→        setStepUpOpen(false);
   190→        setActionError(msgOf(err));
   191→        toast({ variant: "destructive", title: "Could not record the approval", description: msgOf(err) });
   192→      }
   193→    } finally {
   194→      setBusy(null);
   195→    }
   196→  };
   197→
   198→  const submitReject = async () => {
   199→    if (!q) return;
   200→    if (rejectNote.trim().length < 3) {
   201→      setRejectError("A reason is required when rejecting — it goes back to the owner.");
   202→      return;
   203→    }
   204→    setRejectBusy(true);
   205→    setRejectError(null);
   206→    try {
   207→      const res = await salesPost<{ ok: boolean; message?: string }>(
   208→        "quotation.decide",
   209→        { id: q.id, decision: "Rejected", note: rejectNote.trim() },
   210→        role,
   211→      );
   212→      toast({ title: "Quotation rejected", description: res.message });
   213→      setRejectOpen(false);
   214→      reload();
   215→    } catch (err) {
   216→      setRejectError(msgOf(err));
   217→    } finally {
   218→      setRejectBusy(false);
   219→    }
   220→  };
   221→
   222→  const submitCancel = async () => {
   223→    if (!q) return;
   224→    if (cancelReason.trim().length < 3) {
   225→      stayOpenRef.current = true;
   226→      setCancelError("A reason is required to cancel a quotation — the server rejects the request without one.");
   227→      return;
   228→    }
   229→    setCancelError(null);
   230→    setCancelBusy(true);
   231→    try {
   232→      const res = await salesPost<{ ok: boolean; message?: string }>("quotation.cancel", { id: q.id, reason: cancelReason.trim() }, role);
   233→      toast({ title: "Quotation cancelled", description: res.message });
   234→      setCancelOpen(false);
   235→      setCancelReason("");
   236→      reload();
   237→    } catch (err) {
   238→      // 409 locked / invalid_state, 403 scope_denied — reopen so the message stays reviewable, toast it verbatim too.
   239→      setCancelError(msgOf(err));
   240→      setCancelOpen(true);
   241→      toast({ variant: "destructive", title: "Could not cancel the quotation", description: msgOf(err) });
   242→    } finally {
   243→      setCancelBusy(false);
   244→    }
   245→  };
   246→
   247→  if (forbidden) {
   248→    return (
   249→      <SalesGate screenName="Quotation Detail">
   250→        <></>
   251→      </SalesGate>
   252→    );
   253→  }
   254→
   255→  if (!quotationId) {
   256→    return (
   257→      <div>
   258→        <PageHeader
   259→          crumbs={[{ label: "Sales" }, { label: "Quotation Detail" }]}
   260→          title="Quotation Detail"
   261→          description="Quotation lifecycle view — items, approvals, versions and state-aware actions."
   262→        />
   263→        <div className="qubit-card">
   264→          <EmptyState
   265→            icon={<FileText />}
   266→            title="Open a quotation from the Quotation List"
   267→            message="Quotation Detail is a deep-linked workspace — pick a document in the Quotation List (86) and its full lifecycle view opens here."
   268→            action={
   269→              <Button size="sm" className="h-9 gap-1.5" onClick={() => nav.navigate("quotation-list")}>
   270→                <FileText className="h-4 w-4" aria-hidden /> Open Quotation List
   271→              </Button>
   272→            }
   273→          />
   274→        </div>
   275→      </div>
   276→    );
   277→  }
   278→
   279→  if (loading) {
   280→    return (
   281→      <div>
   282→        <PageHeader
   283→          crumbs={[{ label: "Sales" }, { label: "Quotation Detail" }]}
   284→          title="Quotation Detail"
   285→          description="Loading the quotation…"
   286→        />
   287→        <SalesSkeleton rows={5} />
   288→      </div>
   289→    );
   290→  }
   291→
   292→  if (error || !q) {
   293→    return (
   294→      <div>
   295→        <PageHeader
   296→          crumbs={[{ label: "Sales" }, { label: "Quotation Detail" }]}
   297→          title="Quotation Detail"
   298→          description="The quotation could not be opened."
   299→        />
   300→        {error && <SalesErrorBanner message={error} onRetry={reload} />}
   301→        <div className="qubit-card">
   302→          <EmptyState
   303→            icon={<FileText />}
   304→            title="Quotation unavailable"
   305→            message={
   306→              error
   307→                ? "The server refused or could not find this document — the message above is the server's own. Records outside your scope are intentionally indistinguishable from missing ones."
   308→                : "The sales service returned no quotation for this deep link."
   309→            }
   310→            action={
   311→              <Button variant="outline" size="sm" onClick={() => nav.navigate("quotation-list")}>
   312→                Back to Quotation List
   313→              </Button>
   314→            }
   315→          />
   316→        </div>
   317→      </div>
   318→    );
   319→  }
   320→
   321→  // ---------- derived display values ----------
   322→  const gross = q.items.reduce((sum, it) => sum + it.qty * it.unitPrice, 0);
   323→  const discountAmount = Math.round(gross - q.subtotal);
   324→  const highDiscount = q.overallDiscountPct > 15;
   325→  const cancellable = CANCELLABLE.includes(q.status);
   326→  const isDraft = q.status === "Draft";
   327→  const isPending = q.status === "Pending Approval";
   328→  const isApproved = q.status === "Approved";
   329→  const isActiveNegotiation = ["Sent", "Viewed", "Negotiation"].includes(q.status);
   330→  const isAccepted = q.status === "Accepted";
   331→
   332→  return (
   333→    <div>
   334→      {/* Clickable "Quotation List" rides as the chip crumb: PageHeader only makes crumbs[0] clickable and renders the h1 as the current page — same pattern as Lead Detail (75). */}
   335→      <PageHeader
   336→        crumbs={[{ label: "Quotation List", onClick: () => nav.navigate("quotation-list") }, { label: q.number }]}
   337→        title={q.number}
   338→        description={`${q.account?.name ?? "Unknown account"} · ${q.opportunity?.name ?? "No linked opportunity"} · owned by ${q.ownerName}`}
   339→        meta={
   340→          <div className="flex flex-wrap items-center gap-x-3 gap-y-1.5">
   341→            <StatusBadge status={q.status} />
   342→            <span className="text-xs text-slate-500">{STATUS_EXPLAIN[q.status] ?? ""}</span>
   343→          </div>
   344→        }
   345→        actions={
   346→          <>
   347→            {isDraft && (
   348→              <>
   349→                <Button variant="outline" size="sm" className="h-9 gap-1.5" onClick={() => nav.navigate("quotation-builder", { quotationId: q.id })}>
   350→                  <Pencil className="h-3.5 w-3.5" aria-hidden /> Edit draft
   351→                </Button>
   352→                <Button
   353→                  size="sm"
   354→                  className="h-9 gap-1.5"
   355→                  disabled={busy === "quotation.submit"}
   356→                  onClick={() => void runAction("quotation.submit", { id: q.id }, "Quotation submitted", () => reload())}
   357→                >
   358→                  <ArrowUpRight className="h-3.5 w-3.5" aria-hidden />
   359→                  {busy === "quotation.submit" ? "Submitting…" : "Submit for approval"}
   360→                </Button>
   361→              </>
   362→            )}
   363→
   364→            {isPending && canDecide && (
   365→              <>
   366→                <Button variant="outline" size="sm" className="h-9 gap-1.5 border-red-200 text-red-600 hover:bg-red-50" onClick={() => { setRejectNote(""); setRejectError(null); setRejectOpen(true); }}>
   367→                  <Ban className="h-3.5 w-3.5" aria-hidden /> Reject
   368→                </Button>
   369→                <Button size="sm" className="h-9 gap-1.5" disabled={busy === "quotation.decide"} onClick={() => void approve()}>
   370→                  <Check className="h-3.5 w-3.5" aria-hidden />
   371→                  {busy === "quotation.decide" ? "Recording…" : "Approve"}
   372→                </Button>
   373→              </>
   374→            )}
   375→
   376→            {isApproved && (
   377→              <>
   378→                <Button variant="outline" size="sm" className="h-9 gap-1.5" disabled={busy === "quotation.revise"} onClick={() => void runAction("quotation.revise", { id: q.id }, "Revision created", (res) => {
   379→                  if (res.quotation?.id) nav.navigate("quotation-detail", { quotationId: res.quotation.id });
   380→                  else reload();
   381→                })}>
   382→                  <Copy className="h-3.5 w-3.5" aria-hidden /> Revise
   383→                </Button>
   384→                <Button
   385→                  size="sm"
   386→                  className="h-9 gap-1.5"
   387→                  disabled={busy === "quotation.send"}
   388→                  onClick={() => void runAction("quotation.send", { id: q.id }, "Quotation sent", () => reload())}
   389→                >
   390→                  <Send className="h-3.5 w-3.5" aria-hidden />
   391→                  {busy === "quotation.send" ? "Sending…" : "Send to customer"}
   392→                </Button>
   393→              </>
   394→            )}
   395→
   396→            {isActiveNegotiation && (
   397→              <Button
   398→                size="sm"
   399→                variant="outline"
   400→                className="h-9 gap-1.5"
   401→                disabled={busy === "quotation.revise"}
   402→                onClick={() => void runAction("quotation.revise", { id: q.id }, "Revision created", (res) => {
   403→                  if (res.quotation?.id) nav.navigate("quotation-detail", { quotationId: res.quotation.id });
   404→                  else reload();
   405→                })}
   406→              >
   407→                <Copy className="h-3.5 w-3.5" aria-hidden /> Revise
   408→              </Button>
   409→            )}
   410→
   411→            {isAccepted && (
   412→              <Button
   413→                size="sm"
   414→                className="h-9 gap-1.5"
   415→                disabled={busy === "order.create"}
   416→                onClick={() =>
   417→                  q.order
   418→                    ? nav.navigate("order-detail", { orderId: q.order.id })
   419→                    : void runAction("order.create", { quotationId: q.id }, "Sales order created", (res) => {
   420→                        if (res.order?.id) nav.navigate("order-detail", { orderId: res.order.id });
   421→                        else reload();
   422→                      })
   423→                }
   424→              >
   425→                {q.order ? (
   426→                  <>
   427→                    <FileText className="h-3.5 w-3.5" aria-hidden /> View order
   428→                  </>
   429→                ) : (
   430→                  <>
   431→                    <ShoppingCart className="h-3.5 w-3.5" aria-hidden />
   432→                    {busy === "order.create" ? "Creating…" : "Create order"}
   433→                  </>
   434→                )}
   435→              </Button>
   436→            )}
   437→
   438→            {cancellable && (
   439→              <Button
   440→                variant="outline"
   441→                size="sm"
   442→                className="h-9 gap-1.5 border-red-200 text-red-600 hover:bg-red-50"
   443→                onClick={() => {
   444→                  setCancelReason("");
   445→                  setCancelError(null);
   446→                  setCancelOpen(true);
   447→                }}
   448→              >
   449→                <X className="h-3.5 w-3.5" aria-hidden /> Cancel
   450→              </Button>
   451→            )}
   452→          </>
   453→        }
   454→      />
   455→
   456→      {/* Inline 4xx surface — the toast carries the same server message verbatim */}
   457→      {actionError && (
   458→        <div className="mb-3 flex flex-wrap items-center justify-between gap-2 rounded-lg border border-red-200 bg-red-50/60 px-3.5 py-2.5" role="alert">
   459→          <p className="text-[13px] font-medium text-red-700">{actionError}</p>
   460→          <button
   461→            type="button"
   462→            onClick={() => setActionError(null)}
   463→            className="text-xs font-medium text-red-600 transition-colors hover:text-red-700"
   464→          >
   465→            Dismiss
   466→          </button>
   467→        </div>
   468→      )}
   469→
   470→      <div className="space-y-4">
   471→        {/* Summary band — the money story of the document */}
   472→        <div className="qubit-card flex flex-wrap items-center gap-x-10 gap-y-4 px-4 py-4 md:px-5">
   473→          <div>
   474→            <p className="text-[11px] font-semibold uppercase tracking-wide text-slate-400">Grand total (incl. {q.taxPct}% GST)</p>
   475→            <p className="mt-0.5 text-[24px] font-bold leading-none tracking-tight tabular-nums text-slate-900">{fmtRupees(q.total)}</p>
   476→          </div>
   477→          <dl className="flex flex-wrap items-center gap-x-10 gap-y-3">
   478→            <div>
   479→              <dt className="text-[11px] font-semibold uppercase tracking-wide text-slate-400">Subtotal</dt>
   480→              <dd className="mt-0.5 text-sm font-medium tabular-nums text-slate-800">{fmtRupees(q.subtotal)}</dd>
   481→            </div>
   482→            <div>
   483→              <dt className="text-[11px] font-semibold uppercase tracking-wide text-slate-400">Overall discount</dt>
   484→              <dd className="mt-0.5 text-sm tabular-nums">
   485→                {highDiscount ? (
   486→                  <span className="inline-flex items-center rounded-full border border-amber-200 bg-amber-50 px-2 py-0.5 text-[11px] font-medium text-amber-700">
   487→                    {q.overallDiscountPct}% · &gt;15%
   488→                  </span>
   489→                ) : (
   490→                  <span className="font-medium text-slate-800">{q.overallDiscountPct}%</span>
   491→                )}
   492→                {discountAmount > 0 && <span className="ml-1.5 text-xs text-slate-400">−{fmtRupees(discountAmount)} vs list</span>}
   493→              </dd>
   494→            </div>
   495→            <div>
   496→              <dt className="text-[11px] font-semibold uppercase tracking-wide text-slate-400">GST {q.taxPct}%</dt>
   497→              <dd className="mt-0.5 text-sm font-medium tabular-nums text-slate-800">{fmtRupees(q.taxAmount)}</dd>
   498→            </div>
   499→            <div>
   500→              <dt className="text-[11px] font-semibold uppercase tracking-wide text-slate-400">Valid until</dt>
   501→              <dd className="mt-0.5 text-sm font-medium text-slate-800">{q.validUntil}</dd>
   502→            </div>
   503→          </dl>
   504→        </div>
   505→
   506→        <div className="grid gap-4 xl:grid-cols-[minmax(0,7fr)_minmax(0,5fr)]">
   507→          {/* Left column — the document itself */}
   508→          <div className="space-y-4">
   509→            <SectionCard
   510→              title="Items"
   511→              description={`${q.items.length} line ${q.items.length === 1 ? "item" : "items"} — server-computed pricing`}
   512→              noPadding
   513→            >
   514→              <div className="qubit-scroll overflow-x-auto">
   515→                <table className="w-full min-w-[560px] border-collapse text-sm" aria-label={`Line items for ${q.number}`}>
   516→                  <thead>
   517→                    <tr className="border-b border-slate-200/90 bg-slate-50/95 text-[11px] font-semibold uppercase tracking-[0.06em] text-slate-500">
   518→                      <th scope="col" className="px-4 py-2.5 text-left font-semibold md:px-5">Product</th>
   519→                      <th scope="col" className="px-3 py-2.5 text-right font-semibold">Qty</th>

   567→                Pricing pulls from the product catalog — Inventory remains the source of truth for availability.

   578→          </div>
   579→
   580→          {/* Right column — trail and links */}
   581→          <div className="space-y-4">
   582→            <SectionCard
   583→              title="Approval & versions"
   584→              description="One enterprise approval trail — decisions recorded in My Approvals (46) land here."
   585→            >
   586→              {q.approvalNote && (
   587→                <div className="mb-4 rounded-lg border border-slate-200 bg-slate-50 px-3.5 py-3">
   588→                  <p className="text-[11px] font-semibold uppercase tracking-wide text-slate-400">Approval note</p>

   594→                  )}
   595→                </div>
   596→              )}
   597→              {isPending && !q.approvalNote && (

   610→                        {v.by} · {relTime(v.at)} · {fmtDateTime(v.at)}

   612→                    </div>

   617→
   618→            <SectionCard title="Customer & links" description="Everything this document is connected to.">

   622→                    <EntityLink screen="account-detail" id={q.account.id} label={q.account.name} kind="account" />

   624→                    <span className="text-slate-400">—</span>

   629→                    <span className="inline-flex flex-wrap items-center gap-x-2">
   630→                      <EntityLink screen="contact-detail" id={q.contact.id} label={q.contact.name} kind="contact" />

   638→                  {q.opportunity ? (
   639→                    <EntityLink screen="opportunity-detail" id={q.opportunity.id} label={q.opportunity.name} kind="opportunity" />

   653→                <KV label="Owner">{q.ownerName}</KV>

   666→
   667→      {/* Reject — approval decision with a required note (server 422 note_required mirrors this) */}
   668→      <FormDialog
   669→        open={rejectOpen}
   670→        onOpenChange={(o) => {
   671→          setRejectOpen(o);
   672→          if (!o) setRejectError(null);

   674→        title={`Reject ${q.number}`}

   710→        title={`Cancel ${q.number}?`}
   711→        description="Cancelled quotations cannot be reopened. The reason is recorded in the document notes and version history, and the action is written to the audit trail."
   712→        confirmLabel={cancelBusy ? "Cancelling…" : "Cancel quotation"}

   751→    </div>
