    56→/** ₹1.24 Cr / ₹68.0 L / ₹18,400 — a single enterprise formatting voice */
    57→export function formatINR(cr: number): string {
    58→  const abs = Math.abs(cr);
    59→  if (abs >= 1) return `₹${cr.toFixed(2)} Cr`;
    60→  if (abs >= 0.01) return `₹${(cr * 100).toFixed(cr * 100 >= 100 ? 0 : 1)} L`;
    61→  return `₹${Math.round(cr * 1e7).toLocaleString("en-IN")}`;
    62→}
    63→
    64→/** For whole-rupee values (e.g. CPL) */
    65→export function formatRupees(n: number): string {
    66→  return `₹${Math.round(n).toLocaleString("en-IN")}`;
    67→}
    68→
    69→export function formatPct(n: number, digits = 1): string {
    70→  return `${n.toFixed(digits)}%`;
    71→}
