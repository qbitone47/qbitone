    45→interface CalendarPayload {
    46→  meetings: Meeting[];
    47→  events: CalendarEvent[];
    48→  scope: "own";
    49→  note: string | null;
    50→  asOf: string;
    51→  now: number;
    52→}
    53→
    54→const HOUR_H = 48; // px per hour in week grid
    55→const DAY_START = 8; // 08:00
    56→const DAY_END = 20; // 20:00
    57→
    58→const TYPE_STYLES: Record<EventType, { chip: string; dot: string; label: string }> = {
    59→  Meeting: { chip: "border-blue-200 bg-blue-50 text-blue-800", dot: "bg-blue-600", label: "Meetings" },
    60→  Deadline: { chip: "border-amber-200 bg-amber-50 text-amber-800", dot: "bg-amber-500", label: "Deadlines" },
    61→  "Company event": { chip: "border-violet-200 bg-violet-50 text-violet-800", dot: "bg-violet-500", label: "Company events" },
    62→  Leave: { chip: "border-emerald-200 bg-emerald-50 text-emerald-800", dot: "bg-emerald-500", label: "Approved leave" },
    63→  "Focus time": { chip: "border-slate-200 bg-slate-50 text-slate-700", dot: "bg-slate-400", label: "Focus time" },
    64→};
    65→
    66→const TYPE_ORDER: EventType[] = ["Meeting", "Deadline", "Company event", "Leave", "Focus time"];
    67→const DAY_MS = 86_400_000;
    68→const WEEKDAYS = ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"];
    69→
    70→/** Midnight IST (epoch) of the given timestamp's IST day. */
    71→const istMidnight = (ts: number) => {
    72→  const d = ist(ts);
    73→  return Date.UTC(d.getUTCFullYear(), d.getUTCMonth(), d.getUTCDate()) - 330 * 60_000;
    74→};

    76→export function MyCalendarScreen() {
    77→  const nav = useQubitNav();
    78→  const { toast } = useToast();
    79→  const { role } = useRole();
    80→
    81→  const [data, setData] = React.useState<CalendarPayload | null>(null);
    82→  const [error, setError] = React.useState<string | null>(null);
    83→  const [loading, setLoading] = React.useState(true);
    84→  const [view, setView] = React.useState<"day" | "week" | "month">("week");
    85→  const [anchor, setAnchor] = React.useState<number | null>(null);
    86→  const [types, setTypes] = React.useState<EventType[]>(TYPE_ORDER);

   100→      setLoading(false);

   102→  }, [role]);
   103→
   104→  React.useEffect(() => {

   107→

   109→    try {
   110→      const res = await fetch("/api/workspace/meetings", {
   111→        method: "POST",
   112→        headers: { "Content-Type": "application/json", "X-Qubit-Role": role },
   113→        body: JSON.stringify({ action: "create-event", ...input }),
   114→      });
   115→      const resData = await res.json().catch(() => null);
   116→      if (!res.ok) {
   117→        toast({
   118→          variant: "destructive",
   119→          title: "Event not added",
   120→          description: resData?.message ?? "The calendar service rejected this event. Nothing was changed.",
   121→        });
   122→        return;
   123→      }
   124→      toast({ title: "Event added to your calendar", description: resData?.message ?? "The event was added." });
   125→      await load();
   126→    } catch {
   127→      toast({
   128→        variant: "destructive",
   129→        title: "Event not added",
   130→        description: "The calendar service is unreachable. Nothing was changed — retry when you're back online.",
   131→      });
   132→    }
   133→  };
   134→
   135→  if (loading || !data || anchor === null) {
   136→    return (
   137→      <div>
   138→        <PageHeader
   139→          crumbs={[{ label: "Workspace" }, { label: "My time" }]}

   142→          actions={

   148→        {error && (

   150→            <p className="text-sm font-medium text-red-700">{error}</p>
   151→            <Button size="sm" variant="outline" className="h-8 gap-1.5" onClick={load}>
   152→              <RefreshCw className="h-3.5 w-3.5" aria-hidden /> Retry
   153→            </Button>
   154→          </div>
   155→        )}
   156→        {loading && <CalendarSkeleton />}
   157→      </div>

   161→  const now = data.now;
   162→  const allEvents = data.events;

   164→  const meetingsById = new Map(data.meetings.map((m) => [m.id, m]));

   166→  const anchorMid = istMidnight(anchor);

   176→  const monthGrid = Array.from({ length: 42 }, (_, i) => monthGridStart + i * DAY_MS);

   211→
   212→  const isToday = (dayMid: number) => dayMid === istMidnight(now);
   213→  const isPast = (dayMid: number) => dayMid < istMidnight(now);

   216→
   217→  return (
   218→    <div>
   219→      <PageHeader
   220→        crumbs={[{ label: "Workspace" }, { label: "My time" }]}
   221→        title="My Calendar"
   222→        description="Meetings, deadlines, company events and approved leave in one enterprise calendar — with privacy you can rely on."
   223→        actions={
   224→          <Button className="h-9 gap-1.5" onClick={() => setCreateOpen(true)}>
   225→            <CalendarPlus className="h-4 w-4" aria-hidden /> New event
   226→          </Button>
   227→        }
   228→      />
   229→
   230→      {error && (
   231→        <div className="qubit-card mb-3 flex flex-wrap items-center justify-between gap-3 border-red-200 bg-red-50/60 px-4 py-3" role="alert">

   239→      {/* Toolbar */}

   241→        <div className="flex items-center gap-1">
   242→          <Button variant="outline" size="sm" className="h-9" onClick={() => setAnchor(now)}>
   243→            Today
   244→          </Button>

   246→            <ChevronLeft className="h-4 w-4" aria-hidden />

   273→            options={[
   274→              { value: "day", label: "Day" },

   284→      {view === "day" && (
   285→        <DayView dayMid={anchorMid} timedOn={timedOn} allDayOn={allDayOn} onOpen={setSelected} />

   288→      {view === "month" && (

   302→      {/* Legend + privacy note */}
   303→      <div className="mt-3 flex flex-wrap items-center gap-x-4 gap-y-1.5 text-[11px] text-slate-400">

   314→      <NewEventDialog

   326→function WeekGrid({
   327→  days,
   328→  timedOn,
   329→  allDayOn,
   330→  onOpen,
   331→  now,
   332→}: {
   333→  days: number[];
   334→  timedOn: (d: number) => CalendarEvent[];
   335→  allDayOn: (d: number) => CalendarEvent[];
   336→  onOpen: (e: CalendarEvent) => void;
   337→  now: number;
   338→}) {
   339→  const hours = Array.from({ length: DAY_END - DAY_START }, (_, i) => DAY_START + i);
   340→  const todayMid = istMidnight(now);
   341→  const nowOffsetMin = (ist(now).getUTCHours() - DAY_START) * 60 + ist(now).getUTCMinutes();
   342→  const todayIdx = days.findIndex((d) => d === todayMid);
   343→
   344→  return (
   345→    <div className="qubit-card overflow-hidden">
   346→      <div className="qubit-scroll overflow-x-auto">
   347→        <div className="min-w-[860px]">
   348→          {/* day headers */}
   349→          <div className="grid border-b border-slate-200/90 bg-slate-50/80" style={{ gridTemplateColumns: `56px repeat(7, minmax(0, 1fr))` }}>
   350→            <span aria-hidden />
   351→            {days.map((d) => {
   352→              const today = d === todayMid;
   353→              return (
   354→                <div key={d} className={cn("border-l border-slate-100 px-2 py-2 text-center", today && "bg-blue-50/60")}>
   355→                  <p className={cn("text-[10.5px] font-medium uppercase tracking-wide", today ? "text-blue-700" : "text-slate-400")}>
   356→                    {WEEKDAYS[(ist(d).getUTCDay() + 6) % 7]}
   357→                  </p>
   358→                  <p className={cn("text-[15px] font-semibold tabular-nums", today ? "text-blue-700" : d < todayMid ? "text-slate-400" : "text-slate-800")}>
   359→                    {ist(d).getUTCDate()}
   360→                  </p>
   361→                </div>
   362→              );
   363→            })}
   364→          </div>
   365→          {/* all-day strip */}
   366→          <div className="grid border-b border-slate-100" style={{ gridTemplateColumns: `56px repeat(7, minmax(0, 1fr))` }}>
   367→            <span className="pt-1.5 text-right pr-1.5 text-[9.5px] font-medium uppercase tracking-wide text-slate-400">All day</span>
   368→            {days.map((d) => (
   369→              <div key={d} className="min-h-9 space-y-1 border-l border-slate-100 p-1">
   370→                {allDayOn(d).map((e) => (
   371→                  <AllDayChip key={e.id} event={e} onOpen={() => onOpen(e)} />
   372→                ))}
   373→              </div>
   374→            ))}
   375→          </div>
   376→          {/* timed grid */}
   377→          <div className="relative grid" style={{ gridTemplateColumns: `56px repeat(7, minmax(0, 1fr))` }}>
   378→            <div aria-hidden>
   379→              {hours.map((h) => (
   380→                <div key={h} className="relative" style={{ height: HOUR_H }}>
   381→                  <span className="absolute -top-1.5 right-1.5 text-[10px] tabular-nums text-slate-400">{String(h).padStart(2, "0")}:00</span>
   382→                </div>
   383→              ))}
   384→            </div>
   385→            {days.map((d) => (
   386→              <div key={d} className={cn("relative border-l border-slate-100", todayIdx >= 0 && d === days[todayIdx] && "bg-blue-50/25")}>
   387→                {hours.map((h) => (
   388→                  <div key={h} className="border-b border-slate-100/70" style={{ height: HOUR_H }} aria-hidden />
   389→                ))}
   390→                {timedOn(d).map((e) => (
   391→                  <EventBlock key={e.id} event={e} dayMid={d} onOpen={() => onOpen(e)} />
   392→                ))}
   393→                {/* now indicator inside today's column */}
   394→                {d === todayMid &&
   395→                  nowOffsetMin >= 0 &&
   396→                  nowOffsetMin <= (DAY_END - DAY_START) * 60 && (
   397→                    <div aria-hidden className="pointer-events-none absolute left-0 right-0 z-10 flex items-center" style={{ top: (nowOffsetMin / 60) * HOUR_H }}>
   398→                      <span className="h-2 w-2 -translate-x-1 rounded-full bg-red-500" />
   399→                      <span className="h-px flex-1 bg-red-400" />
   400→                    </div>
   401→                  )}
   402→              </div>
   403→            ))}
   404→          </div>
   405→        </div>
   406→      </div>
   407→    </div>
   408→  );
   409→}

   411→function evMinutes(e: CalendarEvent) {
   412→  const d = ist(e.start);
   413→  return (d.getUTCHours() - DAY_START) * 60 + d.getUTCMinutes();
   414→}
   415→
   416→function evDurationMin(e: CalendarEvent) {
   417→  return Math.max(30, Math.round((e.end - e.start) / 60_000));
   418→}
   419→
   420→function EventBlock({ event: e, dayMid, onOpen }: { event: CalendarEvent; dayMid: number; onOpen: () => void }) {
   421→  const style = TYPE_STYLES[e.type];
   422→  const privateEvent = e.visibleTo === "organizer-only" && e.organizerId !== "sagar";
   423→  const top = (evMinutes(e) / 60) * HOUR_H;
   424→  const height = Math.max(22, (evDurationMin(e) / 60) * HOUR_H - 2);
   425→  return (
   426→    <button
   427→      type="button"
   428→      onClick={onOpen}
   429→      aria-label={privateEvent ? "Private event — Busy" : `${e.title}, ${fmtTime(e.start)} to ${fmtTime(e.end)}`}
   430→      className={cn(
   431→        "absolute left-0.5 right-1 z-[5] overflow-hidden rounded-md border px-1.5 py-0.5 text-left transition-all duration-150 hover:z-20 hover:shadow-md",
   432→        privateEvent ? "border-slate-200 bg-slate-100/90" : style.chip,
   433→      )}
   434→      style={{ top, height }}
   435→    >
   436→      <span className="flex items-center gap-1">
   437→        {privateEvent ? (
   438→          <Lock className="h-2.5 w-2.5 shrink-0 text-slate-400" aria-hidden />
   439→        ) : (
   440→          <span className={cn("h-1.5 w-1.5 shrink-0 rounded-full", style.dot)} aria-hidden />
   441→        )}
   442→        <span className="truncate text-[10.5px] font-semibold leading-tight">{privateEvent ? "Busy" : e.title}</span>
   443→      </span>
   444→      {height > 34 && !privateEvent && (
   445→        <span className="mt-px block truncate text-[9.5px] leading-tight opacity-75">
   446→          {fmtTime(e.start)} · {e.location ?? e.type}
   447→        </span>
   448→      )}
   449→    </button>
   450→  );
   451→}

   453→function AllDayChip({ event: e, onOpen }: { event: CalendarEvent; onOpen: () => void }) {
   454→  const privateEvent = e.visibleTo === "organizer-only" && e.organizerId !== "sagar";
   455→  const style = TYPE_STYLES[e.type];
   456→  return (
   457→    <button
   458→      type="button"
   459→      onClick={onOpen}
   460→      aria-label={privateEvent ? "Private event — Busy" : `${e.title} (all day)`}
   461→      className={cn("block w-full truncate rounded border px-1.5 py-0.5 text-left text-[10px] font-medium", privateEvent ? "border-slate-200 bg-slate-100 text-slate-500" : style.chip)}
   462→    >
   463→      {privateEvent ? "Busy" : e.title}
   464→    </button>
   465→  );
   466→}

   470→function DayView({
   471→  dayMid,
   472→  timedOn,
   473→  allDayOn,
   474→  onOpen,
   475→}: {
   476→  dayMid: number;
   477→  timedOn: (d: number) => CalendarEvent[];
   478→  allDayOn: (d: number) => CalendarEvent[];
   479→  onOpen: (e: CalendarEvent) => void;
   480→}) {
   481→  const hours = Array.from({ length: DAY_END - DAY_START }, (_, i) => DAY_START + i);
   482→  const timed = timedOn(dayMid);
   483→  const allDay = allDayOn(dayMid);
   484→  return (
   485→    <div className="qubit-card overflow-hidden">
   486→      {allDay.length > 0 && (
   487→        <div className="flex flex-wrap gap-2 border-b border-slate-100 bg-slate-50/60 px-4 py-2.5">

   489→            <AllDayChip key={e.id} event={e} onOpen={() => onOpen(e)} />
   490→          ))}
   491→        </div>
   492→      )}
   493→      <ol className="divide-y divide-slate-100" aria-label="Day schedule">
   494→        {hours.map((h) => {
   495→          const slotEvents = timed.filter((e) => ist(e.start).getUTCHours() === h);
   496→          return (
   497→            <li key={h} className="flex gap-4 px-4 py-2">
   498→              <span className="w-12 shrink-0 pt-0.5 text-right text-[11px] tabular-nums text-slate-400">{String(h).padStart(2, "0")}:00</span>

   500→                {slotEvents.length === 0 && <div className="h-6 rounded-md bg-slate-50/60" aria-hidden />}
   501→                {slotEvents.map((e) => {
   502→                  const privateEvent = e.visibleTo === "organizer-only" && e.organizerId !== "sagar";
   503→                  return (
   504→                    <button
   505→                      key={e.id}
   506→                      type="button"
   507→                      onClick={() => onOpen(e)}
   508→                      aria-label={privateEvent ? "Private event — Busy" : `${e.title}, ${fmtTime(e.start)}`}
   509→                      className={cn(
   510→                        "flex w-full items-center gap-2.5 rounded-lg border px-3 py-2 text-left transition-colors hover:shadow-sm",
   511→                        privateEvent ? "border-slate-200 bg-slate-100/80" : TYPE_STYLES[e.type].chip,
   512→                      )}
   513→                    >
   514→                      <span className={cn("h-2 w-2 shrink-0 rounded-full", privateEvent ? "bg-slate-400" : TYPE_STYLES[e.type].dot)} aria-hidden />
   515→                      <span className="min-w-0 flex-1">
   516→                        <span className="block truncate text-[13px] font-semibold">{privateEvent ? "Busy — private event" : e.title}</span>
   517→                        {!privateEvent && (
   518→                          <span className="block truncate text-[11px] opacity-80">
   519→                            {fmtTime(e.start)}–{fmtTime(e.end)}
   520→                            {e.location ? ` · ${e.location}` : ""}
   521→                          </span>
   522→                        )}
   523→                      </span>
   524→                      {!privateEvent && e.meetingId && <Video className="h-3.5 w-3.5 shrink-0 opacity-60" aria-hidden />}
   525→                    </button>
   526→                  );
   527→                })}
   528→              </div>
   529→            </li>
   530→          );
   531→        })}
   532→        {timed.length === 0 && allDay.length === 0 && (
   533→          <li className="px-4 py-2">
   534→            <EmptyState icon={<CalendarDays />} title="Nothing scheduled" message="This day is clear — a good place for focused work." className="py-8" />
   535→          </li>
   536→        )}
   537→      </ol>
   538→    </div>
   539→  );

   601→                      key={e.id}

   613→                  <button type="button" onClick={() => onOpenDay(d)} className="w-full px-1 text-left text-[10px] font-medium text-slate-500 hover:text-blue-700">

   615→                  </button>
   616→                )}
   617→              </div>

   619→            </div>
   620→          );
   621→        })}
   622→      </div>
   623→    </div>
   624→  );

   718→// ---------- New event dialog ----------
   719→
   720→function NewEventDialog({
   721→  open,
   722→  onOpenChange,
   723→  defaultDay,
   724→  onCreate,
   725→}: {
   726→  open: boolean;
   727→  onOpenChange: (o: boolean) => void;
   728→  defaultDay: number;
   729→  onCreate: (input: { title: string; type: EventType; start: number; end: number; location?: string }) => void;
   730→}) {
   731→  const [title, setTitle] = React.useState("");
   732→  const [type, setType] = React.useState<EventType>("Meeting");
   733→  const [time, setTime] = React.useState("14:00");
   734→  const [duration, setDuration] = React.useState("30");
   735→  const [location, setLocation] = React.useState("");
   736→  const [error, setError] = React.useState<string | null>(null);
   737→
   738→  React.useEffect(() => {
   739→    if (open) {
   740→      setTitle("");
   741→      setType("Meeting");
   742→      setTime("14:00");
   743→      setDuration("30");
   744→      setLocation("");
   745→      setError(null);
   746→    }
   747→  }, [open]);
   748→
   749→  const submit = () => {
   750→    if (title.trim().length < 3) {
   751→      setError("Give the event a title (at least 3 characters).");
   752→      return;
   753→    }
   754→    const [h, m] = time.split(":").map(Number);
   755→    const dur = Number(duration);
   756→    const start = defaultDay + h * 3_600_000 + m * 60_000;
   757→    onCreate({
   758→      title: title.trim(),
   759→      type,
   760→      start,
   761→      end: start + dur * 60_000,
   762→      location: location.trim() || undefined,
   763→    });
   764→    onOpenChange(false);
   765→  };
   766→
   767→  return (
   768→    <FormDialog open={open} onOpenChange={onOpenChange} title="New event" description="Adds to your calendar for this session. Meeting invites with participants run through the Meetings module." submitLabel="Add event" onSubmit={submit}>
   769→      <div className="space-y-1.5">
   770→        <Label htmlFor="ev-title" className="text-xs font-medium text-slate-700">
   771→          Title <span className="text-red-500" aria-hidden>*</span>
   772→        </Label>
   773→        <Input id="ev-title" value={title} onChange={(e) => { setTitle(e.target.value); setError(null); }} placeholder="e.g. Focus time — board deck" autoFocus aria-invalid={Boolean(error)} />
   774→        {error && <p role="alert" className="text-[11.5px] font-medium text-red-600">{error}</p>}
   775→      </div>
   776→      <div className="grid gap-3 sm:grid-cols-3">
   777→        <div className="space-y-1.5 sm:col-span-1">
   778→          <Label className="text-xs font-medium text-slate-700">Type</Label>
   779→          <Select value={type} onValueChange={(v) => setType(v as EventType)}>
   780→            <SelectTrigger aria-label="Event type"><SelectValue /></SelectTrigger>
   781→            <SelectContent>
   782→              <SelectItem value="Meeting">Meeting</SelectItem>
   783→              <SelectItem value="Deadline">Deadline</SelectItem>
   784→              <SelectItem value="Focus time">Focus time</SelectItem>
   785→            </SelectContent>
   786→          </Select>
   787→        </div>
   788→        <div className="space-y-1.5">
   789→          <Label htmlFor="ev-time" className="text-xs font-medium text-slate-700">Start (IST)</Label>
   790→          <Input id="ev-time" type="time" value={time} onChange={(e) => setTime(e.target.value)} />
   791→        </div>
   792→        <div className="space-y-1.5">
   793→          <Label htmlFor="ev-duration" className="text-xs font-medium text-slate-700">Duration</Label>
   794→          <Select value={duration} onValueChange={setDuration}>
   795→            <SelectTrigger id="ev-duration" aria-label="Duration"><SelectValue /></SelectTrigger>
   796→            <SelectContent>
   797→              <SelectItem value="15">15 min</SelectItem>
   798→              <SelectItem value="30">30 min</SelectItem>
   799→              <SelectItem value="45">45 min</SelectItem>
   800→              <SelectItem value="60">1 hour</SelectItem>
   801→              <SelectItem value="120">2 hours</SelectItem>
   802→            </SelectContent>
   803→          </Select>
   804→        </div>
   805→      </div>
   806→      <div className="space-y-1.5">
   807→        <Label htmlFor="ev-location" className="text-xs font-medium text-slate-700">Location</Label>
   808→        <Input id="ev-location" value={location} onChange={(e) => setLocation(e.target.value)} placeholder="Room, link or platform" />
   809→      </div>
   810→    </FormDialog>
   811→  );
   812→}
