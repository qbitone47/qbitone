   171→export function decideApproval(
   172→  role: WorkspaceRole,
   173→  input: { id: string; decision: "Approved" | "Rejected" | "Changes requested"; note?: string; stepUpPassword?: string },
   174→) {
   175→  const s = store();
   176→  const req = s.approvals.find((a) => a.id === input.id);
   177→  if (!req) return { ok: false as const, status: 404, code: "not_found", message: "Approval request not found." };
   178→  if (req.status !== "Pending") {
   179→    return { ok: false as const, status: 409, code: "already_decided", message: `This request was already ${req.status.toLowerCase()} by another approver.` };
   180→  }
   181→
   182→  // Server-side authorization: employee boundaries hold no approval authority.
   183→  if (role === "employee" || role === "sales-exec") {
   184→    writeAudit({ actorId: SESSION_USER_ID, action: "Approval decision denied", resource: req.title, result: "Blocked", severity: "Warning", context: "Role holds no approval authority for this request type" });
   185→    return { ok: false as const, status: 403, code: "forbidden", message: "Your role does not hold approval authority for this request." };
   186→  }
   187→
   188→  // Step-up authentication for sensitive approvals (server-enforced).
   189→  if (req.stepUpRequired) {
   190→    if (!input.stepUpPassword || input.stepUpPassword.length < 8) {
   191→      return { ok: false as const, status: 401, code: "step_up_required", message: "Reauthentication is required for this approval." };
   192→    }
   193→  }
   194→
   195→  // High-value decisions require a note for the audit trail.
   196→  if (input.decision === "Rejected" && (!input.note || input.note.trim().length < 4)) {
   197→    return { ok: false as const, status: 422, code: "note_required", message: "A reason is required when rejecting a request." };
   198→  }
   199→
   200→  req.status = input.decision;
   201→  req.decidedAt = Date.now();
   202→  req.decidedBy = SESSION_USER_ID;
   203→  req.decisionNote = input.note?.trim() || undefined;
   204→  req.history = [
   205→    ...(req.history ?? []),
   206→    { at: Date.now(), actor: "Sagar Singh", event: input.decision + (input.note ? ` — ${input.note.trim()}` : "") },
   207→  ];
   208→
   209→  writeAudit({
   210→    actorId: SESSION_USER_ID,
   211→    action: `Approval ${input.decision.toLowerCase()}`,
   212→    resource: `${req.kind} · ${req.title}`,
   213→    result: "Success",
   214→    severity: req.stepUpRequired ? "High" : "Info",
   215→    source: "Web · Mumbai",
   216→    context: req.stepUpRequired
   217→      ? "Step-up authentication verified before the decision"
   218→      : `Routed via Approval Matrix (${req.rule})`,
   219→  });
   220→
   221→  return { ok: true as const, approval: req };
   222→}

   232→export function createTask(role: WorkspaceRole, input: { title: string; detail?: string; priority?: Task["priority"]; due?: number | null; /** Phase 20: origin of the task (defaults to Manual for the quick action) */ source?: Task["source"]; related?: Task["related"] }) {
   233→  const title = input.title?.trim() ?? "";
   234→  if (title.length < 4) return { ok: false as const, status: 422, code: "invalid_title", message: "Task title must be at least 4 characters." };
   235→  if (title.length > 140) return { ok: false as const, status: 422, code: "invalid_title", message: "Task title must be 140 characters or fewer." };
   236→  if (input.due !== null && input.due !== undefined && Number.isNaN(input.due)) {
   237→    return { ok: false as const, status: 422, code: "invalid_due", message: "Due date is invalid." };
   238→  }
   239→
   240→  const t: Task = {
   241→    id: `task-${Math.floor(Date.now() / 1000) % 100000}`,
   242→    title,
   243→    detail: input.detail?.trim() || "",
   244→    status: "Not Started",
   245→    priority: input.priority ?? "Medium",
   246→    due: input.due ?? null,
   247→    source: input.source ?? "Manual",
   248→    ownerId: SESSION_USER_ID,
   249→    assigneeId: SESSION_USER_ID,
   250→    related: input.related,
   251→  };
   252→  store().tasks.unshift(t);
   253→  writeAudit({ actorId: SESSION_USER_ID, action: "Task created", resource: t.title, result: "Success", context: input.source ? `Created from ${input.source}` : "Created from Employee Home quick action" });
   254→  return { ok: true as const, task: t };
   255→}

   258→  "Not Started": ["In Progress", "Cancelled"],
   259→  "In Progress": ["Waiting", "Blocked", "Completed", "Cancelled"],
   260→  Blocked: ["In Progress", "Waiting", "Cancelled"],

   266→export function updateTask(role: WorkspaceRole, input: { id: string; status: Task["status"] }) {
   267→  const t = store().tasks.find((x) => x.id === input.id);
   268→  if (!t) return { ok: false as const, status: 404, code: "not_found", message: "Task not found." };
   269→  // Authorization: only the assignee transitions their own task.
   270→  if (t.assigneeId !== SESSION_USER_ID) {
   271→    writeAudit({ actorId: SESSION_USER_ID, action: "Task update denied", resource: t.title, result: "Blocked", severity: "Warning", context: "Caller is not the task assignee" });
   272→    return { ok: false as const, status: 403, code: "forbidden", message: "Only the task assignee can update this task." };
   273→  }
   274→  const allowed = TASK_TRANSITIONS[t.status] ?? [];
   275→  if (!allowed.includes(input.status)) {
   276→    return { ok: false as const, status: 409, code: "invalid_transition", message: `A task cannot move from ${t.status} to ${input.status}.` };
   277→  }
   278→  t.status = input.status;
   279→  if (input.status === "Completed") t.completedAt = Date.now();
   280→  writeAudit({ actorId: SESSION_USER_ID, action: "Task status changed", resource: t.title, result: "Success", context: `${t.status === "Completed" ? "Marked completed" : `Moved to ${input.status}`}` });
   281→  return { ok: true as const, task: t };
   282→}

   290→export function listMail(role: WorkspaceRole, params: { folder?: string; q?: string; unreadOnly?: boolean; label?: string; limit?: number }) {
   291→  const s = store();
   292→  const folder = params.folder ?? "inbox";
   293→  let items = folder === "all" ? [...s.mail] : s.mail.filter((m) => m.folder === folder);
   294→  // Authorization scope: the session mailbox only (demo: single-user store).
   295→  const q = params.q?.trim().toLowerCase();
   296→  if (q) {
   297→    items = items.filter((m) => {
   298→      const fromName = m.fromId ? employees.find((e) => e.id === m.fromId)?.name ?? "" : m.fromExternal?.name ?? "";
   299→      const fromEmail = m.fromId ? employees.find((e) => e.id === m.fromId)?.email ?? "" : m.fromExternal?.email ?? "";
   300→      return [m.subject, m.preview, m.body, fromName, fromEmail, ...(m.to ?? [])].join(" ").toLowerCase().includes(q);
   301→    });
   302→  }
   303→  if (params.unreadOnly) items = items.filter((m) => !m.read);
   304→  if (params.label) items = items.filter((m) => m.labels?.includes(params.label as never));
   305→  items.sort((a, b) => b.ts - a.ts);
   306→  const limit = params.limit && params.limit > 0 ? Math.min(params.limit, 200) : 100;
   307→  return { items: items.slice(0, limit), total: items.length, scopeNote: "Search is scoped to your authorized mailbox server-side." };
   308→}

   310→export function sendMail(role: WorkspaceRole, input: { to: string[]; cc?: string[]; bcc?: string[]; subject: string; body: string; attachments?: { name: string; sizeKb: number; type: string }[] }) {
   311→  const to = (input.to ?? []).map((v) => v.trim()).filter(Boolean);
   312→  const cc = (input.cc ?? []).map((v) => v.trim()).filter(Boolean);
   313→  if (to.length === 0) return { ok: false as const, status: 422, code: "no_recipients", message: "Add at least one recipient." };
   314→  if (to.some((v) => !emailLike(v))) return { ok: false as const, status: 422, code: "invalid_recipient", message: "One or more recipient addresses are not valid." };
   315→  if (!input.subject?.trim() && !input.body?.trim()) {
   316→    return { ok: false as const, status: 422, code: "empty_message", message: "Add a subject or a message before sending." };
   317→  }
   318→  const atts = input.attachments ?? [];
   319→  if (atts.length > MAX_ATTACHMENTS) return { ok: false as const, status: 422, code: "too_many_attachments", message: `A message can carry at most ${MAX_ATTACHMENTS} attachments.` };
   320→  const totalKb = atts.reduce((n, a) => n + (a.sizeKb || 0), 0);
   321→  if (totalKb > MAX_TOTAL_KB) return { ok: false as const, status: 422, code: "attachment_too_large", message: "Attachments exceed the 10 MB message limit." };
   322→
   323→  const msg: MailMessage = {
   324→    id: `s-${Math.floor(Date.now() / 1000) % 100000}`,
   325→    folder: "sent",
   326→    fromId: SESSION_USER_ID,
   327→    to,
   328→    cc: cc.length ? cc : undefined,
   329→    subject: input.subject?.trim() || "(no subject)",
   330→    preview: (input.body ?? "").replace(/\s+/g, " ").trim().slice(0, 140),
   331→    body: input.body ?? "",
   332→    ts: Date.now(),
   333→    read: true,
   334→    sendStatus: "Sent",
   335→    attachments: atts.length
   336→      ? atts.map((a) => ({ name: a.name, sizeKb: a.sizeKb, type: (a.type?.toUpperCase() || "BIN") as never }))
   337→      : undefined,
   338→  };
   339→  store().mail.unshift(msg);
   340→  writeAudit({ actorId: SESSION_USER_ID, action: "Mail sent", resource: msg.subject, result: "Success", context: `${to.length} recipient(s)${cc.length ? `, ${cc.length} cc` : ""}${atts.length ? `, ${atts.length} attachment(s)` : ""}` });
   341→  return { ok: true as const, message: msg };
   342→}

   375→export function retryMail(role: WorkspaceRole, id: string) {
   376→  const m = store().mail.find((x) => x.id === id);
   377→  if (!m || m.folder !== "sent") return { ok: false as const, status: 404, code: "not_found", message: "Message not found in Sent." };
   378→  if (m.sendStatus !== "Failed" && m.sendStatus !== "Pending") {
   379→    return { ok: false as const, status: 409, code: "not_retryable", message: `This message is already ${(m.sendStatus ?? "sent").toLowerCase()}.` };
   380→  }
   381→  m.sendStatus = "Sent";
   382→  m.sendDetail = undefined;
   383→  writeAudit({ actorId: SESSION_USER_ID, action: "Mail retry succeeded", resource: m.subject, result: "Success", context: "Delivery retried after failure" });
   384→  return { ok: true as const, message: m };
   385→}

   404→export function documentAction(
   405→  role: WorkspaceRole,
   406→  input: { action: "download" | "share" | "rename" | "move" | "upload"; docId?: string; name?: string; category?: string; toEmployeeId?: string; fileMeta?: { name: string; sizeKb: number; type: string } },
   407→) {
   408→  const s = store();
   409→
   410→  if (input.action === "upload") {
   411→    const f = input.fileMeta;
   412→    if (!f || !f.name?.trim()) return { ok: false as const, status: 422, code: "invalid_file", message: "Choose a file to upload." };
   413→    if (f.sizeKb > MAX_TOTAL_KB) return { ok: false as const, status: 422, code: "file_too_large", message: "Files above 10 MB must go through the enterprise storage vault." };
   414→    const doc: WorkspaceDoc = {
   415→      id: `doc-${Math.floor(Date.now() / 1000) % 100000}`,
   416→      name: f.name,
   417→      type: (f.type?.toUpperCase() || "DOCX") as WorkspaceDoc["type"],
   418→      category: "Personal",
   419→      ownerId: SESSION_USER_ID,
   420→      sizeKb: f.sizeKb,
   421→      modified: Date.now(),
   422→      sensitivity: "Internal",
   423→      description: "Uploaded from My Documents. File content remains in your enterprise storage; Qubit One holds the record.",
   424→    };
   425→    s.documents.unshift(doc);
   426→    writeAudit({ actorId: SESSION_USER_ID, action: "Document uploaded", resource: doc.name, result: "Success", context: "My Documents · metadata record only" });
   427→    return { ok: true as const, doc };
   428→  }
   429→
   430→  const doc = s.documents.find((d) => d.id === input.docId);
   431→  if (!doc) return { ok: false as const, status: 404, code: "not_found", message: "Document not found." };
   432→
   433→  if (SENSITIVE_DOC_ACTIONS.has(input.action) && doc.sensitivity === "Restricted") {
   434→    writeAudit({ actorId: SESSION_USER_ID, action: `Document ${input.action} denied`, resource: doc.name, result: "Blocked", severity: "Warning", context: "Restricted document — HR-managed access only" });
   435→    return { ok: false as const, status: 403, code: "restricted", message: "This document is restricted. Access is managed by HR — request access from the document owner." };
   436→  }
   437→
   438→  // Phase 21 (audit L7): Confidential download/share was previously only
   439→  // gated for the employee role — the same document was exposed to every
   440→  // other non-owner role. The matrix is now symmetric with Restricted:
   441→  // only the document owner, an administrator or a department head may
   442→  // download or share a Confidential document; everyone else gets the
   443→  // same structured 403 with an audited denial. Restricted behavior above
   444→  // is unchanged (refused for all non-HR scope).
   445→  if (
   446→    SENSITIVE_DOC_ACTIONS.has(input.action) &&
   447→    doc.sensitivity === "Confidential" &&
   448→    doc.ownerId !== SESSION_USER_ID &&
   449→    role !== "administrator" &&
   450→    role !== "dept-head"
   451→  ) {
   452→    writeAudit({ actorId: SESSION_USER_ID, action: `Document ${input.action} denied`, resource: doc.name, result: "Blocked", severity: "Warning", context: "Confidential document — only the owner, an administrator or a department head may download or share it" });
   453→    return { ok: false as const, status: 403, code: "confidential", message: "Confidential documents can be downloaded or shared only by the document owner or an administrator — request access from the owner." };
   454→  }
   455→
   456→  if (input.action === "download") {
   457→    writeAudit({ actorId: SESSION_USER_ID, action: "Document downloaded", resource: doc.name, result: "Success", context: `Sensitivity: ${doc.sensitivity}` });
   458→    return { ok: true as const, doc };
   459→  }
   460→
   461→  if (input.action === "share") {
   462→    if (!input.toEmployeeId || !employees.some((e) => e.id === input.toEmployeeId)) {
   463→      return { ok: false as const, status: 422, code: "invalid_recipient", message: "Choose a colleague to share with." };
   464→    }
   465→    writeAudit({ actorId: SESSION_USER_ID, action: "Document shared", resource: doc.name, result: "Success", context: `Shared with ${employees.find((e) => e.id === input.toEmployeeId)?.name}` });
   466→    return { ok: true as const, doc };
   467→  }
   468→
   469→  if (doc.ownerId !== SESSION_USER_ID) {
   470→    return { ok: false as const, status: 403, code: "forbidden", message: "Only the owner can rename or move this document." };
   471→  }
   472→  if (input.action === "rename") {
   473→    const name = input.name?.trim();
   474→    if (!name || name.length < 3) return { ok: false as const, status: 422, code: "invalid_name", message: "Enter a document name (3+ characters)." };
   475→    doc.name = name;
   476→  } else if (input.action === "move") {
   477→    const cat = input.category as WorkspaceDoc["category"];
   478→    if (!["Personal", "Shared with me", "HR", "Company", "Workflow"].includes(cat)) {
   479→      return { ok: false as const, status: 422, code: "invalid_category", message: "Choose a valid category." };
   480→    }
   481→    doc.category = cat;
   482→  }
   483→  return { ok: true as const, doc };
   484→}

   527→// for every role preview. Roles with governance authority

   529→// organize; that permission is enforced here, never in the UI.

   531→// Minutes flow: notes and action items are recorded AGAINST the
   532→// store's session maps (meetingNotes / meetingActions, keyed by
   533→// meeting id — mirroring mobile-server's meetingNotes/meetingActions
   534→// concept) for COMPLETED meetings, and only by the organizer.
   535→// materializeMeeting() merges the fixture record with those session
   536→// entries at read time, so every workspace reader — list, ?id=
   537→// detail, cancel response and the calendar drawer — sees the same
   538→// state without the store ever mutating the fixture-shaped record.
   539→//
   540→// Phase 20 completion of the chain:
   541→//   · createMeeting — full meeting creation with participants; every
   542→//     invited participant gets a notification in the SAME store that
   543→//     seeds the notification center (no second notification system).
   544→//   · cancelMeeting — now actually notifies participants (the message
   545→//     always claimed it; the store makes it true).
   546→//   · addMeetingAction(asTask) / promoteMeetingAction — meeting action
   547→//     items become tasks in the ONE centralized task store via the
   548→//     existing createTask, and promotion is tracked so an item can
   549→//     never become two tasks.
   550→// ============================================================

   552→/** Session display name (demo identity is fixed). */
   553→const sessionName = employees.find((e) => e.id === SESSION_USER_ID)?.name ?? SESSION_USER_ID;
   554→
   555→/** Fixture record + session minutes merged into the wire shape.
   556→ *  Phase 20: also stamps `promoted` onto action items from the
   557→ *  meetingActionsPromoted map so detail readers can hide the
   558→ *  "Add to My Tasks" affordance for items already promoted. */
   559→function materializeMeeting(s: WorkspaceStore, m: Meeting): Meeting {
   560→  const notes = s.meetingNotes.get(m.id) ?? [];
   561→  const actions = s.meetingActions.get(m.id) ?? [];
   562→  const promoted = s.meetingActionsPromoted.get(m.id);
   563→  if (notes.length === 0 && actions.length === 0 && (!promoted || promoted.size === 0)) return m;
   564→  const all: NonNullable<Meeting["actionItems"]> = [
   565→    ...(m.actionItems ?? []).map((a) => ({ ...a })),
   566→    ...actions.map((a) => ({ title: a.title, ownerId: a.ownerId, due: a.due })),
   567→  ];
   568→  for (const i of promoted ?? []) {
   569→    if (all[i]) all[i] = { ...all[i], promoted: true };
   570→  }
   571→  return {
   572→    ...m,
   573→    notes: notes.length ? [m.notes, ...notes.map((n) => n.text)].filter(Boolean).join("\n\n") : m.notes,
   574→    actionItems: all.length ? all : m.actionItems,
   575→  };
   576→}

   613→/**
   614→ * Phase 20 — participant notifications. Appends one WorkspaceNotification
   615→ * per participant EXCLUDING the acting session user, into the SAME store
   616→ * that seeds the notification center (single notification system for
   617→ * desktop and mobile — no second engine). Idempotent by id:
   618→ * ntf-mtg-<meetingId>-<kind>-<participantId>.
   619→ * ts uses the fixed demo clock (WORKSPACE_NOW) so dynamic notifications
   620→ * render "just now" against the workspace timeline every other surface
   621→ * anchors to — consistent with relTime's contract in workspace-data.

   623→function notifyMeetingParticipants(meeting: Meeting, kind: "created" | "cancelled", actorName: string): number {
   624→  const s = store();
   625→  let notified = 0;
   626→  for (const pid of meeting.participantIds) {
   627→    if (pid === SESSION_USER_ID) continue; // the acting organizer/manager never notifies themselves
   628→    const id = `ntf-mtg-${meeting.id}-${kind}-${pid}`;
   629→    if (s.notifications.some((n) => n.id === id)) continue; // idempotent
   630→    const when = `${fmtDate(meeting.start)} · ${fmtTime(meeting.start)}–${fmtTime(meeting.end)}`;
   631→    const notification: WorkspaceNotification = {
   632→      id,
   633→      recipientId: pid, // Phase 21 (M3): addressed to THIS participant only
   634→      title: kind === "created" ? `Meeting invite — "${meeting.title}"` : `"${meeting.title}" was cancelled`,
   635→      body:
   636→        kind === "created"
   637→          ? `${actorName} invited you. ${when} · ${meeting.platform}${meeting.location ? ` · ${meeting.location}` : ""}.`
   638→          : `${actorName} cancelled the ${when} slot (${meeting.platform}). Your calendar is updated.`,
   639→      category: "Meetings",
   640→      ts: WORKSPACE_NOW,
   641→      unread: true,
   642→      priority: kind === "cancelled" ? "High" : "Medium",
   643→      source: "Meetings",
   644→      link: { screen: "my-meetings", params: { meetingId: meeting.id } },
   645→    };
   646→    s.notifications.push(notification);
   647→    notified += 1;
   648→  }
   649→  return notified;
   650→}

   685→export function cancelMeeting(role: WorkspaceRole, input: { id: string; reason: string }) {
   686→  const s = store();
   687→  const m = s.meetings.find((x) => x.id === input.id);
   688→  if (!m) return { ok: false as const, status: 404, code: "not_found", message: "Meeting not found." };
   689→  if (m.status !== "Scheduled") {
   690→    return {
   691→      ok: false as const,
   692→      status: 409,
   693→      code: "invalid_state",
   694→      message: m.status === "Cancelled" ? "This meeting was already cancelled." : `A meeting that is ${m.status.toLowerCase()} cannot be cancelled.`,
   695→    };
   696→  }
   697→  const isOrganizer = m.organizerId === SESSION_USER_ID;
   698→  const governance = role === "administrator" || role === "dept-head";
   699→  if (!isOrganizer && !governance) {
   700→    writeAudit({ actorId: SESSION_USER_ID, action: "Meeting cancellation denied", resource: `${m.title} (${m.id})`, result: "Blocked", severity: "Warning", source: "Workspace", context: "Only the organizer — or an administrator / department head — can cancel a meeting" });
   701→    return { ok: false as const, status: 403, code: "forbidden", message: "Only the meeting organizer — or an administrator / department head — can cancel this meeting." };
   702→  }
   703→  const reason = input.reason?.trim() ?? "";
   704→  if (reason.length < 4) {
   705→    return { ok: false as const, status: 422, code: "reason_required", message: "A short reason is required when cancelling a meeting." };
   706→  }
   707→  m.status = "Cancelled";
   708→  // Phase 20 §5 — a cancelled meeting leaves the Calendar: remove its linked
   709→  // event(s) from the same store so the two surfaces never disagree.
   710→  const before = s.events.length;
   711→  s.events = s.events.filter((e) => e.meetingId !== m.id);
   712→  const removedEvents = before - s.events.length;
   713→  const notified = notifyMeetingParticipants(m, "cancelled", sessionName);
   714→  writeAudit({
   715→    actorId: SESSION_USER_ID,
   716→    action: "Meeting cancelled",
   717→    resource: `${m.id} — ${m.title}`,
   718→    result: "Success",
   719→    severity: "Info",
   720→    source: "Workspace",
   721→    context: `${isOrganizer ? "Organizer" : "Governance"} cancellation — reason: ${reason} · ${notified} participant${notified === 1 ? "" : "s"} notified · ${removedEvents} calendar event${removedEvents === 1 ? "" : "s"} removed`,
   722→  });
   723→  return {
   724→    ok: true as const,
   725→    meeting: materializeMeeting(s, m),
   726→    message: `"${m.title}" was cancelled — ${notified} participant${notified === 1 ? "" : "s"} notified and the calendar updated.`,
   727→  };
   728→}

   756→export function addMeetingAction(role: WorkspaceRole, input: { id: string; title: string; ownerId?: string; due?: number | null; /** Phase 20: also create a task in the ONE task store */ asTask?: boolean }) {
   757→  const s = store();
   758→  const m = s.meetings.find((x) => x.id === input.id);
   759→  if (!m) return { ok: false as const, status: 404, code: "not_found", message: "Meeting not found." };
   760→  if (m.organizerId !== SESSION_USER_ID) {
   761→    writeAudit({ actorId: SESSION_USER_ID, action: "Meeting action item denied", resource: `${m.id} — ${m.title}`, result: "Blocked", severity: "Warning", source: "Workspace", context: "Only the organizer can record action items" });
   762→    return { ok: false as const, status: 403, code: "forbidden", message: "Only the meeting organizer can add action items." };
   763→  }
   764→  if (m.status !== "Completed") {
   765→    return { ok: false as const, status: 409, code: "invalid_state", message: `Action items are recorded on completed meetings — this one is ${m.status.toLowerCase()}.` };
   766→  }
   767→  const title = input.title?.trim() ?? "";
   768→  if (title.length < 4) return { ok: false as const, status: 422, code: "invalid_title", message: "Describe the action item (4+ characters)." };
   769→  const ownerId = input.ownerId ?? SESSION_USER_ID;
   770→  if (!employees.some((e) => e.id === ownerId)) {
   771→    return { ok: false as const, status: 422, code: "invalid_owner", message: "Choose a valid owner." };
   772→  }
   773→  if (input.due !== null && input.due !== undefined && Number.isNaN(input.due)) {
   774→    return { ok: false as const, status: 422, code: "invalid_due", message: "Due date is invalid." };
   775→  }
   776→
   777→  // Phase 20 — asTask: create the task FIRST (via the existing createTask,
   778→  // the ONE centralized task store) so a failed task creation leaves the
   779→  // meeting record untouched. The item's index in the MATERIALIZED array
   780→  // is fixture items first, session items after — compute it pre-insert.
   781→  let taskId: string | undefined;
   782→  if (input.asTask) {
   783→    const taskRes = createTask(role, {
   784→      title,
   785→      detail: `Meeting action item from ${m.id} — "${m.title}".`,
   786→      due: input.due ?? null,
   787→      source: "Meeting action item",
   788→      related: { label: m.title, kind: "Meeting", ref: m.id },
   789→    });
   790→    if (!taskRes.ok) return taskRes;
   791→    taskId = taskRes.task.id;
   792→  }
   793→
   794→  const existing = s.meetingActions.get(m.id) ?? [];
   795→  const index = (m.actionItems?.length ?? 0) + existing.length;
   796→  s.meetingActions.set(m.id, [...existing, { title, ownerId, due: input.due ?? null, at: Date.now() }]);
   797→  if (taskId) {
   798→    const promoted = s.meetingActionsPromoted.get(m.id) ?? new Set<number>();
   799→    promoted.add(index);
   800→    s.meetingActionsPromoted.set(m.id, promoted);
   801→  }
   802→
   803→  // ONE audit event carries both facts when a task is also created.
   804→  if (taskId) {
   805→    writeAudit({ actorId: SESSION_USER_ID, action: "Meeting action item recorded", resource: `${m.id} — ${title}`, result: "Success", source: "Workspace", context: `Owner: ${employees.find((e) => e.id === ownerId)?.name ?? ownerId} · task created (${taskId})` });
   806→    return { ok: true as const, meeting: materializeMeeting(s, m), taskId, message: "Action item recorded on the meeting — task added to My Tasks." };
   807→  }
   808→  writeAudit({ actorId: SESSION_USER_ID, action: "Meeting action item added", resource: `${m.id} — ${title}`, result: "Success", source: "Workspace", context: `Owner: ${employees.find((e) => e.id === ownerId)?.name ?? ownerId}` });
   809→  return { ok: true as const, meeting: materializeMeeting(s, m), message: "Action item recorded on the meeting." };
   810→}

   812→/**
   813→ * Phase 20 — promote an already-recorded action item (by index in the
   814→ * MATERIALIZED actionItems array) into a task via the existing createTask.
   815→ * The organizer OR any participant may promote into their own task queue
   816→ * (the task is always created for the session user — "Add to My Tasks");
   817→ * promotion is tracked per meeting/index so an item can never become two
   818→ * tasks (409 on repeat).
   819→ */
   820→export function promoteMeetingAction(role: WorkspaceRole, input: { id: string; index: number }) {
   821→  const s = store();
   822→  const m = s.meetings.find((x) => x.id === input.id);
   823→  if (!m) return { ok: false as const, status: 404, code: "not_found", message: "Meeting not found." };
   824→  const participating = m.organizerId === SESSION_USER_ID || m.participantIds.includes(SESSION_USER_ID);
   825→  if (!participating) {
   826→    writeAudit({ actorId: SESSION_USER_ID, action: "Meeting action promotion denied", resource: `${m.id} — ${m.title}`, result: "Blocked", severity: "Warning", source: "Workspace", context: "Caller is neither the organizer nor a participant" });
   827→    return { ok: false as const, status: 403, code: "forbidden", message: "You are not a participant of this meeting." };
   828→  }
   829→  const item = materializeMeeting(s, m).actionItems?.[input.index];
   830→  if (!item) return { ok: false as const, status: 404, code: "not_found", message: "Action item not found." };
   831→  const promoted = s.meetingActionsPromoted.get(m.id) ?? new Set<number>();
   832→  if (promoted.has(input.index)) {
   833→    return { ok: false as const, status: 409, code: "already_promoted", message: `"${item.title}" is already in My Tasks.` };
   834→  }
   835→  const taskRes = createTask(role, {
   836→    title: item.title,
   837→    detail: `Promoted from meeting action item — ${m.id} "${m.title}".`,
   838→    due: item.due,
   839→    source: "Meeting action item",
   840→    related: { label: m.title, kind: "Meeting", ref: m.id },
   841→  });
   842→  if (!taskRes.ok) return taskRes;
   843→  promoted.add(input.index);
   844→  s.meetingActionsPromoted.set(m.id, promoted);
   845→  writeAudit({ actorId: SESSION_USER_ID, action: "Meeting action item promoted to task", resource: `${m.id} — ${item.title}`, result: "Success", source: "Workspace", context: `Task ${taskRes.task.id} created in My Tasks` });
   846→  return { ok: true as const, task: taskRes.task, message: `"${item.title}" was added to My Tasks.` };
   847→}

   849→/**
   850→ * Phase 20 — full meeting creation with participants. Any role may create
   851→ * their own meeting (the session user is always the organizer and a
   852→ * participant). Every invited participant receives a notification in the
   853→ * SAME store that seeds the notification center, and the creation is
   854→ * audited with participant count and time.
   855→ */
   856→export function createMeeting(role: WorkspaceRole, input: { title: string; agenda?: string; start: number; end: number; location?: string; platform?: string; participantIds: string[] }) {
   857→  const s = store();
   858→  const title = input.title?.trim() ?? "";
   859→  if (!title) return { ok: false as const, status: 422, code: "invalid_title", message: "Give the meeting a title." };
   860→  if (title.length < 4) return { ok: false as const, status: 422, code: "invalid_title", message: "Meeting title must be at least 4 characters." };
   861→  if (title.length > 140) return { ok: false as const, status: 422, code: "invalid_title", message: "Keep the meeting title within 140 characters." };
   862→  if (!Number.isFinite(input.start) || !Number.isFinite(input.end)) {
   863→    return { ok: false as const, status: 400, code: "invalid_time", message: "Start and end must be epoch milliseconds." };
   864→  }
   865→  if (input.end <= input.start) {
   866→    return { ok: false as const, status: 400, code: "invalid_time", message: "The meeting end must be after its start." };
   867→  }
   868→  if (input.location !== undefined && (typeof input.location !== "string" || input.location.trim().length > 120)) {
   869→    return { ok: false as const, status: 422, code: "invalid_location", message: "Location must be 120 characters or fewer." };
   870→  }
   871→  const VALID_PLATFORMS: Meeting["platform"][] = ["Qubit Meet", "Zoom", "Microsoft Teams", "Google Meet", "Room", "In person"];
   872→  if (input.platform !== undefined && !VALID_PLATFORMS.includes(input.platform as Meeting["platform"])) {
   873→    return { ok: false as const, status: 422, code: "invalid_platform", message: "Choose a valid platform (Qubit Meet, Zoom, Microsoft Teams, Google Meet, Room or In person)." };
   874→  }
   875→  if (!Array.isArray(input.participantIds)) {
   876→    return { ok: false as const, status: 400, code: "invalid_participants", message: "participantIds must be a list of employee ids." };
   877→  }
   878→  const cleaned = Array.from(new Set(input.participantIds.map((v) => String(v ?? "").trim()).filter(Boolean)));
   879→  const unknown = cleaned.filter((pid) => !employees.some((e) => e.id === pid));
   880→  if (unknown.length > 0) {
   881→    return { ok: false as const, status: 400, code: "invalid_participants", message: `Unknown participant${unknown.length === 1 ? "" : "s"}: ${unknown.join(", ")}.` };
   882→  }
   883→  const participantIds = Array.from(new Set([SESSION_USER_ID, ...cleaned])); // organizer joins and duplicates collapse
   884→  if (participantIds.length > 12) {
   885→    return { ok: false as const, status: 422, code: "too_many_participants", message: "A meeting can have at most 12 participants." };
   886→  }
   887→
   888→  const platform = (input.platform as Meeting["platform"]) ?? "Qubit Meet";
   889→  const seq = s.meetings.filter((x) => /^mtg-9\d\d$/.test(x.id)).length; // dynamic ids start at mtg-900 — never collides with fixture mtg-3xx
   890→  const meeting: Meeting = {
   891→    id: `mtg-${900 + seq}`,
   892→    title,
   893→    start: input.start,
   894→    end: input.end,
   895→    organizerId: SESSION_USER_ID,
   896→    participantIds,
   897→    location: input.location?.trim() || platform,
   898→    platform,
   899→    status: "Scheduled",
   900→    agenda: input.agenda?.trim() || undefined,
   901→  };
   902→  s.meetings.push(meeting);
   903→  // Phase 20 §5 — the meeting appears in the Calendar (screen 44): a calendar
   904→  // event of type "Meeting" is created in the SAME store the calendar reads,
   905→  // linked by meetingId so detail navigation and cancellation stay consistent.
   906→  const event: CalendarEvent = {
   907→    id: `ev-${900 + seq}`,
   908→    title,
   909→    type: "Meeting",
   910→    start: meeting.start,
   911→    end: meeting.end,
   912→    organizerId: SESSION_USER_ID,
   913→    meetingId: meeting.id,
   914→    location: meeting.location,
   915→  };
   916→  s.events.push(event);
   917→  const notified = notifyMeetingParticipants(meeting, "created", sessionName);
   918→  writeAudit({
   919→    actorId: SESSION_USER_ID,
   920→    action: "Meeting scheduled",
   921→    resource: `${meeting.id} — ${title}`,
   922→    result: "Success",
   923→    severity: "Info",
   924→    source: "Workspace",
   925→    context: `${participantIds.length} participant${participantIds.length === 1 ? "" : "s"} (${notified} notified) · ${fmtDate(meeting.start)} ${fmtTime(meeting.start)}–${fmtTime(meeting.end)} IST`,
   926→  });
   927→  return {
   928→    ok: true as const,
   929→    meeting: materializeMeeting(s, meeting),
   930→    message: `"${title}" scheduled with ${participantIds.length} participant${participantIds.length === 1 ? "" : "s"} — ${notified} notified.`,
   931→  };
   932→}

   934→// Calendar entries created from My Calendar (screen 44) — personal events
   935→// such as focus blocks and deadlines. Meeting invites WITH participants run
   936→// through the Meetings module; this records the entry on the session
   937→// calendar server-side so the screen holds no client-only business state.
   938→export function createCalendarEvent(role: WorkspaceRole, input: { title: string; type: CalendarEvent["type"]; start: number; end: number; location?: string }) {
   939→  const title = input.title?.trim() ?? "";
   940→  if (title.length < 3) return { ok: false as const, status: 422, code: "invalid_title", message: "Give the event a title (at least 3 characters)." };
   941→  const VALID_TYPES: CalendarEvent["type"][] = ["Meeting", "Deadline", "Company event", "Leave", "Focus time"];
   942→  if (!VALID_TYPES.includes(input.type)) {
   943→    return { ok: false as const, status: 422, code: "invalid_type", message: "Choose a valid event type." };
   944→  }
   945→  if (!Number.isFinite(input.start) || !Number.isFinite(input.end) || input.end <= input.start) {
   946→    return { ok: false as const, status: 422, code: "invalid_time", message: "The event end must be after its start." };
   947→  }
   948→  const ev: CalendarEvent = {
   949→    id: `ev-w${Math.floor(Date.now() / 1000) % 100000}`,
   950→    title,
   951→    type: input.type,
   952→    start: input.start,
   953→    end: input.end,
   954→    organizerId: SESSION_USER_ID,
   955→    location: input.location?.trim() || undefined,
   956→  };
   957→  store().events.push(ev);
   958→  writeAudit({ actorId: SESSION_USER_ID, action: "Calendar event created", resource: title, result: "Success", source: "Workspace", context: `${ev.type} added from My Calendar` });
   959→  return { ok: true as const, event: ev, message: `"${title}" was added to your calendar.` };
   960→}
